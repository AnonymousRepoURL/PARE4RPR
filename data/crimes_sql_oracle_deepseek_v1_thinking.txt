WITH day_agg AS (
  SELECT BLOCK,
         PRIMARY_TYPE,
         FLOOR(TS / 86400000) AS DAY,
         COUNT(*) AS CNT_ALL,
         SUM(CASE WHEN TS > 1786118400000 THEN 1 ELSE 0 END) AS CNT_GT,
         MIN(CASE WHEN TS > 1786118400000 THEN TS END) AS MIN_TS_GT,
         MIN(TS) AS MIN_TS_ALL
  FROM CRIMES
  GROUP BY BLOCK, PRIMARY_TYPE, FLOOR(TS / 86400000)
),
day_run AS (
  SELECT BLOCK,
         PRIMARY_TYPE,
         DAY,
         CNT_ALL,
         CNT_GT,
         MIN_TS_GT,
         MIN_TS_ALL,
         DAY - ROW_NUMBER() OVER (PARTITION BY BLOCK, PRIMARY_TYPE ORDER BY DAY) AS GRP
  FROM day_agg
),
run_bounds AS (
  SELECT BLOCK,
         PRIMARY_TYPE,
         GRP,
         MAX(DAY) AS D_MAX,
         MIN(CASE WHEN CNT_GT > 0 THEN DAY END) AS D_START
  FROM day_run
  GROUP BY BLOCK, PRIMARY_TYPE, GRP
)
SELECT b.BLOCK,
       b.PRIMARY_TYPE,
       MAX(CASE WHEN r.DAY = b.D_START THEN r.MIN_TS_GT END) AS FIRST_DAY_TS,
       MAX(CASE WHEN r.DAY = b.D_MAX THEN r.MIN_TS_ALL END) AS LAST_DAY_TS,
       SUM(CASE WHEN r.DAY = b.D_START THEN r.CNT_GT ELSE r.CNT_ALL END) AS INCIDENT_COUNT
FROM run_bounds b
JOIN day_run r
  ON r.BLOCK = b.BLOCK
 AND r.PRIMARY_TYPE = b.PRIMARY_TYPE
 AND r.GRP = b.GRP
WHERE b.D_START IS NOT NULL
  AND b.D_MAX - b.D_START + 1 >= 3
  AND r.DAY >= b.D_START
GROUP BY b.BLOCK, b.PRIMARY_TYPE, b.GRP, b.D_START, b.D_MAX;

        ELSE 0
      END AS new_run

WITH base AS (
  SELECT 
    DISTRICT,
    ID,
    TS,
    PRIMARY_TYPE,
    LOCATION_DESCRIPTION,
    ARREST,
    ROW_NUMBER() OVER (PARTITION BY DISTRICT ORDER BY TS, ID) AS rn,
    CASE 
      WHEN PRIMARY_TYPE = 'BATTERY' AND LOCATION_DESCRIPTION = 'STREET' AND ARREST = 'false' THEN 'REPORT'
      WHEN PRIMARY_TYPE = 'CRIMINAL TRESPASS' THEN 'THEFT'
      WHEN PRIMARY_TYPE = 'THEFT' AND ARREST = 'true' THEN 'DAMAGE'
      WHEN PRIMARY_TYPE = 'OTHER OFFENSE' AND LOCATION_DESCRIPTION = 'RESIDENCE' AND ARREST = 'false' THEN 'ARRESTED'
      ELSE 'OTHER'
    END AS cat
  FROM CRIMES
),
grouped AS (
  SELECT 
    b.*,
    COUNT(CASE WHEN cat NOT IN ('THEFT','DAMAGE') THEN 1 END) OVER (PARTITION BY DISTRICT ORDER BY rn ROWS UNBOUNDED PRECEDING) AS grp
  FROM base b
),
enriched AS (
  SELECT 
    g.*,
    LAST_VALUE(CASE WHEN g.cat = 'THEFT' THEN g.ID END IGNORE NULLS) 
      OVER (PARTITION BY g.DISTRICT, g.grp ORDER BY g.rn ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING) AS last_theft_id,
    LAST_VALUE(CASE WHEN g.cat = 'DAMAGE' THEN g.ID END IGNORE NULLS) 
      OVER (PARTITION BY g.DISTRICT, g.grp ORDER BY g.rn ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING) AS last_damage_id
  FROM grouped g
)
SELECT 
  r.DISTRICT,
  r.ID AS REPORT_ID,
  r.last_theft_id AS LAST_THEFT_ID,
  r.last_damage_id AS LAST_DAMAGE_ID,
  s.ID AS ARRESTED_ID
FROM enriched r
JOIN enriched s 
  ON (s.DISTRICT = r.DISTRICT OR (s.DISTRICT IS NULL AND r.DISTRICT IS NULL))
 AND s.grp = r.grp + 1 
 AND s.cat = 'ARRESTED'
WHERE r.cat = 'REPORT'
  AND s.rn > r.rn + 1;

WITH base AS (
  SELECT ID, DISTRICT, TS, PRIMARY_TYPE, LATITUDE
  FROM CRIMES
  WHERE PRIMARY_TYPE IN ('THEFT', 'BATTERY', 'ROBBERY')
),
l1 AS (
  SELECT ID AS L1_ID, DISTRICT, TS AS L1_TS, LATITUDE AS L1_LAT
  FROM base
  WHERE PRIMARY_TYPE = 'THEFT'
),
l3 AS (
  SELECT 
    l1.L1_ID,
    l1.DISTRICT AS L1_DISTRICT,
    l1.L1_TS,
    l1.L1_LAT,
    MAX(r.ID) KEEP (DENSE_RANK LAST ORDER BY r.TS, r.ID) AS L3_ID,
    MAX(r.TS) KEEP (DENSE_RANK LAST ORDER BY r.TS, r.ID) AS L3_TS
  FROM l1
  JOIN base r ON (r.DISTRICT = l1.DISTRICT OR (r.DISTRICT IS NULL AND l1.DISTRICT IS NULL))
    AND r.PRIMARY_TYPE = 'ROBBERY'
    AND (r.TS > l1.L1_TS OR (r.TS = l1.L1_TS AND r.ID > l1.L1_ID))
    AND r.TS - l1.L1_TS <= 3600000
    AND r.LATITUDE > l1.L1_LAT
  GROUP BY l1.L1_ID, l1.DISTRICT, l1.L1_TS, l1.L1_LAT
),
l2 AS (
  SELECT 
    l3.L1_ID,
    l3.L3_ID,
    l3.L1_DISTRICT,
    l3.L1_TS,
    l3.L1_LAT,
    l3.L3_TS,
    MAX(l2.ID) KEEP (DENSE_RANK LAST ORDER BY l2.TS, l2.ID) AS L2_ID
  FROM l3
  JOIN base l2 ON (l2.DISTRICT = l3.L1_DISTRICT OR (l2.DISTRICT IS NULL AND l3.L1_DISTRICT IS NULL))
    AND l2.PRIMARY_TYPE = 'BATTERY'
    AND (l2.TS > l3.L1_TS OR (l2.TS = l3.L1_TS AND l2.ID > l3.L1_ID))
    AND (l2.TS < l3.L3_TS OR (l2.TS = l3.L3_TS AND l2.ID < l3.L3_ID))

  GROUP BY l3.L1_ID, l3.L3_ID, l3.L1_DISTRICT, l3.L1_TS, l3.L1_LAT, l3.L3_TS
)

WITH B_C AS (
  SELECT B.ID AS B_ID,
         B.DISTRICT AS B_DISTRICT,
         B.TS AS B_TS,
         C.ID AS C_ID,
         C.TS AS C_TS,
         ROW_NUMBER() OVER (PARTITION BY B.ID ORDER BY C.TS, C.ID) AS rn
  FROM CRIMES B
  JOIN CRIMES C
    ON C.PRIMARY_TYPE = 'MOTOR VEHICLE THEFT'
   AND C.LONGITUDE BETWEEN B.LONGITUDE - 0.1 AND B.LONGITUDE + 0.1
   AND (C.TS > B.TS OR (C.TS = B.TS AND C.ID > B.ID))
  WHERE B.PRIMARY_TYPE = 'BATTERY'
),
B_with_C AS (
  SELECT B_ID, B_DISTRICT, B_TS, C_ID, C_TS
  FROM B_C
  WHERE rn = 1
),
A_B_candidates AS (
  SELECT A.ID AS A_ID,
         A.TS AS A_TS,
         B.B_ID,
         B.B_TS,
         B.C_ID,
         B.C_TS,
         ROW_NUMBER() OVER (PARTITION BY A.ID ORDER BY B.B_TS, B.B_ID) AS rn
  FROM CRIMES A
  JOIN B_with_C B
    ON B.B_DISTRICT = A.DISTRICT
   AND (B.B_TS > A.TS OR (B.B_TS = A.TS AND B.B_ID > A.ID))
   AND B.C_TS - A.TS <= 1800000
  WHERE A.PRIMARY_TYPE = 'ROBBERY'
),
matches AS (
  SELECT A_ID, B_ID, C_ID
  FROM A_B_candidates
  WHERE rn = 1
)
SELECT A.*
FROM matches m
JOIN CRIMES A ON m.A_ID = A.ID
UNION ALL
SELECT B.*
FROM matches m
JOIN CRIMES B ON m.B_ID = B.ID
UNION ALL
SELECT C.*
FROM matches m
JOIN CRIMES C ON m.C_ID = C.ID;

WITH base AS (
  SELECT 
    COMMUNITY_AREA,
    TS,
    ID,
    CASE WHEN PRIMARY_TYPE = 'ASSAULT' AND WARD = 48 AND ARREST = 'false' THEN 1 ELSE 0 END AS is_E,
    CASE WHEN PRIMARY_TYPE = 'CRIMINAL TRESPASS' AND WARD = 1 THEN 1 ELSE 0 END AS is_C,
    ROW_NUMBER() OVER (PARTITION BY COMMUNITY_AREA ORDER BY TS, ID) AS rn
  FROM CRIMES
),
flags AS (
  SELECT 
    b.*,
    NVL(LAG(is_E, 1, 0) OVER (PARTITION BY COMMUNITY_AREA ORDER BY rn), 0) AS prev_E,
    LEAD(is_E, 1, 0) OVER (PARTITION BY COMMUNITY_AREA ORDER BY rn) AS next_E,
    LEAD(is_C, 1, 0) OVER (PARTITION BY COMMUNITY_AREA ORDER BY rn) AS next_C,
    LEAD(TS, 1, NULL) OVER (PARTITION BY COMMUNITY_AREA ORDER BY rn) AS next_TS
  FROM base b
),
runs AS (
  SELECT 
    f.*,
    CASE WHEN is_E = 1 AND prev_E = 0 THEN 1 ELSE 0 END AS start_flag
  FROM flags f
),
grouped AS (
  SELECT 
    r.*,
    SUM(start_flag) OVER (PARTITION BY COMMUNITY_AREA ORDER BY rn ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS grp_id
  FROM runs r
),
run_stats AS (
  SELECT 
    g.*,
    COUNT(CASE WHEN is_E = 1 THEN 1 END) OVER (PARTITION BY COMMUNITY_AREA, grp_id) AS run_len,
    MIN(CASE WHEN is_E = 1 THEN TS END) OVER (PARTITION BY COMMUNITY_AREA, grp_id) AS start_ts
  FROM grouped g
)
SELECT 
  COMMUNITY_AREA,
  start_ts AS START_TS,
  next_TS AS END_TS,
  run_len AS ESCALATING_COUNT
FROM run_stats
WHERE is_E = 1 
  AND next_E = 0 
  AND next_C = 1 
  AND run_len >= 3;

  FROM CRIMES
  WHERE (PRIMARY_TYPE = 'BURGLARY' AND DISTRICT = '008' AND DOMESTIC = 'false')
     OR (PRIMARY_TYPE = 'OFFENSE INVOLVING CHILDREN' AND DOMESTIC = 'true')
),
numbered AS (
  SELECT
    f.*,
    ROW_NUMBER() OVER (PARTITION BY COMMUNITY_AREA ORDER BY TS, ID) AS rn,
    ROW_NUMBER() OVER (PARTITION BY COMMUNITY_AREA, is_e ORDER BY TS, ID) AS rn_e,
    LEAD(is_c) OVER (PARTITION BY COMMUNITY_AREA ORDER BY TS, ID) AS next_is_c,
    LEAD(TS) OVER (PARTITION BY COMMUNITY_AREA ORDER BY TS, ID) AS next_ts
  FROM filtered f
),
escalating_groups AS (
  SELECT
    n.*,
    rn - rn_e AS group_id
  FROM numbered n
  WHERE is_e = 1
),
aggregated AS (
  SELECT
    COMMUNITY_AREA,
    group_id,
    COUNT(*) AS ESCALATING_COUNT,
    MIN(TS) AS START_TS,
    MAX(next_is_c) AS followed_by_c,
    MAX(next_ts) AS END_TS
  FROM escalating_groups
  GROUP BY COMMUNITY_AREA, group_id
)
SELECT
  COMMUNITY_AREA,
  START_TS,
  END_TS,
  ESCALATING_COUNT
FROM aggregated
WHERE ESCALATING_COUNT >= 3
  AND followed_by_c = 1;

WITH marked AS (
  SELECT
    COMMUNITY_AREA,
    TS,
    ID,
    CASE WHEN PRIMARY_TYPE = 'NARCOTICS' AND ARREST = 'true' THEN 1 ELSE 0 END AS is_e,
    CASE WHEN PRIMARY_TYPE = 'OTHER OFFENSE' AND LOCATION_DESCRIPTION = 'APARTMENT' THEN 1 ELSE 0 END AS is_c
  FROM CRIMES
),
computed AS (
  SELECT
    COMMUNITY_AREA,
    TS,
    ID,
    is_e,
    is_c,
    SUM(1 - is_e) OVER (
      PARTITION BY COMMUNITY_AREA
      ORDER BY TS, ID
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS grp
  FROM marked
),
runs AS (
  SELECT
    COMMUNITY_AREA,
    TS,
    ID,
    is_e,
    is_c,
    grp,
    COUNT(CASE WHEN is_e = 1 THEN 1 END) OVER (
      PARTITION BY COMMUNITY_AREA, grp
    ) AS run_length,
    MIN(CASE WHEN is_e = 1 THEN TS END) OVER (
      PARTITION BY COMMUNITY_AREA, grp
    ) AS run_start_ts
  FROM computed
)
SELECT
  COMMUNITY_AREA,
  START_TS,
  END_TS,
  ESCALATING_COUNT
FROM (
  SELECT
    COMMUNITY_AREA,
    TS AS END_TS,
    is_c,
    LAG(run_start_ts) OVER (
      PARTITION BY COMMUNITY_AREA
      ORDER BY TS, ID
    ) AS START_TS,
    LAG(run_length) OVER (
      PARTITION BY COMMUNITY_AREA
      ORDER BY TS, ID
    ) AS ESCALATING_COUNT
  FROM runs
)
WHERE is_c = 1
  AND ESCALATING_COUNT >= 3;

WITH base AS (
  SELECT ID, BEAT, TS,
         CASE
           WHEN PRIMARY_TYPE = 'ARSON' AND LOCATION_DESCRIPTION = 'RESIDENCE' AND ARREST = 'false' THEN 'B'
           WHEN PRIMARY_TYPE = 'PUBLIC PEACE VIOLATION' AND DISTRICT = '005' THEN 'S'
           WHEN PRIMARY_TYPE = 'BURGLARY' AND COMMUNITY_AREA = 13 AND ARREST = 'false' THEN 'U'
           WHEN YEAR = 2001 AND PRIMARY_TYPE = 'BATTERY' AND DISTRICT = '016' THEN 'R'
           ELSE 'O'
         END AS typ,
         ROW_NUMBER() OVER (PARTITION BY BEAT ORDER BY TS, ID) AS rn
  FROM CRIMES
),
runs AS (
  SELECT b.*,
         b.rn - ROW_NUMBER() OVER (PARTITION BY b.BEAT, b.typ ORDER BY b.rn) AS grp
  FROM base b
),
valid_spikes AS (
  SELECT r.BEAT, r.rn AS s_rn, r.TS AS s_ts
  FROM (
    SELECT r.*,
           LAG(r.typ) OVER (PARTITION BY r.BEAT ORDER BY r.rn) AS prev_typ,
           LEAD(r.typ) OVER (PARTITION BY r.BEAT ORDER BY r.rn) AS next_typ
    FROM runs r
  ) r
  WHERE r.typ = 'S' AND r.prev_typ = 'B' AND r.next_typ = 'U'
),
u_run_info AS (
  SELECT r.BEAT, r.rn, r.grp,
         MIN(r.rn) OVER (PARTITION BY r.BEAT, r.grp) AS run_start,
         MAX(r.rn) OVER (PARTITION BY r.BEAT, r.grp) AS run_end,
         COUNT(*) OVER (PARTITION BY r.BEAT, r.grp) AS run_count
  FROM runs r
  WHERE r.typ = 'U'
),
valid_sequences AS (
  SELECT vs.BEAT, vs.s_rn, vs.s_ts,
         u.run_start AS u_start,
         u.run_end AS u_end,
         u.run_count AS u_count,
         r.rn AS r_rn,
         r.TS AS r_ts
  FROM valid_spikes vs
  JOIN u_run_info u ON vs.BEAT = u.BEAT AND u.rn = vs.s_rn + 1
  JOIN runs r ON r.BEAT = u.BEAT AND r.rn = u.run_end + 1 AND r.typ = 'R'
),
b_run_info AS (
  SELECT r.BEAT, r.rn, r.grp,
         MIN(r.rn) OVER (PARTITION BY r.BEAT, r.grp) AS run_start,
         MAX(r.rn) OVER (PARTITION BY r.BEAT, r.grp) AS run_end
  FROM runs r
  WHERE r.typ = 'B'
)
SELECT b_rows.TS AS START_TS, vs.r_ts AS END_TS, vs.u_count AS SUSTAINED_COUNT
FROM valid_sequences vs
JOIN b_run_info b ON vs.BEAT = b.BEAT AND b.rn = vs.s_rn - 1
JOIN runs b_rows ON b_rows.BEAT = b.BEAT AND b_rows.grp = b.grp AND b_rows.typ = 'B';

WITH Ordered AS (
  SELECT 
    ID, TS, PRIMARY_TYPE, LATITUDE, LONGITUDE,
    ROW_NUMBER() OVER (ORDER BY TS, ID) AS rn,
    CASE WHEN PRIMARY_TYPE = 'ROBBERY' AND LATITUDE BETWEEN 41.6994435912654 AND 41.7146453168236 AND LONGITUDE BETWEEN -87.6587416716118 AND -87.6330873595214 THEN 1 ELSE 0 END AS is_A,
    CASE WHEN PRIMARY_TYPE = 'BATTERY' AND LATITUDE BETWEEN 41.6994435912654 AND 41.7146453168236 AND LONGITUDE BETWEEN -87.6587416716118 AND -87.6330873595214 THEN 1 ELSE 0 END AS is_B,
    CASE WHEN PRIMARY_TYPE = 'MOTOR VEHICLE THEFT' AND LATITUDE BETWEEN 41.6994435912654 AND 41.7146453168236 AND LONGITUDE BETWEEN -87.6587416716118 AND -87.6330873595214 THEN 1 ELSE 0 END AS is_C
  FROM CRIMES
),
Metrics AS (
  SELECT 
    rn, ID, TS, is_A, is_B, is_C,
    MIN(CASE WHEN is_B = 1 THEN rn END) OVER (ORDER BY rn ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING) AS next_B,
    MAX(CASE WHEN is_C = 1 THEN rn END) OVER (ORDER BY rn ROWS BETWEEN 1 FOLLOWING AND 49 FOLLOWING) AS max_C_49,
    MAX(CASE WHEN is_B = 1 THEN rn END) OVER (ORDER BY rn ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) AS prev_B
  FROM Ordered
)
SELECT 
  A.TS AS TS,
  A.ID AS ID,
  A.ID AS ROBBERY_ID,
  B.ID AS BATTERY_ID,
  C.ID AS MOTOR_VEHICLE_THEFT_ID
FROM Metrics A
JOIN Metrics C ON C.rn = A.max_C_49
JOIN Metrics B ON B.rn = C.prev_B
WHERE A.is_A = 1
  AND A.next_B IS NOT NULL
  AND A.max_C_49 IS NOT NULL
  AND A.max_C_49 > A.next_B
ORDER BY A.rn;

WITH ordered_crimes AS (
  SELECT
    BEAT,
    ID,
    PRIMARY_TYPE,
    LEAD(ID, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS NEXT_ID,
    LEAD(PRIMARY_TYPE, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS NEXT_TYPE,
    LEAD(ID, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS NEXT2_ID,
    LEAD(PRIMARY_TYPE, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS NEXT2_TYPE
  FROM CRIMES
)
SELECT
  BEAT,
  CASE
    WHEN PRIMARY_TYPE = 'THEFT' THEN ID
    WHEN NEXT_TYPE = 'THEFT' THEN NEXT_ID
    WHEN NEXT2_TYPE = 'THEFT' THEN NEXT2_ID
  END AS THEFT_ID,
  CASE
    WHEN PRIMARY_TYPE = 'BATTERY' THEN ID
    WHEN NEXT_TYPE = 'BATTERY' THEN NEXT_ID
    WHEN NEXT2_TYPE = 'BATTERY' THEN NEXT2_ID
  END AS BATTERY_ID,
  CASE
    WHEN PRIMARY_TYPE = 'CRIMINAL DAMAGE' THEN ID
    WHEN NEXT_TYPE = 'CRIMINAL DAMAGE' THEN NEXT_ID
    WHEN NEXT2_TYPE = 'CRIMINAL DAMAGE' THEN NEXT2_ID
  END AS DAMAGE_ID
FROM ordered_crimes
WHERE
  (PRIMARY_TYPE = 'THEFT' OR NEXT_TYPE = 'THEFT' OR NEXT2_TYPE = 'THEFT')
  AND (PRIMARY_TYPE = 'BATTERY' OR NEXT_TYPE = 'BATTERY' OR NEXT2_TYPE = 'BATTERY')
  AND (PRIMARY_TYPE = 'CRIMINAL DAMAGE' OR NEXT_TYPE = 'CRIMINAL DAMAGE' OR NEXT2_TYPE = 'CRIMINAL DAMAGE')
  AND NEXT2_TYPE IS NOT NULL;

WITH CRIMES_WITH_TYPE AS (
  SELECT
    BEAT,
    TS,
    ID,
    CASE
      WHEN PRIMARY_TYPE = 'ASSAULT' AND COMMUNITY_AREA = 28 AND ARREST = 'true' THEN 'A'
      WHEN YEAR = 2024 AND PRIMARY_TYPE = 'BATTERY' AND DISTRICT = '011' THEN 'B'
      WHEN PRIMARY_TYPE = 'OTHER OFFENSE' AND COMMUNITY_AREA = 33 AND ARREST = 'false' THEN 'C'
      ELSE NULL
    END AS MATCH_TYPE
  FROM CRIMES
),
CRIMES_WITH_LEAD AS (
  SELECT
    BEAT,
    TS,
    ID,
    MATCH_TYPE,
    LEAD(ID, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS N1_ID,
    LEAD(MATCH_TYPE, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS N1_TYPE,
    LEAD(ID, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS N2_ID,
    LEAD(MATCH_TYPE, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS N2_TYPE
  FROM CRIMES_WITH_TYPE
)
SELECT
  BEAT,
  CASE
    WHEN MATCH_TYPE = 'A' THEN ID
    WHEN N1_TYPE = 'A' THEN N1_ID
    WHEN N2_TYPE = 'A' THEN N2_ID
  END AS A_ID,
  CASE
    WHEN MATCH_TYPE = 'B' THEN ID
    WHEN N1_TYPE = 'B' THEN N1_ID
    WHEN N2_TYPE = 'B' THEN N2_ID
  END AS B_ID,
  CASE
    WHEN MATCH_TYPE = 'C' THEN ID
    WHEN N1_TYPE = 'C' THEN N1_ID
    WHEN N2_TYPE = 'C' THEN N2_ID
  END AS C_ID
FROM CRIMES_WITH_LEAD
WHERE MATCH_TYPE || N1_TYPE || N2_TYPE IN ('ABC', 'ACB', 'BAC', 'BCA', 'CAB', 'CBA');

WITH marked AS (

    ID,
    BEAT,
    TS,
    CASE WHEN YEAR = 2004 AND PRIMARY_TYPE = 'ASSAULT' THEN 1 ELSE 0 END AS A_FLAG,
    CASE WHEN PRIMARY_TYPE = 'BURGLARY' AND DISTRICT = '006' THEN 1 ELSE 0 END AS B_FLAG,
    CASE WHEN PRIMARY_TYPE = 'CRIMINAL DAMAGE' AND WARD = 17 AND ARREST = 'false' THEN 1 ELSE 0 END AS C_FLAG
  FROM CRIMES
),
windows AS (
  SELECT
    ID AS ID0,
    LEAD(ID, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS ID1,
    LEAD(ID, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS ID2,
    A_FLAG AS A0,
    LEAD(A_FLAG, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS A1,
    LEAD(A_FLAG, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS A2,
    B_FLAG AS B0,
    LEAD(B_FLAG, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS B1,
    LEAD(B_FLAG, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS B2,
    C_FLAG AS C0,
    LEAD(C_FLAG, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS C1,
    LEAD(C_FLAG, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS C2
  FROM marked
)
SELECT
  CASE WHEN A0 = 1 THEN ID0 WHEN A1 = 1 THEN ID1 ELSE ID2 END AS A_ID,
  CASE WHEN B0 = 1 THEN ID0 WHEN B1 = 1 THEN ID1 ELSE ID2 END AS B_ID,
  CASE WHEN C0 = 1 THEN ID0 WHEN C1 = 1 THEN ID1 ELSE ID2 END AS C_ID
FROM windows
WHERE A0 + A1 + A2 = 1
  AND B0 + B1 + B2 = 1
  AND C0 + C1 + C2 = 1;

WITH base AS (
  SELECT
    BEAT,
    TS,
    ID,
    CASE WHEN PRIMARY_TYPE = 'BURGLARY' AND ARREST = 'false' THEN 1 ELSE 0 END AS is_A,
    CASE WHEN PRIMARY_TYPE = 'CRIMINAL TRESPASS' AND ARREST = 'false' THEN 1 ELSE 0 END AS is_B,
    CASE WHEN PRIMARY_TYPE = 'CRIMINAL DAMAGE' AND LOCATION_DESCRIPTION = 'STREET' THEN 1 ELSE 0 END AS is_C
  FROM CRIMES
),
ranked AS (
  SELECT
    BEAT,
    TS,
    ID,
    is_A,
    is_B,
    is_C,
    LEAD(ID, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next1_ID,
    LEAD(is_A, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next1_is_A,
    LEAD(is_B, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next1_is_B,
    LEAD(is_C, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next1_is_C,
    LEAD(ID, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next2_ID,
    LEAD(is_A, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next2_is_A,
    LEAD(is_B, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next2_is_B,
    LEAD(is_C, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next2_is_C
  FROM base
)
SELECT
  BEAT,
  TS,
  ID,
  CASE
    WHEN is_A = 1 THEN ID
    WHEN next1_is_A = 1 THEN next1_ID
    WHEN next2_is_A = 1 THEN next2_ID
  END AS A_ID,
  CASE
    WHEN is_B = 1 THEN ID
    WHEN next1_is_B = 1 THEN next1_ID
    WHEN next2_is_B = 1 THEN next2_ID
  END AS B_ID,
  CASE
    WHEN is_C = 1 THEN ID
    WHEN next1_is_C = 1 THEN next1_ID
    WHEN next2_is_C = 1 THEN next2_ID
  END AS C_ID
FROM ranked
WHERE (is_A + next1_is_A + next2_is_A) = 1
  AND (is_B + next1_is_B + next2_is_B) = 1
  AND (is_C + next1_is_C + next2_is_C) = 1;

WITH src AS (
  SELECT c.ID, c.CASE_NUMBER, c.TS, c.BLOCK, c.IUCR, c.PRIMARY_TYPE, c.DESCRIPTION,
         c.LOCATION_DESCRIPTION, c.ARREST, c.DOMESTIC, c.BEAT, c.DISTRICT, c.WARD,
         c.COMMUNITY_AREA, c.FBI_CODE, c.X_COORDINATE, c.Y_COORDINATE, c.YEAR,
         c.UPDATED_ON, c.LATITUDE, c.LONGITUDE, c.LOCATION,
         ROW_NUMBER() OVER (PARTITION BY c.BEAT ORDER BY c.TS, c.ID) AS rn,
         LEAD(c.ARREST) OVER (PARTITION BY c.BEAT ORDER BY c.TS, c.ID) AS next_arrest,
         LEAD(c.TS)     OVER (PARTITION BY c.BEAT ORDER BY c.TS, c.ID) AS next_ts
  FROM CRIMES c
),
f AS (
  SELECT s.*,
         CASE WHEN s.next_arrest = 'true'
               AND s.next_ts - s.TS <= 86400000
               AND s.PRIMARY_TYPE IN ('THEFT','BURGLARY','CRIMINAL DAMAGE',
                                      'ASSAULT','BATTERY','ROBBERY')
              THEN 1 ELSE 0 END AS cp
  FROM src s
),
g AS (
  SELECT f.*,
         MAX(CASE WHEN f.cp = 0 THEN f.rn END)
           OVER (PARTITION BY f.BEAT ORDER BY f.TS, f.ID ROWS UNBOUNDED PRECEDING) AS last_zero_rn
  FROM f
),
h AS (
  SELECT g.*,
         CASE WHEN g.cp = 1
               AND MOD(g.rn - COALESCE(g.last_zero_rn, 0), 2) = 1
              THEN 1 ELSE 0 END AS is_start
  FROM g
),
m AS (
  SELECT h.*,
         LAG(h.is_start, 1, 0)
           OVER (PARTITION BY h.BEAT ORDER BY h.TS, h.ID) AS prev_is_start,
         SUM(h.is_start)
           OVER (PARTITION BY h.BEAT ORDER BY h.TS, h.ID ROWS UNBOUNDED PRECEDING) AS match_no
  FROM h
)
SELECT ID, CASE_NUMBER, TS, BLOCK, IUCR, PRIMARY_TYPE, DESCRIPTION, LOCATION_DESCRIPTION,
       ARREST, DOMESTIC, BEAT, DISTRICT, WARD, COMMUNITY_AREA, FBI_CODE,
       X_COORDINATE, Y_COORDINATE, YEAR, UPDATED_ON, LATITUDE, LONGITUDE, LOCATION,
       match_no AS MATCH_NO,
       CASE WHEN is_start = 1 THEN
              CASE WHEN PRIMARY_TYPE IN ('THEFT','BURGLARY','CRIMINAL DAMAGE')
                   THEN 'PROPERTY_EVENT'
                   ELSE 'VIOLENT_EVENT' END
            ELSE 'ARREST_EVENT' END AS EVENT_CLASS
FROM m
WHERE is_start = 1 OR prev_is_start = 1
ORDER BY BEAT, MATCH_NO, TS, ID;

WITH CLASSIFIED AS (

      WHEN C.PRIMARY_TYPE = 'ARSON' AND C.LOCATION_DESCRIPTION = 'VEHICLE NON-COMMERCIAL' THEN 'PROPERTY_EVENT'
      WHEN C.PRIMARY_TYPE = 'KIDNAPPING' AND C.LOCATION_DESCRIPTION = 'RESIDENCE' THEN 'VIOLENT_EVENT'
      WHEN C.YEAR = 2020 AND C.PRIMARY_TYPE = 'THEFT' AND C.DISTRICT = '012' THEN 'ARREST_EVENT'
    END AS EVENT_CLS
  FROM CRIMES C
),
FLAGGED AS (
  SELECT
    CLASSIFIED.*,
    CASE
      WHEN EVENT_CLS IN ('PROPERTY_EVENT', 'VIOLENT_EVENT')
           AND LEAD(EVENT_CLS) OVER (PARTITION BY BEAT ORDER BY TS, ID) = 'ARREST_EVENT'
      THEN 1 ELSE 0
    END AS IS_START
  FROM CLASSIFIED
),
NUMBERED AS (
  SELECT
    FLAGGED.*,
    SUM(IS_START) OVER (PARTITION BY BEAT ORDER BY TS, ID ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS MATCH_NO,
    LAG(IS_START) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS PREV_IS_START
  FROM FLAGGED
)
SELECT
  ID, CASE_NUMBER, TS, BLOCK, IUCR, PRIMARY_TYPE, DESCRIPTION, LOCATION_DESCRIPTION,
  ARREST, DOMESTIC, BEAT, DISTRICT, WARD, COMMUNITY_AREA, FBI_CODE,
  X_COORDINATE, Y_COORDINATE, YEAR, UPDATED_ON, LATITUDE, LONGITUDE, LOCATION,
  MATCH_NO,
  CASE WHEN IS_START = 1 THEN EVENT_CLS ELSE 'ARREST_EVENT' END AS EVENT_CLASS
FROM NUMBERED
WHERE IS_START = 1 OR PREV_IS_START = 1
ORDER BY BEAT, TS, ID;

WITH base AS (

         CASE WHEN c.YEAR = 2003 AND c.PRIMARY_TYPE = 'ASSAULT' THEN 1 ELSE 0 END AS IS_P,
         CASE WHEN c.PRIMARY_TYPE = 'NARCOTICS' AND c.WARD = 3 THEN 1 ELSE 0 END AS IS_V,

ordered AS (

  FROM base b

matched AS (

         SUM(CASE WHEN (o.IS_P = 1 OR o.IS_V = 1) AND o.NEXT_IS_A = 1 THEN 1 ELSE 0 END)

  FROM ordered o

SELECT

       WHEN IS_A = 1 THEN 'ARREST_EVENT'

WITH base AS (

classified AS (

matched AS (

  FROM classified

WITH ordered AS (
  SELECT 
    c.*,
    ROW_NUMBER() OVER (PARTITION BY COMMUNITY_AREA ORDER BY TS, ID) AS rn,
    CASE WHEN PRIMARY_TYPE IN ('THEFT','MOTOR VEHICLE THEFT') THEN 1 ELSE 0 END AS is_theft,
    CASE WHEN PRIMARY_TYPE = 'ROBBERY' THEN 1 ELSE 0 END AS is_robbery
  FROM CRIMES c
),
theft_groups AS (
  SELECT 
    o.*,
    SUM(CASE WHEN is_theft = 0 THEN 1 ELSE 0 END) OVER (PARTITION BY COMMUNITY_AREA ORDER BY rn) AS theft_grp
  FROM ordered o
),
run_info AS (
  SELECT 
    tg.*,
    MAX(rn) OVER (PARTITION BY COMMUNITY_AREA, theft_grp) AS theft_run_end_rn
  FROM theft_groups tg
),
robbery_window AS (
  SELECT 
    ri.*,
    MAX(CASE WHEN is_robbery = 1 THEN rn END) OVER (
      PARTITION BY COMMUNITY_AREA 
      ORDER BY TS 
      RANGE BETWEEN CURRENT ROW AND 86400000 FOLLOWING
    ) AS last_robbery_rn
  FROM run_info ri

SELECT 
  rw.TS AS THEFT_START_TS,
  rob.TS AS ROBBERY_END_TS,
  rw.theft_run_end_rn - rw.rn + 1 AS THEFT_COUNT,
  rw.last_robbery_rn - rw.theft_run_end_rn - 1 AS INTERVENING_COUNT,
  1 AS ROBBERY_COUNT
FROM robbery_window rw
JOIN robbery_window rob 
  ON NVL(rob.COMMUNITY_AREA, -1) = NVL(rw.COMMUNITY_AREA, -1)
  AND rob.rn = rw.last_robbery_rn
WHERE rw.is_theft = 1
  AND rw.last_robbery_rn > rw.rn;

WITH crime_flags AS (
  SELECT
    c.*,
    CASE WHEN PRIMARY_TYPE = 'BURGLARY' THEN 1 ELSE 0 END AS is_start,
    CASE WHEN PRIMARY_TYPE IN ('THEFT', 'CRIMINAL DAMAGE') THEN 1 ELSE 0 END AS is_middle,
    CASE WHEN PRIMARY_TYPE = 'ROBBERY' THEN 1 ELSE 0 END AS is_end,
    CASE WHEN PRIMARY_TYPE IN ('THEFT', 'CRIMINAL DAMAGE') THEN 0 ELSE 1 END AS is_non_middle
  FROM CRIMES c
),
crime_windows AS (
  SELECT
    f.*,
    LAST_VALUE(CASE WHEN is_non_middle = 1 THEN is_start END) IGNORE NULLS
      OVER (PARTITION BY BLOCK ORDER BY TS, ID ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) AS prev_non_middle_is_start,
    LAST_VALUE(CASE WHEN is_non_middle = 1 THEN ID END) IGNORE NULLS
      OVER (PARTITION BY BLOCK ORDER BY TS, ID ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) AS prev_non_middle_id
  FROM crime_flags f
),
matches AS (
  SELECT
    w.*,
    CASE
      WHEN is_start = 1 THEN 1
      WHEN (is_middle = 1 OR is_end = 1) AND prev_non_middle_is_start = 1 THEN 1
      ELSE 0
    END AS in_match,
    CASE
      WHEN is_start = 1 THEN ID
      WHEN (is_middle = 1 OR is_end = 1) AND prev_non_middle_is_start = 1 THEN prev_non_middle_id
      ELSE NULL
    END AS match_start_id
  FROM crime_windows w
)
SELECT
  ID, CASE_NUMBER, TS, BLOCK, IUCR, PRIMARY_TYPE, DESCRIPTION, LOCATION_DESCRIPTION,
  ARREST, DOMESTIC, BEAT, DISTRICT, WARD, COMMUNITY_AREA, FBI_CODE,
  X_COORDINATE, Y_COORDINATE, YEAR, UPDATED_ON, LATITUDE, LONGITUDE, LOCATION,
  MAX(CASE WHEN is_start = 1 THEN TS END) OVER (PARTITION BY BLOCK, match_start_id) AS START_TS,
  MAX(CASE WHEN is_end = 1 THEN TS END) OVER (PARTITION BY BLOCK, match_start_id) AS END_TS,
  MAX(CASE WHEN is_start = 1 THEN ID END) OVER (PARTITION BY BLOCK, match_start_id) AS START_ID,
  MAX(CASE WHEN is_end = 1 THEN ID END) OVER (PARTITION BY BLOCK, match_start_id) AS END_ID
FROM matches
WHERE in_match = 1;

    CASE WHEN PRIMARY_TYPE = 'BATTERY' AND DISTRICT = '001' AND DOMESTIC = 'false' THEN 1 ELSE 0 END AS is_b,
    CASE WHEN PRIMARY_TYPE = 'BURGLARY' AND DISTRICT = '025' AND ARREST = 'false' THEN 1 ELSE 0 END AS is_c
  FROM CRIMES
),
grp AS (
  SELECT
    b.BLOCK, b.ID, b.rn, b.is_a, b.is_b, b.is_c,
    SUM(CASE WHEN b.is_b = 1 AND LAG(b.is_b, 1, 0) OVER (PARTITION BY b.BLOCK ORDER BY b.rn) = 0 THEN 1 ELSE 0 END)
      OVER (PARTITION BY b.BLOCK ORDER BY b.rn) AS b_grp
  FROM base b
),
runs AS (
  SELECT
    g.BLOCK, g.ID, g.rn, g.is_a, g.is_b, g.is_c, g.b_grp,
    MAX(CASE WHEN g.is_b = 1 THEN g.rn END) OVER (PARTITION BY g.BLOCK, g.b_grp) AS b_run_end
  FROM grp g
),
ar AS (
  SELECT
    r.BLOCK, r.ID, r.rn, r.is_a, r.is_c,
    LEAD(CASE WHEN r.is_b = 1 THEN r.b_run_end END, 1) OVER (PARTITION BY r.BLOCK ORDER BY r.rn) AS e
  FROM runs r
)
SELECT
  a.BLOCK,
  a.ID AS START_ID,
  mb.ID AS LAST_MIDDLE_ID,
  ec.ID AS END_ID
FROM ar a
LEFT JOIN base mb
  ON mb.BLOCK = a.BLOCK AND mb.rn = a.e
LEFT JOIN base ec
  ON ec.BLOCK = a.BLOCK AND ec.rn = COALESCE(a.e, a.rn) + 1 AND ec.is_c = 1
WHERE a.is_a = 1;

WITH ordered AS (

         CASE WHEN PRIMARY_TYPE = 'ASSAULT' AND WARD = 39 THEN 1 ELSE 0 END AS is_open,

grouped AS (
  SELECT o.*,
         LAG(ID) OVER (PARTITION BY DISTRICT ORDER BY TS, ID) AS prev_id,
         LAG(is_open) OVER (PARTITION BY DISTRICT ORDER BY TS, ID) AS prev_is_open,
         SUM(1 - is_close) OVER (PARTITION BY DISTRICT ORDER BY TS, ID ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS close_group
  FROM ordered o
),
match_info AS (
  SELECT g.*,
         COUNT(*) OVER (PARTITION BY DISTRICT, close_group) AS CLOSE_COUNT,
         FIRST_VALUE(ID) OVER (PARTITION BY DISTRICT, close_group ORDER BY TS DESC, ID DESC) AS LAST_CLOSE_ID
  FROM grouped g
  WHERE is_close = 1

         LAG(is_open) OVER (ORDER BY TS, ID) AS prev_is_open,

         LAG(TS) OVER (ORDER BY TS, ID) AS prev_ts
  FROM base

runs AS (

         CASE WHEN is_close = 1 AND prev_is_close = 0 AND prev_is_open = 1 THEN prev_id END AS run_open_id,
         CASE WHEN is_close = 1 AND prev_is_close = 0 AND prev_is_open = 1 THEN prev_ts END AS run_open_ts,

WITH base AS (
  SELECT
    ID,
    TS,
    BEAT,
    PRIMARY_TYPE,
    ARREST,
    ROW_NUMBER() OVER (PARTITION BY BEAT ORDER BY TS, ID) AS rn,
    CASE WHEN (PRIMARY_TYPE IN ('THEFT', 'MOTOR VEHICLE THEFT') AND ARREST = 'false')
              OR (PRIMARY_TYPE = 'CRIMINAL DAMAGE' AND ARREST = 'false')
         THEN 1 ELSE 0 END AS is_P,
    CASE WHEN PRIMARY_TYPE = 'CRIMINAL DAMAGE' AND ARREST = 'true' AND TS > 1640966400000
         THEN 1 ELSE 0 END AS is_A
  FROM CRIMES
),
computed AS (
  SELECT
    *,
    LAST_VALUE(CASE WHEN is_P = 1 THEN TS END) OVER (
      PARTITION BY BEAT
      ORDER BY rn
      ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
    ) AS last_P_TS,
    LAG(is_P, 1) OVER (PARTITION BY BEAT ORDER BY rn) AS lag_is_P1,
    LAG(is_P, 2) OVER (PARTITION BY BEAT ORDER BY rn) AS lag_is_P2,
    LAG(TS, 1) OVER (PARTITION BY BEAT ORDER BY rn) AS lag_TS1,
    LAG(TS, 2) OVER (PARTITION BY BEAT ORDER BY rn) AS lag_TS2
  FROM base
),
flags AS (
  SELECT
    *,
    CASE WHEN is_A = 1 AND last_P_TS IS NOT NULL AND TS - last_P_TS <= 86400000
         THEN 1 ELSE 0 END AS valid_A,
    CASE WHEN is_A = 1 AND lag_is_P1 = 1 AND lag_is_P2 = 1
         THEN 1 ELSE 0 END AS is_start
  FROM computed
),
blocks AS (
  SELECT
    *,
    SUM(CASE WHEN valid_A = 1 AND (LAG(valid_A, 1) OVER (PARTITION BY BEAT ORDER BY rn) IS NULL OR LAG(valid_A, 1) OVER (PARTITION BY BEAT ORDER BY rn) = 0) THEN 1 ELSE 0 END) OVER (
      PARTITION BY BEAT
      ORDER BY rn
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS grp
  FROM flags
)
SELECT
  BEAT,
  MIN(lag_TS2) KEEP (DENSE_RANK FIRST ORDER BY rn) AS PROPERTY_START_TS,
  MIN(lag_TS1) KEEP (DENSE_RANK FIRST ORDER BY rn) AS PROPERTY_END_TS,
  MIN(TS) KEEP (DENSE_RANK LAST ORDER BY rn) AS ARREST_END_TS,
  2 AS PROPERTY_COUNT,
  COUNT(*) AS ARREST_COUNT
FROM blocks
WHERE valid_A = 1
GROUP BY BEAT, grp
HAVING MIN(is_start) KEEP (DENSE_RANK FIRST ORDER BY rn) = 1
ORDER BY BEAT, MIN(rn) KEEP (DENSE_RANK FIRST ORDER BY rn);

WITH base AS (
  SELECT
    ID,
    DISTRICT,
    TS,
    PRIMARY_TYPE,
    CASE WHEN PRIMARY_TYPE IN ('THEFT', 'CRIMINAL DAMAGE', 'BATTERY', 'ROBBERY') THEN 1 ELSE 0 END AS is_n,
    CASE WHEN PRIMARY_TYPE = 'HOMICIDE' THEN 1 ELSE 0 END AS is_h,
    SUM(CASE WHEN PRIMARY_TYPE NOT IN ('THEFT', 'CRIMINAL DAMAGE', 'BATTERY', 'ROBBERY') THEN 1 ELSE 0 END)
      OVER (PARTITION BY DISTRICT ORDER BY TS, ID) AS grp
  FROM CRIMES
),
n_rows AS (
  SELECT
    DISTRICT,
    TS,
    ID,
    grp,
    COUNT(*) OVER (PARTITION BY DISTRICT, grp ORDER BY TS, ID ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING) AS ambiguous_count
  FROM base
  WHERE is_n = 1
),
h_rows AS (
  SELECT
    DISTRICT,
    TS,
    ID,
    grp
  FROM base
  WHERE is_h = 1
)
SELECT
  n.DISTRICT,
  n.TS AS START_TS,
  h.TS AS END_TS,
  n.ambiguous_count AS AMBIGUOUS_COUNT
FROM n_rows n
JOIN h_rows h
  ON (n.DISTRICT = h.DISTRICT OR (n.DISTRICT IS NULL AND h.DISTRICT IS NULL))
 AND h.grp = n.grp + 1;
