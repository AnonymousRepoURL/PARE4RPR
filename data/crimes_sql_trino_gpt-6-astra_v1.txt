  DEFINE
    FIRST_HOMICIDE AS FIRST_HOMICIDE.PRIMARY_TYPE = 'HOMICIDE',
    FOLLOWUP AS FOLLOWUP.TS - FIRST_HOMICIDE.TS <= 3456000000
);

    LAST(N.TS) AS LAST_DAY_TS,
    COUNT(*) AS INCIDENT_COUNT

  AFTER MATCH SKIP PAST LAST ROW
  PATTERN (F S* (N S*){2,})
  SUBSET X = (F, N)
  DEFINE
    F AS TRUE,
    S AS FLOOR(S.TS / 86400000) = FLOOR(PREV(S.TS) / 86400000),
    N AS FLOOR(N.TS / 86400000) = FLOOR(PREV(X.TS) / 86400000) + 1
);

SELECT *
FROM (
  SELECT ID, TS, PRIMARY_TYPE, DISTRICT, LONGITUDE
  FROM CRIMES
  WHERE (PRIMARY_TYPE = 'ROBBERY' AND DISTRICT IS NOT NULL)
     OR (PRIMARY_TYPE = 'BATTERY' AND DISTRICT IS NOT NULL AND LONGITUDE IS NOT NULL)
     OR (PRIMARY_TYPE = 'MOTOR VEHICLE THEFT' AND LONGITUDE IS NOT NULL)
) MATCH_RECOGNIZE (
  ORDER BY TS, ID
  MEASURES
    A.ID AS ROBBERY_ID,
    B.ID AS BATTERY_ID,
    C.ID AS MOTOR_VEHICLE_THEFT_ID
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO NEXT ROW
  PATTERN (A N1*? B N2*? C)
  DEFINE
    A AS A.PRIMARY_TYPE = 'ROBBERY',
    N1 AS N1.TS - A.TS <= 1800000,
    B AS B.PRIMARY_TYPE = 'BATTERY'
      AND B.DISTRICT = A.DISTRICT
      AND B.TS - A.TS <= 1800000,
    N2 AS N2.TS - A.TS <= 1800000,
    C AS C.PRIMARY_TYPE = 'MOTOR VEHICLE THEFT'
      AND C.LONGITUDE <= B.LONGITUDE + 0.1
      AND C.LONGITUDE >= B.LONGITUDE - 0.1
      AND C.TS - A.TS <= 1800000
);

  SELECT
    COMMUNITY_AREA,
    TS,
    ID,
    PRIMARY_TYPE IN ('ASSAULT', 'BATTERY', 'ROBBERY')
      AND ARREST = 'false' AS IS_ESCALATING,
    PRIMARY_TYPE IN ('WEAPONS VIOLATION', 'HOMICIDE') AS IS_CRITICAL
  FROM CRIMES
) MATCH_RECOGNIZE (
  PARTITION BY COMMUNITY_AREA
  ORDER BY TS, ID
  MEASURES
    FIRST(ESCALATING.TS) AS ESCALATION_START_TS,
    LAST(ESCALATING.TS) AS ESCALATION_END_TS,

    COUNT(ESCALATING.*) AS ESCALATION_COUNT

  AFTER MATCH SKIP PAST LAST ROW
  PATTERN (ESCALATING{3,} CRITICAL)
  DEFINE
    ESCALATING AS ESCALATING.IS_ESCALATING,
    CRITICAL AS CRITICAL.IS_CRITICAL
      AND CRITICAL.TS - FIRST(ESCALATING.TS) <= 86400000
);

SELECT BEAT, THEFT_ID, BATTERY_ID, DAMAGE_ID
FROM (
  SELECT
    BEAT,
    MAX(CASE WHEN PRIMARY_TYPE = 'THEFT' THEN ID END)
      OVER w AS THEFT_ID,
    MAX(CASE WHEN PRIMARY_TYPE = 'BATTERY' THEN ID END)
      OVER w AS BATTERY_ID,
    MAX(CASE WHEN PRIMARY_TYPE = 'CRIMINAL DAMAGE' THEN ID END)
      OVER w AS DAMAGE_ID
  FROM CRIMES
  WINDOW w AS (
    PARTITION BY BEAT
    ORDER BY TS, ID
    ROWS BETWEEN CURRENT ROW AND 2 FOLLOWING
  )
) AS event_windows
WHERE THEFT_ID IS NOT NULL
  AND BATTERY_ID IS NOT NULL
  AND DAMAGE_ID IS NOT NULL;

WITH classified AS (
  SELECT
    BEAT,
    TS,
    ID,
    CASE
      WHEN PRIMARY_TYPE = 'ASSAULT'
        AND COMMUNITY_AREA = 28
        AND ARREST = 'true' THEN 1
      WHEN YEAR = 2024
        AND PRIMARY_TYPE = 'BATTERY'
        AND DISTRICT = '011' THEN 2
      WHEN PRIMARY_TYPE = 'OTHER OFFENSE'
        AND COMMUNITY_AREA = 33
        AND ARREST = 'false' THEN 3
    END AS event_code
  FROM CRIMES
),
sequenced AS (
  SELECT
    BEAT,
    ID,
    event_code,
    LEAD(ID, 1) OVER w AS next_id,
    LEAD(ID, 2) OVER w AS second_id,
    LEAD(event_code, 1) OVER w AS next_code,
    LEAD(event_code, 2) OVER w AS second_code
  FROM classified
  WINDOW w AS (PARTITION BY BEAT ORDER BY TS, ID)
)
SELECT
  BEAT,
  CASE
    WHEN event_code = 1 THEN ID
    WHEN next_code = 1 THEN next_id
    ELSE second_id
  END AS A_ID,
  CASE
    WHEN event_code = 2 THEN ID
    WHEN next_code = 2 THEN next_id
    ELSE second_id
  END AS B_ID,
  CASE
    WHEN event_code = 3 THEN ID
    WHEN next_code = 3 THEN next_id
    ELSE second_id
  END AS C_ID
FROM sequenced
WHERE event_code <> next_code
  AND event_code <> second_code
  AND next_code <> second_code;

        LAG(PRIMARY_TYPE = 'ASSAULT') OVER w AS PRECEDED_BY_ASSAULT,
        LEAD(
            CASE
                WHEN PRIMARY_TYPE = 'BATTERY'
                     AND TS BETWEEN 1755864000000 AND 1755873000000
                THEN ID
            END
        ) OVER w AS BATTERY_ID,
        LEAD(
            CASE WHEN PRIMARY_TYPE = 'MOTOR VEHICLE THEFT' THEN ID END,
            2
        ) OVER w AS MOTOR_VEHICLE_THEFT_ID
    FROM CRIMES
    WINDOW w AS (ORDER BY TS, ID)
) AS candidates

  AND PRECEDED_BY_ASSAULT
  AND BATTERY_ID IS NOT NULL
  AND MOTOR_VEHICLE_THEFT_ID IS NOT NULL;

  MEASURES MATCH_NUMBER() AS MATCH_NO, CLASSIFIER() AS EVENT_CLASS, FINAL COUNT(VIOLENT_EVENT.*) AS MATCH_SIZE
  ALL ROWS PER MATCH WITH UNMATCHED ROWS
  AFTER MATCH SKIP PAST LAST ROW
  PATTERN(VIOLENT_EVENT{2,})
  DEFINE VIOLENT_EVENT AS VIOLENT_EVENT.PRIMARY_TYPE IN ('ASSAULT', 'BATTERY', 'ROBBERY') AND VIOLENT_EVENT.TS - FIRST(VIOLENT_EVENT.TS) <= 3600000
);

SELECT * FROM CRIMES MATCH_RECOGNIZE(
  PARTITION BY BLOCK, PRIMARY_TYPE
  ORDER BY TS, ID
  MEASURES CLASSIFIER() AS EVENT_CLASS, RUNNING COUNT(FOLLOWUP.*) AS FOLLOWUPS_SO_FAR, FINAL COUNT(FOLLOWUP.*) AS FOLLOWUP_TOTAL, RUNNING LAST(FOLLOWUP.TS) AS CURRENT_END_TS, FINAL LAST(FOLLOWUP.TS) AS FINAL_END_TS
  ALL ROWS PER MATCH
  AFTER MATCH SKIP PAST LAST ROW
  PATTERN(START_EVENT FOLLOWUP+)
  DEFINE
    START_EVENT AS 1 = 1,
    FOLLOWUP AS TS - START_EVENT.TS <= 86400000
);

SELECT * FROM CRIMES MATCH_RECOGNIZE(
  PARTITION BY BEAT
  ORDER BY TS, ID
  MEASURES
    FIRST(PROPERTY_EVENT.TS) AS PROPERTY_START_TS,
    LAST(PROPERTY_EVENT.TS) AS PROPERTY_END_TS,
    LAST(ARRESTED.TS) AS ARREST_END_TS,
    COUNT(PROPERTY_EVENT.*) AS PROPERTY_COUNT,
    COUNT(ARRESTED.*) AS ARREST_COUNT
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO LAST PROPERTY_EVENT
  PATTERN(PROPERTY_EVENT{2} ARRESTED+)
  DEFINE
    PROPERTY_EVENT AS PRIMARY_TYPE IN ('THEFT', 'MOTOR VEHICLE THEFT', 'CRIMINAL DAMAGE') AND ARREST = 'false',
    ARRESTED AS PRIMARY_TYPE = 'CRIMINAL DAMAGE' AND ARREST = 'true' AND TS > 1640966400000 AND TS - PROPERTY_EVENT.TS <= 86400000
);

SELECT *
FROM CRIMES MATCH_RECOGNIZE(
  PARTITION BY DISTRICT
  ORDER BY TS, ID
  MEASURES
    FIRST(AMBIGUOUS_EVENT.TS) AS START_TS,
    HOMICIDE_EVENT.TS AS END_TS,
    COUNT(AMBIGUOUS_EVENT.*) AS AMBIGUOUS_COUNT
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO NEXT ROW
  PATTERN(AMBIGUOUS_EVENT+ HOMICIDE_EVENT)
  DEFINE
    AMBIGUOUS_EVENT AS PRIMARY_TYPE IN ('THEFT', 'CRIMINAL DAMAGE', 'BATTERY', 'ROBBERY'),
    HOMICIDE_EVENT AS PRIMARY_TYPE = 'HOMICIDE'
);
