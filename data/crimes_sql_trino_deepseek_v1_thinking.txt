WITH
base AS (
  SELECT
    BLOCK,
    PRIMARY_TYPE,
    TS,
    ID,
    FLOOR(TS / 86400000) AS day
  FROM CRIMES
),
ordered AS (
  SELECT
    BLOCK,
    PRIMARY_TYPE,
    TS,
    ID,
    day,
    LAG(day) OVER (PARTITION BY BLOCK, PRIMARY_TYPE ORDER BY TS, ID) AS prev_day
  FROM base
),
runs AS (
  SELECT
    BLOCK,
    PRIMARY_TYPE,
    TS,
    ID,
    day,
    SUM(CASE WHEN prev_day IS NULL OR day > prev_day + 1 THEN 1 ELSE 0 END)
      OVER (PARTITION BY BLOCK, PRIMARY_TYPE ORDER BY TS, ID) AS run_id
  FROM ordered
),
numbered AS (
  SELECT
    BLOCK,
    PRIMARY_TYPE,
    TS,
    ID,
    day,
    run_id,
    ROW_NUMBER() OVER (PARTITION BY BLOCK, PRIMARY_TYPE, run_id ORDER BY TS, ID) AS rn_in_run
  FROM runs
),
run_stats AS (
  SELECT
    BLOCK,
    PRIMARY_TYPE,
    run_id,
    MAX(day) AS max_day,
    COUNT(*) AS total_rows
  FROM runs
  GROUP BY BLOCK, PRIMARY_TYPE, run_id
),
start_day_per_run AS (
  SELECT
    BLOCK,
    PRIMARY_TYPE,
    run_id,
    MIN(day) AS start_day
  FROM runs
  WHERE TS > 1786118400000
  GROUP BY BLOCK, PRIMARY_TYPE, run_id
),
start_rows AS (
  SELECT
    n.BLOCK,
    n.PRIMARY_TYPE,
    n.run_id,
    MIN(n.rn_in_run) AS start_rn,
    MIN_BY(n.TS, n.rn_in_run) AS start_ts
  FROM numbered n
  JOIN start_day_per_run sd
    ON n.BLOCK = sd.BLOCK
    AND n.PRIMARY_TYPE = sd.PRIMARY_TYPE
    AND n.run_id = sd.run_id
    AND n.day = sd.start_day
  WHERE n.TS > 1786118400000
  GROUP BY n.BLOCK, n.PRIMARY_TYPE, n.run_id
),
last_day_rows AS (
  SELECT
    n.BLOCK,
    n.PRIMARY_TYPE,
    n.run_id,
    MIN_BY(n.TS, n.rn_in_run) AS last_day_ts
  FROM numbered n
  JOIN run_stats r
    ON n.BLOCK = r.BLOCK
    AND n.PRIMARY_TYPE = r.PRIMARY_TYPE
    AND n.run_id = r.run_id
    AND n.day = r.max_day
  GROUP BY n.BLOCK, n.PRIMARY_TYPE, n.run_id
)
SELECT
  s.BLOCK,
  s.PRIMARY_TYPE,
  s.start_ts AS FIRST_DAY_TS,
  l.last_day_ts AS LAST_DAY_TS,
  r.total_rows - s.start_rn + 1 AS INCIDENT_COUNT
FROM start_rows s
JOIN run_stats r
  ON s.BLOCK = r.BLOCK
  AND s.PRIMARY_TYPE = r.PRIMARY_TYPE
  AND s.run_id = r.run_id
JOIN last_day_rows l
  ON s.BLOCK = l.BLOCK
  AND s.PRIMARY_TYPE = l.PRIMARY_TYPE
  AND s.run_id = l.run_id
WHERE r.max_day - s.start_day >= 2;

WITH base AS (
    SELECT
        DISTRICT,
        TS,
        ID,
        PRIMARY_TYPE,
        LOCATION_DESCRIPTION,
        COMMUNITY_AREA,
        ARREST,
        YEAR
    FROM CRIMES
    WHERE DISTRICT = '016'
),
numbered AS (
    SELECT
        *,
        ROW_NUMBER() OVER (PARTITION BY DISTRICT ORDER BY TS, ID) AS rn,
        PRIMARY_TYPE = 'ARSON' AND LOCATION_DESCRIPTION = 'RESIDENCE' AS is_report,
        PRIMARY_TYPE = 'STALKING' AND LOCATION_DESCRIPTION = 'RESIDENCE' AS is_theft,
        PRIMARY_TYPE = 'WEAPONS VIOLATION' AND COMMUNITY_AREA = 46 AND ARREST = 'true' AS is_damage,
        YEAR = 2001 AND PRIMARY_TYPE = 'BATTERY' AND DISTRICT = '016' AS is_arrested
    FROM base
),
flagged AS (
    SELECT
        *,
        is_theft OR is_damage AS is_middle,
        LAG(is_theft OR is_damage, 1, false) OVER (PARTITION BY DISTRICT ORDER BY TS, ID) AS prev_is_middle
    FROM numbered
),
runs AS (
    SELECT
        *,
        SUM(CASE WHEN is_middle AND NOT prev_is_middle THEN 1 ELSE 0 END) OVER (PARTITION BY DISTRICT ORDER BY TS, ID) AS mid_run_id
    FROM flagged
),
with_lag AS (
    SELECT
        *,
        LAG(mid_run_id) OVER (PARTITION BY DISTRICT ORDER BY TS, ID) AS prev_mid_run_id
    FROM runs
),
middle_runs AS (
    SELECT
        DISTRICT,
        mid_run_id,
        MIN(rn) AS run_start_rn,
        MAX(CASE WHEN is_theft THEN rn END) AS last_theft_rn,
        MAX(CASE WHEN is_damage THEN rn END) AS last_damage_rn
    FROM runs
    WHERE is_middle
    GROUP BY DISTRICT, mid_run_id
),
arrested_with_run AS (
    SELECT
        DISTRICT,
        TS,
        ID,
        rn,
        prev_mid_run_id,
        prev_is_middle
    FROM with_lag
    WHERE is_arrested
),
matches AS (
    SELECT
        a.DISTRICT,
        a.TS,
        a.ID AS ARRESTED_ID,
        mr.run_start_rn,
        mr.last_theft_rn,
        mr.last_damage_rn
    FROM arrested_with_run a
    JOIN middle_runs mr ON a.DISTRICT = mr.DISTRICT AND a.prev_mid_run_id = mr.mid_run_id
    WHERE a.prev_is_middle = true
),
report_rows AS (
    SELECT
        m.DISTRICT,
        m.TS,
        m.ARRESTED_ID,
        m.run_start_rn,
        m.last_theft_rn,
        m.last_damage_rn,
        r.ID AS REPORT_ID
    FROM matches m
    JOIN numbered r ON m.DISTRICT = r.DISTRICT AND r.rn = m.run_start_rn - 1
    WHERE r.is_report
),
final AS (
    SELECT
        rr.DISTRICT,
        rr.TS,
        rr.ARRESTED_ID AS ID,
        rr.REPORT_ID,
        lt.ID AS LAST_THEFT_ID,
        ld.ID AS LAST_DAMAGE_ID,
        rr.ARRESTED_ID
    FROM report_rows rr
    LEFT JOIN numbered lt ON rr.DISTRICT = lt.DISTRICT AND rr.last_theft_rn = lt.rn
    LEFT JOIN numbered ld ON rr.DISTRICT = ld.DISTRICT AND rr.last_damage_rn = ld.rn
)
SELECT
    DISTRICT,
    TS,
    ID,
    REPORT_ID,
    LAST_THEFT_ID,
    LAST_DAMAGE_ID,
    ARRESTED_ID
FROM final;

WITH base AS (
  SELECT
    COMMUNITY_AREA,
    TS,
    ID,
    PRIMARY_TYPE,
    ARREST,
    CASE WHEN PRIMARY_TYPE IN ('ASSAULT', 'BATTERY', 'ROBBERY') AND ARREST = 'false' THEN true ELSE false END AS is_esc,
    CASE WHEN PRIMARY_TYPE IN ('WEAPONS VIOLATION', 'HOMICIDE') THEN true ELSE false END AS is_crit
  FROM CRIMES
),
numbered AS (
  SELECT
    *,
    ROW_NUMBER() OVER (PARTITION BY COMMUNITY_AREA ORDER BY TS, ID) AS rn,
    SUM(CASE WHEN is_esc THEN 0 ELSE 1 END) OVER (PARTITION BY COMMUNITY_AREA ORDER BY TS, ID ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS grp
  FROM base
),
runs AS (
  SELECT
    COMMUNITY_AREA,
    grp,
    COUNT(*) AS run_len,
    MIN(rn) AS run_start_rn,
    MAX(rn) AS run_end_rn,
    MAX(TS) AS run_end_ts
  FROM numbered
  WHERE is_esc = true
  GROUP BY COMMUNITY_AREA, grp
),
runs_with_next AS (
  SELECT
    r.COMMUNITY_AREA,
    r.grp,
    r.run_len,
    r.run_start_rn,
    r.run_end_rn,
    r.run_end_ts,
    n.TS AS next_ts,
    n.is_crit AS next_is_crit
  FROM runs r
  LEFT JOIN numbered n
    ON r.COMMUNITY_AREA = n.COMMUNITY_AREA
    AND n.rn = r.run_end_rn + 1
),
candidate_runs AS (
  SELECT *
  FROM runs_with_next
  WHERE next_is_crit = true
    AND run_len >= 3
),
matched_rows AS (
  SELECT
    c.COMMUNITY_AREA,
    c.grp,
    c.run_len,
    c.next_ts,
    c.run_end_ts,
    n.TS,
    n.rn - c.run_start_rn + 1 AS row_in_run
  FROM candidate_runs c
  JOIN numbered n
    ON c.COMMUNITY_AREA = n.COMMUNITY_AREA
    AND c.grp = n.grp
  WHERE n.is_esc = true
    AND n.rn <= c.run_end_rn - 2
    AND n.TS >= c.next_ts - 86400000
)
SELECT
  COMMUNITY_AREA,
  MIN(TS) AS ESCALATION_START_TS,
  run_end_ts AS ESCALATION_END_TS,
  next_ts AS CRITICAL_TS,
  run_len - MIN(row_in_run) + 1 AS ESCALATION_COUNT
FROM matched_rows
GROUP BY COMMUNITY_AREA, grp, next_ts, run_len, run_end_ts;

WITH base AS (

    TS,
    ID,
    PRIMARY_TYPE = 'ASSAULT' AND WARD = 48 AND ARREST = 'false' AS is_e,
    PRIMARY_TYPE = 'CRIMINAL TRESPASS' AND WARD = 1 AS is_c
  FROM CRIMES
),
marked AS (
  SELECT
    COMMUNITY_AREA,
    TS,
    ID,
    is_e,
    is_c,
    SUM(CASE WHEN is_e THEN 0 ELSE 1 END) OVER (
      PARTITION BY COMMUNITY_AREA
      ORDER BY TS, ID
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS grp
  FROM base
),
runs AS (
  SELECT
    COMMUNITY_AREA,
    TS,
    ID,
    is_e,
    is_c,
    SUM(CASE WHEN is_e THEN 1 ELSE 0 END) OVER (
      PARTITION BY COMMUNITY_AREA, grp
    ) AS e_run_len,
    MIN(CASE WHEN is_e THEN TS END) OVER (
      PARTITION BY COMMUNITY_AREA, grp
    ) AS e_run_start_ts
  FROM marked
),
lagged AS (
  SELECT
    COMMUNITY_AREA,
    TS,
    is_c,
    LAG(is_e) OVER (
      PARTITION BY COMMUNITY_AREA
      ORDER BY TS, ID
    ) AS prev_is_e,
    LAG(e_run_len) OVER (
      PARTITION BY COMMUNITY_AREA
      ORDER BY TS, ID
    ) AS prev_e_run_len,
    LAG(e_run_start_ts) OVER (
      PARTITION BY COMMUNITY_AREA
      ORDER BY TS, ID
    ) AS prev_e_run_start_ts
  FROM runs

  COMMUNITY_AREA,
  prev_e_run_start_ts AS START_TS,
  TS AS END_TS,
  prev_e_run_len AS ESCALATING_COUNT
FROM lagged
WHERE is_c
  AND prev_is_e
  AND prev_e_run_len >= 3;

WITH base AS (
    SELECT
        COMMUNITY_AREA,
        TS,
        ID,
        (PRIMARY_TYPE = 'BURGLARY' AND DISTRICT = '008' AND DOMESTIC = 'false') AS IS_E,
        (PRIMARY_TYPE = 'OFFENSE INVOLVING CHILDREN' AND DOMESTIC = 'true') AS IS_C
    FROM CRIMES
),
seq AS (
    SELECT
        COMMUNITY_AREA, TS, ID, IS_E, IS_C,
        ROW_NUMBER() OVER (PARTITION BY COMMUNITY_AREA ORDER BY TS, ID) AS RN,
        LEAD(TS) OVER (PARTITION BY COMMUNITY_AREA ORDER BY TS, ID) AS NEXT_TS,
        LEAD(IS_C) OVER (PARTITION BY COMMUNITY_AREA ORDER BY TS, ID) AS NEXT_IS_C
    FROM base
),
graded AS (
    SELECT
        COMMUNITY_AREA, TS, ID, IS_E, IS_C, RN, NEXT_TS, NEXT_IS_C,
        RN - SUM(CASE WHEN IS_E THEN 1 ELSE 0 END) OVER (
            PARTITION BY COMMUNITY_AREA
            ORDER BY TS, ID
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ) AS GRP
    FROM seq
),
runs AS (
    SELECT
        COMMUNITY_AREA,
        GRP,
        COUNT(*) AS RUN_LEN,
        MIN_BY(TS, RN) AS START_TS,
        MAX_BY(NEXT_TS, RN) AS END_TS,
        MAX_BY(NEXT_IS_C, RN) AS NEXT_IS_C
    FROM graded
    WHERE IS_E
    GROUP BY COMMUNITY_AREA, GRP
)
SELECT
    COMMUNITY_AREA,
    START_TS,
    END_TS,
    RUN_LEN AS ESCALATING_COUNT
FROM runs
WHERE RUN_LEN >= 3
  AND NEXT_IS_C;

WITH classified AS (
  SELECT
    ID,
    BEAT,
    TS,
    ROW_NUMBER() OVER (PARTITION BY BEAT ORDER BY TS, ID) AS rn,
    CASE
      WHEN PRIMARY_TYPE = 'ARSON' AND ARREST = 'true' THEN 'A'
      WHEN PRIMARY_TYPE = 'INTERFERENCE WITH PUBLIC OFFICER' AND DISTRICT = '006' THEN 'Z'
      WHEN YEAR = 2001 AND PRIMARY_TYPE = 'CRIMINAL DAMAGE' AND DISTRICT = '015' THEN 'B'
      WHEN PRIMARY_TYPE = 'ASSAULT' AND LOCATION_DESCRIPTION = 'HOSPITAL BUILDING/GROUNDS' AND ARREST = 'false' THEN 'C'
      WHEN YEAR = 2010 AND PRIMARY_TYPE = 'BATTERY' AND DISTRICT = '017' THEN 'D'
      ELSE 'O'
    END AS row_type
  FROM CRIMES
),
with_prev_next AS (
  SELECT
    *,
    MAX(CASE WHEN row_type != 'Z' THEN rn END) OVER (PARTITION BY BEAT ORDER BY rn ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) AS prev_non_z_rn,
    MIN(CASE WHEN row_type != 'Z' THEN rn END) OVER (PARTITION BY BEAT ORDER BY rn ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING) AS next_non_z_rn
  FROM classified
),
with_runs AS (
  SELECT
    *,
    SUM(CASE WHEN row_type = 'B' AND (LAG(row_type) OVER (PARTITION BY BEAT ORDER BY rn) IS DISTINCT FROM 'B') THEN 1 ELSE 0 END) OVER (PARTITION BY BEAT ORDER BY rn) AS b_run_id,
    SUM(CASE WHEN row_type = 'C' AND (LAG(row_type) OVER (PARTITION BY BEAT ORDER BY rn) IS DISTINCT FROM 'C') THEN 1 ELSE 0 END) OVER (PARTITION BY BEAT ORDER BY rn) AS c_run_id
  FROM with_prev_next
),
b_runs AS (
  SELECT
    BEAT,
    b_run_id,
    MIN(rn) AS run_start_rn,
    MAX(rn) AS run_end_rn
  FROM with_runs
  WHERE row_type = 'B'
  GROUP BY BEAT, b_run_id
),
c_runs AS (
  SELECT
    BEAT,
    c_run_id,
    MIN(rn) AS run_start_rn,
    MAX(rn) AS run_end_rn
  FROM with_runs
  WHERE row_type = 'C'
  GROUP BY BEAT, c_run_id
),
b_matches AS (
  SELECT
    br.BEAT,
    prev.ID AS START_ID,
    nxt.ID AS END_ID,
    nxt.rn - prev.rn + 1 AS MATCH_SIZE
  FROM b_runs br
  JOIN with_runs r_start ON r_start.BEAT = br.BEAT AND r_start.rn = br.run_start_rn
  JOIN with_runs r_end ON r_end.BEAT = br.BEAT AND r_end.rn = br.run_end_rn
  JOIN with_runs prev ON prev.BEAT = br.BEAT AND prev.rn = r_start.prev_non_z_rn
  JOIN with_runs nxt ON nxt.BEAT = br.BEAT AND nxt.rn = r_end.next_non_z_rn
  WHERE prev.row_type = 'A' AND nxt.row_type = 'D'
),
c_matches AS (
  SELECT
    cr.BEAT,
    prev.ID AS START_ID,
    nxt.ID AS END_ID,
    nxt.rn - prev.rn + 1 AS MATCH_SIZE
  FROM c_runs cr
  JOIN with_runs r_start ON r_start.BEAT = cr.BEAT AND r_start.rn = cr.run_start_rn
  JOIN with_runs r_end ON r_end.BEAT = cr.BEAT AND r_end.rn = cr.run_end_rn
  JOIN with_runs prev ON prev.BEAT = cr.BEAT AND prev.rn = r_start.prev_non_z_rn
  JOIN with_runs nxt ON nxt.BEAT = cr.BEAT AND nxt.rn = r_end.next_non_z_rn
  WHERE prev.row_type = 'A' AND nxt.row_type = 'D'
)
SELECT START_ID, END_ID, MATCH_SIZE FROM b_matches
UNION ALL
SELECT START_ID, END_ID, MATCH_SIZE FROM c_matches;

WITH typed AS (

    *,
    CASE
      WHEN PRIMARY_TYPE = 'ASSAULT' AND COMMUNITY_AREA = 43 THEN 'A'
      WHEN PRIMARY_TYPE = 'ROBBERY' AND DISTRICT = '011' AND ARREST = 'false' THEN 'B'
      WHEN PRIMARY_TYPE = 'BATTERY' AND DISTRICT = '006' AND ARREST = 'true' THEN 'C'
      WHEN PRIMARY_TYPE = 'DECEPTIVE PRACTICE' AND LOCATION_DESCRIPTION = 'BANK' THEN 'D'
      ELSE 'O'
    END AS type
  FROM CRIMES
),
numbered AS (
  SELECT
    *,
    ROW_NUMBER() OVER (PARTITION BY BEAT ORDER BY TS, ID) AS seq
  FROM typed

with_lag AS (
  SELECT
    numbered.*,
    LAG(type) OVER (PARTITION BY BEAT ORDER BY seq) AS prev_type
  FROM numbered
),
windowed AS (
  SELECT
    with_lag.*,
    MIN(CASE WHEN type = 'D' THEN seq END) OVER (PARTITION BY BEAT ORDER BY seq ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING) AS next_d_seq,
    LEAD(type) OVER (PARTITION BY BEAT ORDER BY seq) AS next_type,
    SUM(CASE WHEN type NOT IN ('B', 'C') THEN 1 ELSE 0 END) OVER (PARTITION BY BEAT ORDER BY seq ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS cum_invalid,
    SUM(CASE WHEN type = 'C' AND prev_type = 'C' THEN 1 ELSE 0 END) OVER (PARTITION BY BEAT ORDER BY seq ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS cum_cc,
    SUM(CASE WHEN type = 'B' THEN 1 ELSE 0 END) OVER (PARTITION BY BEAT ORDER BY seq ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS cum_B,
    SUM(CASE WHEN type = 'C' THEN 1 ELSE 0 END) OVER (PARTITION BY BEAT ORDER BY seq ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS cum_C
  FROM with_lag
),
a_rows AS (
  SELECT * FROM windowed WHERE type = 'A'
),
d_rows AS (
  SELECT * FROM windowed WHERE type = 'D'
)
SELECT
  a.BEAT,
  a.ID AS START_ID,
  d.ID AS END_ID,
  d.cum_B - a.cum_B AS B_COUNT,
  d.cum_C - a.cum_C AS C_COUNT
FROM a_rows a
JOIN d_rows d
  ON a.BEAT = d.BEAT AND a.next_d_seq = d.seq
WHERE a.next_type = 'B'
  AND d.prev_type = 'C'
  AND d.cum_invalid - a.cum_invalid = 1
  AND d.cum_cc - a.cum_cc = 0
ORDER BY a.BEAT, a.seq;

WITH ranked AS (
  SELECT BEAT,
         ID,
         PRIMARY_TYPE,
         LEAD(ID, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next1_id,
         LEAD(PRIMARY_TYPE, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next1_type,
         LEAD(ID, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next2_id,
         LEAD(PRIMARY_TYPE, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next2_type
  FROM CRIMES
)
SELECT BEAT,
       CASE WHEN PRIMARY_TYPE = 'THEFT' THEN ID
            WHEN next1_type = 'THEFT' THEN next1_id
            ELSE next2_id END AS THEFT_ID,
       CASE WHEN PRIMARY_TYPE = 'BATTERY' THEN ID
            WHEN next1_type = 'BATTERY' THEN next1_id
            ELSE next2_id END AS BATTERY_ID,
       CASE WHEN PRIMARY_TYPE = 'CRIMINAL DAMAGE' THEN ID
            WHEN next1_type = 'CRIMINAL DAMAGE' THEN next1_id
            ELSE next2_id END AS DAMAGE_ID
FROM ranked
WHERE PRIMARY_TYPE IN ('THEFT', 'BATTERY', 'CRIMINAL DAMAGE')
  AND next1_type IN ('THEFT', 'BATTERY', 'CRIMINAL DAMAGE')
  AND next2_type IN ('THEFT', 'BATTERY', 'CRIMINAL DAMAGE')
  AND PRIMARY_TYPE <> next1_type
  AND PRIMARY_TYPE <> next2_type
  AND next1_type <> next2_type;

WITH classified AS (
  SELECT
    BEAT,
    ID,
    TS,
    CASE
      WHEN PRIMARY_TYPE = 'ASSAULT' AND COMMUNITY_AREA = 28 AND ARREST = 'true' THEN 'A'
      WHEN YEAR = 2024 AND PRIMARY_TYPE = 'BATTERY' AND DISTRICT = '011' THEN 'B'
      WHEN PRIMARY_TYPE = 'OTHER OFFENSE' AND COMMUNITY_AREA = 33 AND ARREST = 'false' THEN 'C'
    END AS typ
  FROM CRIMES
),
leads AS (
  SELECT
    BEAT,
    ID,
    typ,
    LEAD(ID) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next_id,
    LEAD(typ) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next_typ,
    LEAD(ID, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next2_id,
    LEAD(typ, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next2_typ
  FROM classified
)
SELECT
  BEAT,
  CASE WHEN typ = 'A' THEN ID WHEN next_typ = 'A' THEN next_id ELSE next2_id END AS A_ID,
  CASE WHEN typ = 'B' THEN ID WHEN next_typ = 'B' THEN next_id ELSE next2_id END AS B_ID,
  CASE WHEN typ = 'C' THEN ID WHEN next_typ = 'C' THEN next_id ELSE next2_id END AS C_ID
FROM leads
WHERE typ IS NOT NULL
  AND next_typ IS NOT NULL
  AND next2_typ IS NOT NULL
  AND typ <> next_typ
  AND typ <> next2_typ
  AND next_typ <> next2_typ;

WITH tagged AS (

      WHEN PRIMARY_TYPE = 'BURGLARY' AND DISTRICT = '006' THEN 'B'
      WHEN PRIMARY_TYPE = 'CRIMINAL DAMAGE' AND WARD = 17 AND ARREST = 'false' THEN 'C'
      ELSE NULL
    END AS tag
  FROM CRIMES

leads AS (

    BEAT,

    LEAD(ID, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next_id2
  FROM tagged

    WHEN tag = 'A' THEN ID
    WHEN next_tag1 = 'A' THEN next_id1
    WHEN next_tag2 = 'A' THEN next_id2

  CASE
    WHEN tag = 'B' THEN ID
    WHEN next_tag1 = 'B' THEN next_id1
    WHEN next_tag2 = 'B' THEN next_id2

  CASE
    WHEN tag = 'C' THEN ID
    WHEN next_tag1 = 'C' THEN next_id1
    WHEN next_tag2 = 'C' THEN next_id2
  END AS C_ID

  (tag = 'A' AND next_tag1 = 'C' AND next_tag2 = 'B') OR
  (tag = 'B' AND next_tag1 = 'A' AND next_tag2 = 'C') OR
  (tag = 'B' AND next_tag1 = 'C' AND next_tag2 = 'A') OR
  (tag = 'C' AND next_tag1 = 'A' AND next_tag2 = 'B') OR
  (tag = 'C' AND next_tag1 = 'B' AND next_tag2 = 'A')

SELECT * FROM CRIMES MATCH_RECOGNIZE(
  PARTITION BY COMMUNITY_AREA
  ORDER BY TS, ID
  MEASURES FIRST(THEFT_EVENT.TS) AS THEFT_START_TS, LAST(ROBBERY_EVENT.TS) AS ROBBERY_END_TS, COUNT(THEFT_EVENT.*) AS THEFT_COUNT, COUNT(GAP.*) AS INTERVENING_COUNT, COUNT(ROBBERY_EVENT.*) AS ROBBERY_COUNT
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO NEXT ROW
  PATTERN(THEFT_EVENT+ GAP* ROBBERY_EVENT+)
  DEFINE
    THEFT_EVENT AS THEFT_EVENT.PRIMARY_TYPE IN ('THEFT', 'MOTOR VEHICLE THEFT'),
    GAP AS GAP.TS - FIRST(THEFT_EVENT.TS) <= 86400000,
    ROBBERY_EVENT AS ROBBERY_EVENT.PRIMARY_TYPE = 'ROBBERY' AND ROBBERY_EVENT.TS - FIRST(THEFT_EVENT.TS) <= 86400000
);

SELECT * FROM CRIMES MATCH_RECOGNIZE(
  PARTITION BY DISTRICT
  ORDER BY TS, ID
  MEASURES START_EVENT.ID AS START_ID, FIRST(MIDDLE_EVENT.ID) AS FIRST_MIDDLE_ID, LAST(MIDDLE_EVENT.ID) AS LAST_MIDDLE_ID, END_EVENT.ID AS END_ID, COUNT(MIDDLE_EVENT.*) AS MIDDLE_COUNT
  ONE ROW PER MATCH
  PATTERN(START_EVENT MIDDLE_EVENT{2,10} END_EVENT)
  DEFINE
    START_EVENT AS PRIMARY_TYPE = 'BURGLARY' AND ARREST = 'false',
    MIDDLE_EVENT AS PRIMARY_TYPE IN ('THEFT', 'CRIMINAL DAMAGE', 'CRIMINAL TRESPASS') AND ARREST = 'false' AND TS - START_EVENT.TS <= 604800000,
    END_EVENT AS ARREST = 'true' AND TS - START_EVENT.TS <= 604800000
);

  FROM base

    COMMUNITY_AREA,
    TS,
    IS_OVERLAP,
    RN,
    MAX(RN) OVER (PARTITION BY COMMUNITY_AREA, RID) AS RUN_END_RN,

SELECT * FROM (
  SELECT BEAT, TS, ID, PRIMARY_TYPE
  FROM CRIMES
  WHERE BEAT IN (SELECT BEAT FROM CRIMES WHERE PRIMARY_TYPE = 'HOMICIDE')
) AS t
MATCH_RECOGNIZE(
  PARTITION BY BEAT
  ORDER BY TS, ID
  MEASURES FIRST(TS) AS START_TS, HOMICIDE_EVENT.TS AS END_TS, COUNT(*) AS INCIDENT_COUNT
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO NEXT ROW
  PATTERN(PROPERTY_A* PROPERTY_B* HOMICIDE_EVENT)
  DEFINE
    PROPERTY_A AS PRIMARY_TYPE IN ('THEFT', 'BURGLARY', 'CRIMINAL DAMAGE'),
    PROPERTY_B AS PRIMARY_TYPE IN ('THEFT', 'CRIMINAL DAMAGE', 'MOTOR VEHICLE THEFT'),
    HOMICIDE_EVENT AS PRIMARY_TYPE = 'HOMICIDE'
);

SELECT * FROM CRIMES MATCH_RECOGNIZE(
  PARTITION BY COMMUNITY_AREA
  ORDER BY TS, ID
  MEASURES FIRST(TS) AS START_TS, HOMICIDE_EVENT.TS AS END_TS, COUNT(*) AS INCIDENT_COUNT
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO NEXT ROW
  PATTERN(COMMON_A* COMMON_B* COMMON_C* HOMICIDE_EVENT)
  DEFINE
    COMMON_A AS ARREST = 'false',
    COMMON_B AS DOMESTIC = 'false',
    COMMON_C AS PRIMARY_TYPE != 'HOMICIDE',
    HOMICIDE_EVENT AS PRIMARY_TYPE = 'HOMICIDE'
);

SELECT c.*,
       LAST_VALUE(c.LATITUDE) IGNORE NULLS OVER (ORDER BY c.TS, c.ID ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING) AS LAST_LATITUDE,
       LAST_VALUE(c.LONGITUDE) IGNORE NULLS OVER (ORDER BY c.TS, c.ID ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING) AS LAST_LONGITUDE
FROM CRIMES c
WHERE c.TS < 1325347200000
QUALIFY COUNT(*) OVER () >= 2;

WITH classified AS (
  SELECT
    ID,
    TS,
    BEAT,
    CASE
      WHEN PRIMARY_TYPE = 'ASSAULT' AND DISTRICT = '010' AND ARREST = 'false' THEN 'A'
      WHEN PRIMARY_TYPE = 'BATTERY' AND WARD = 10 AND ARREST = 'false' THEN 'B'
      WHEN PRIMARY_TYPE = 'OTHER OFFENSE' AND DISTRICT = '005' AND ARREST = 'false' THEN 'C'
      WHEN PRIMARY_TYPE = 'DECEPTIVE PRACTICE' AND LOCATION_DESCRIPTION = 'BANK' THEN 'D'
      ELSE 'O'
    END AS row_type
  FROM CRIMES
)
SELECT START_ID, END_ID, MATCH_SIZE
FROM classified
MATCH_RECOGNIZE(
  PARTITION BY BEAT
  ORDER BY TS, ID
  MEASURES A.ID AS START_ID, D.ID AS END_ID, COUNT(*) AS MATCH_SIZE
  PATTERN(A (B+ C?)+ D)
  DEFINE
    A AS A.row_type = 'A',
    B AS B.row_type = 'B',
    C AS C.row_type = 'C',
    D AS D.row_type = 'D'
);

WITH base AS (
  SELECT
    DISTRICT,
    TS,
    ID,
    CASE
      WHEN PRIMARY_TYPE = 'NARCOTICS' AND LOCATION_DESCRIPTION = 'SIDEWALK' THEN 'A'
      WHEN PRIMARY_TYPE = 'WEAPONS VIOLATION' AND ARREST = 'true' THEN 'B'
      WHEN PRIMARY_TYPE = 'ROBBERY' AND DOMESTIC = 'false' THEN 'C'
      WHEN PRIMARY_TYPE = 'BATTERY' AND LOCATION_DESCRIPTION = 'STREET' AND ARREST = 'false' THEN 'D'
      ELSE 'X'
    END AS ptype
  FROM CRIMES
),
segmented AS (
  SELECT
    DISTRICT,
    TS,
    ID,
    ptype,
    SUM(CASE WHEN ptype = 'X' THEN 1 ELSE 0 END) OVER (
      PARTITION BY DISTRICT
      ORDER BY TS, ID
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS segment_id
  FROM base
),
filtered AS (
  SELECT
    DISTRICT,
    TS,
    ID,
    ptype,
    segment_id,
    ROW_NUMBER() OVER (
      PARTITION BY DISTRICT, segment_id
      ORDER BY TS, ID
    ) AS rn
  FROM segmented
  WHERE ptype <> 'X'
),
with_lag AS (
  SELECT
    *,
    LAG(ptype) OVER (
      PARTITION BY DISTRICT, segment_id
      ORDER BY rn
    ) AS prev_ptype
  FROM filtered
),
cc_flagged AS (
  SELECT
    *,
    CASE WHEN ptype = 'C' AND prev_ptype = 'C' THEN rn END AS cc_rn
  FROM with_lag
),
windowed AS (
  SELECT
    *,
    LEAD(ptype) OVER (
      PARTITION BY DISTRICT, segment_id
      ORDER BY rn
    ) AS next_ptype,
    MIN(CASE WHEN ptype = 'D' THEN rn END) OVER (
      PARTITION BY DISTRICT, segment_id
      ORDER BY rn
      ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING
    ) AS next_D_rn,
    MIN(CASE WHEN ptype = 'A' THEN rn END) OVER (
      PARTITION BY DISTRICT, segment_id
      ORDER BY rn
      ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING
    ) AS next_A_rn,
    MIN(cc_rn) OVER (
      PARTITION BY DISTRICT, segment_id
      ORDER BY rn
      ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING
    ) AS next_CC_rn
  FROM cc_flagged
),
matches AS (
  SELECT
    w.DISTRICT,
    w.ID AS START_ID,
    d.ID AS END_ID,
    d.rn - w.rn + 1 AS MATCH_SIZE
  FROM windowed w
  JOIN windowed d
    ON w.DISTRICT IS NOT DISTINCT FROM d.DISTRICT
    AND w.segment_id = d.segment_id
    AND w.next_D_rn = d.rn
  WHERE w.ptype = 'A'
    AND w.next_D_rn IS NOT NULL
    AND (w.next_A_rn IS NULL OR w.next_A_rn > w.next_D_rn)
    AND w.next_D_rn > w.rn + 1
    AND w.next_ptype = 'B'
    AND (w.next_CC_rn IS NULL OR w.next_CC_rn > w.next_D_rn)
)
SELECT
  DISTRICT,
  START_ID,
  END_ID,
  MATCH_SIZE
FROM matches;

WITH typed AS (
  SELECT *,
    CASE
      WHEN PRIMARY_TYPE = 'BURGLARY' THEN 'S'
      WHEN PRIMARY_TYPE IN ('THEFT', 'CRIMINAL DAMAGE') THEN 'M'
      WHEN PRIMARY_TYPE = 'ROBBERY' THEN 'E'
      ELSE 'O'
    END AS type
  FROM CRIMES
),
filled AS (
  SELECT *,
    last_value(CASE WHEN type <> 'M' THEN type END) IGNORE NULLS OVER (PARTITION BY BLOCK ORDER BY TS, ID ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS last_non_m_type,
    last_value(CASE WHEN type <> 'M' THEN ID END) IGNORE NULLS OVER (PARTITION BY BLOCK ORDER BY TS, ID ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS last_non_m_id
  FROM typed
),
prev_filled AS (
  SELECT *,
    LAG(last_non_m_type) OVER (PARTITION BY BLOCK ORDER BY TS, ID) AS prev_last_non_m_type,
    LAG(last_non_m_id) OVER (PARTITION BY BLOCK ORDER BY TS, ID) AS prev_last_non_m_id
  FROM filled
),
matched AS (
  SELECT *,
    CASE
      WHEN type = 'S' THEN ID
      WHEN type = 'M' AND last_non_m_type = 'S' THEN last_non_m_id
      WHEN type = 'E' AND prev_last_non_m_type = 'S' THEN prev_last_non_m_id
      ELSE NULL
    END AS match_start_id
  FROM prev_filled
)
SELECT
  ID, CASE_NUMBER, TS, BLOCK, IUCR, PRIMARY_TYPE, DESCRIPTION, LOCATION_DESCRIPTION, ARREST, DOMESTIC, BEAT, DISTRICT, WARD, COMMUNITY_AREA, FBI_CODE, X_COORDINATE, Y_COORDINATE, YEAR, UPDATED_ON, LATITUDE, LONGITUDE, LOCATION,
  min(TS) OVER (PARTITION BY BLOCK, match_start_id) AS START_TS,
  max(CASE WHEN type = 'E' THEN TS END) OVER (PARTITION BY BLOCK, match_start_id) AS END_TS,
  match_start_id AS START_ID,
  max(CASE WHEN type = 'E' THEN ID END) OVER (PARTITION BY BLOCK, match_start_id) AS END_ID
FROM matched
WHERE match_start_id IS NOT NULL;

WITH base AS (
  SELECT
    DISTRICT,
    TS,
    ID,
    PRIMARY_TYPE = 'BATTERY' AND ARREST = 'false' AS is_open,
    PRIMARY_TYPE = 'BATTERY' AND ARREST = 'true' AS is_close
  FROM CRIMES
),
runs AS (
  SELECT
    DISTRICT,
    TS,
    ID,
    is_open,
    is_close,
    SUM(CASE WHEN NOT is_close THEN 1 ELSE 0 END) OVER (
      PARTITION BY DISTRICT
      ORDER BY TS, ID
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS run_id
  FROM base
),
close_runs AS (
  SELECT
    DISTRICT,
    run_id,
    COUNT(*) AS arrested_count,
    MAX(TS) AS last_arrested_ts
  FROM runs
  WHERE is_close
  GROUP BY DISTRICT, run_id
),
non_close AS (
  SELECT
    DISTRICT,
    run_id,
    is_open,
    TS,
    ID
  FROM runs
  WHERE NOT is_close
)
SELECT
  nr.DISTRICT AS DISTRICT,
  nr.TS AS TS,
  nr.ID AS ID,
  nr.TS AS FIRST_UNARRESTED_TS,
  cr.last_arrested_ts AS LAST_ARRESTED_TS,
  cr.arrested_count AS ARRESTED_COUNT
FROM close_runs cr
JOIN non_close nr
  ON cr.DISTRICT IS NOT DISTINCT FROM nr.DISTRICT
  AND cr.run_id = nr.run_id
WHERE nr.is_open = true;

WITH typed AS (
    SELECT
        BLOCK,
        ID,
        ROW_NUMBER() OVER (PARTITION BY BLOCK ORDER BY TS, ID) AS rn,
        CASE
            WHEN PRIMARY_TYPE = 'ARSON' AND LOCATION_DESCRIPTION = 'VEHICLE NON-COMMERCIAL' THEN 'S'
            WHEN YEAR = 2025 AND PRIMARY_TYPE = 'CRIMINAL SEXUAL ASSAULT' THEN 'M'
            WHEN PRIMARY_TYPE = 'WEAPONS VIOLATION' AND WARD = 7 AND ARREST = 'true' THEN 'E'
            ELSE 'O'
        END AS typ
    FROM CRIMES
),
grouped AS (
    SELECT
        BLOCK,
        ID,
        rn,
        typ,
        SUM(CASE WHEN typ <> 'M' THEN 1 ELSE 0 END) OVER (
            PARTITION BY BLOCK
            ORDER BY rn
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ) AS g
    FROM typed
),
flagged AS (
    SELECT
        BLOCK,
        ID,
        rn,
        typ,
        g,
        MAX(CASE WHEN typ = 'S' THEN 1 ELSE 0 END) OVER (PARTITION BY BLOCK, g) AS cur_flag
    FROM grouped
),
withlag AS (
    SELECT
        BLOCK,
        ID,
        rn,
        typ,
        g,
        cur_flag,
        LAG(cur_flag) OVER (PARTITION BY BLOCK ORDER BY rn) AS prev_flag
    FROM flagged
),
withmatch AS (
    SELECT
        BLOCK,
        ID,
        rn,
        typ,
        CASE
            WHEN typ = 'S' THEN g
            WHEN typ = 'M' AND cur_flag = 1 THEN g
            WHEN typ = 'E' AND prev_flag = 1 THEN g - 1
        END AS match_g
    FROM withlag
)
SELECT
    BLOCK,
    MAX(ID) FILTER (WHERE typ = 'S') AS START_ID,
    MAX_BY(ID, rn) FILTER (WHERE typ = 'M') AS LAST_MIDDLE_ID,
    MAX(ID) FILTER (WHERE typ = 'E') AS END_ID
FROM withmatch
WHERE match_g IS NOT NULL
GROUP BY BLOCK, match_g;

WITH numbered AS (
  SELECT
    BLOCK,
    ID,
    TS,
    YEAR,
    PRIMARY_TYPE,
    DISTRICT,
    DOMESTIC,
    ARREST,
    ROW_NUMBER() OVER (PARTITION BY BLOCK ORDER BY TS, ID) AS rn,
    CASE WHEN YEAR = 2003 AND PRIMARY_TYPE = 'ASSAULT' THEN 1 ELSE 0 END AS is_A,
    CASE WHEN PRIMARY_TYPE = 'BATTERY' AND DISTRICT = '001' AND DOMESTIC = 'false' THEN 1 ELSE 0 END AS is_B,
    CASE WHEN PRIMARY_TYPE = 'BURGLARY' AND DISTRICT = '025' AND ARREST = 'false' THEN 1 ELSE 0 END AS is_C
  FROM CRIMES
),
with_next AS (
  SELECT
    *,
    MIN(CASE WHEN is_B = 0 THEN rn END) OVER (
      PARTITION BY BLOCK
      ORDER BY rn
      ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING
    ) AS next_non_B_rn,
    MAX(rn) OVER (PARTITION BY BLOCK) AS max_rn
  FROM numbered
)
SELECT
  a.BLOCK,
  a.ID AS START_ID,
  m.ID AS LAST_MIDDLE_ID,
  e.ID AS END_ID
FROM with_next a
LEFT JOIN with_next m
  ON a.BLOCK = m.BLOCK
  AND m.rn = CASE
    WHEN a.next_non_B_rn IS NULL THEN
      CASE WHEN a.rn < a.max_rn THEN a.max_rn ELSE NULL END
    WHEN a.next_non_B_rn = a.rn + 1 THEN NULL
    ELSE a.next_non_B_rn - 1
  END
LEFT JOIN with_next e
  ON a.BLOCK = e.BLOCK
  AND e.rn = a.next_non_B_rn
  AND e.is_C = 1
WHERE a.is_A = 1;

WITH base AS (
  SELECT 
    BLOCK,
    TS,
    ID,
    (PRIMARY_TYPE = 'DECEPTIVE PRACTICE' AND ARREST = 'false') AS is_start,
    (PRIMARY_TYPE = 'NARCOTICS' AND COMMUNITY_AREA = 25) AS is_middle,
    (PRIMARY_TYPE = 'MOTOR VEHICLE THEFT' AND ARREST = 'false') AS is_end
  FROM CRIMES
),
mid_flags AS (
  SELECT 
    *,
    CASE WHEN NOT is_middle THEN is_start ELSE NULL END AS start_flag_for_mid
  FROM base
),
mid_matched AS (
  SELECT 
    *,
    LAST_VALUE(start_flag_for_mid) IGNORE NULLS OVER (
      PARTITION BY BLOCK 
      ORDER BY TS, ID 
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS last_start_flag
  FROM mid_flags
),
mid_matched_calc AS (
  SELECT 
    *,
    (is_middle AND last_start_flag) AS is_mid_matched
  FROM mid_matched
),
end_matched AS (
  SELECT 
    *,
    LAG(is_start) OVER (PARTITION BY BLOCK ORDER BY TS, ID) AS prev_is_start,
    LAG(is_middle) OVER (PARTITION BY BLOCK ORDER BY TS, ID) AS prev_is_middle,
    LAG(is_mid_matched) OVER (PARTITION BY BLOCK ORDER BY TS, ID) AS prev_is_mid_matched
  FROM mid_matched_calc
),
matched_rows AS (
  SELECT 
    BLOCK,
    TS,
    ID,
    is_start,
    is_middle,
    is_end,
    (is_start OR is_mid_matched OR (is_end AND (prev_is_start OR (prev_is_middle AND prev_is_mid_matched)))) AS is_matched
  FROM end_matched
),
filtered_matched AS (
  SELECT * FROM matched_rows WHERE is_matched
),
with_match_id AS (
  SELECT *,
    SUM(CASE WHEN is_start THEN 1 ELSE 0 END) OVER (
      PARTITION BY BLOCK 
      ORDER BY TS, ID 
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS match_id
  FROM filtered_matched
)
SELECT 
  BLOCK,
  MAX(CASE WHEN is_start THEN ID END) AS START_ID,
  MAX_BY(ID, ARRAY[TS, ID]) FILTER (WHERE is_middle) AS LAST_MIDDLE_ID,
  MAX(CASE WHEN is_end THEN ID END) AS END_ID
FROM with_match_id
GROUP BY BLOCK, match_id;

WITH base AS (
  SELECT
    ID,
    DISTRICT,
    TS,
    PRIMARY_TYPE = 'ASSAULT' AND WARD = 39 AS is_open,
    PRIMARY_TYPE = 'BATTERY' AND WARD = 35 AND ARREST = 'true' AS is_close
  FROM CRIMES
),
grp_calc AS (
  SELECT
    ID,
    DISTRICT,
    TS,
    is_open,
    is_close,
    SUM(CASE WHEN is_close THEN 0 ELSE 1 END) OVER (PARTITION BY DISTRICT ORDER BY TS, ID) AS grp
  FROM base
),
stats AS (
  SELECT
    ID,
    is_open,
    is_close,
    grp,
    SUM(CASE WHEN is_close THEN 1 ELSE 0 END) OVER (PARTITION BY DISTRICT, grp) AS close_count,
    LAST_VALUE(ID) OVER (PARTITION BY DISTRICT, grp ORDER BY TS, ID ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING) AS last_close_id
  FROM grp_calc
)
SELECT
  ID AS OPEN_ID,
  last_close_id AS LAST_CLOSE_ID,
  close_count AS CLOSE_COUNT
FROM stats
WHERE is_open AND close_count > 0;

WITH filtered AS (
  SELECT ID, TS, PRIMARY_TYPE, DOMESTIC, WARD
  FROM CRIMES
  WHERE DISTRICT = '008'
),
flagged AS (
  SELECT
    ID,
    TS,
    (PRIMARY_TYPE = 'BATTERY' AND DOMESTIC = 'false') AS is_open,
    COALESCE(PRIMARY_TYPE = 'THEFT' AND WARD = 44, false) AS is_close
  FROM filtered
),
seq AS (
  SELECT
    ID,
    TS,
    is_open,
    is_close,
    COALESCE(LAG(is_close) OVER (ORDER BY TS, ID), false) AS prev_close,
    COALESCE(LEAD(is_close) OVER (ORDER BY TS, ID), false) AS next_close
  FROM flagged
),
runs AS (
  SELECT
    ID,
    TS,
    is_open,
    is_close,
    next_close,
    SUM(CASE WHEN is_close AND NOT prev_close THEN 1 ELSE 0 END)
      OVER (ORDER BY TS, ID ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS crg
  FROM seq
),
match_starts AS (
  SELECT ID AS open_id, TS AS open_ts, crg + 1 AS run_crg
  FROM runs
  WHERE is_open AND next_close
),
run_end AS (
  SELECT crg, ID AS last_close_id
  FROM runs
  WHERE is_close AND NOT next_close
),
run_count AS (
  SELECT crg, COUNT(*) AS close_count
  FROM runs
  WHERE is_close
  GROUP BY crg
)
SELECT
  ms.open_id AS OPEN_ID,
  re.last_close_id AS LAST_CLOSE_ID,
  rc.close_count AS CLOSE_COUNT
FROM match_starts ms
JOIN run_count rc ON rc.crg = ms.run_crg
JOIN run_end re ON re.crg = ms.run_crg
ORDER BY ms.open_ts, ms.open_id;

WITH base AS (

    CASE WHEN PRIMARY_TYPE = 'BURGLARY' AND DOMESTIC = 'false' THEN TRUE ELSE FALSE END AS is_b
  FROM CRIMES
),
runs AS (
  SELECT
    *,
    SUM(CASE WHEN is_b AND LAG(is_b) OVER (PARTITION BY DISTRICT ORDER BY TS, ID) IS NOT TRUE THEN 1 ELSE 0 END)
      OVER (PARTITION BY DISTRICT ORDER BY TS, ID) AS run_id
  FROM base
),
with_b_run_id AS (
  SELECT
    *,
    CASE WHEN is_b THEN run_id ELSE NULL END AS b_run_id
  FROM runs

b_run_aggs AS (

    *,

  FROM with_b_run_id
),
lead_vals AS (

  FROM b_run_aggs
)
SELECT
  DISTRICT,
  ID AS OPEN_ID,
  next_run_end_id AS LAST_CLOSE_ID,
  next_run_count AS CLOSE_COUNT
FROM lead_vals
WHERE is_a AND next_is_b;

WITH ranked AS (

SELECT * FROM (
  SELECT * FROM CRIMES WHERE PRIMARY_TYPE IN ('THEFT', 'BATTERY', 'ROBBERY')
) MATCH_RECOGNIZE(
  PARTITION BY DISTRICT
  ORDER BY TS, ID
  MEASURES A.ID AS THEFT_ID, B.ID AS BATTERY_ID, C.ID AS ROBBERY_ID
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO NEXT ROW
  PATTERN(A X* B Y* C)
  DEFINE
    A AS A.PRIMARY_TYPE = 'THEFT',
    X AS X.TS - A.TS <= 86400000,
    B AS B.PRIMARY_TYPE = 'BATTERY' AND B.TS - A.TS <= 86400000,
    Y AS Y.TS - A.TS <= 86400000,
    C AS C.PRIMARY_TYPE = 'ROBBERY' AND C.TS - A.TS <= 86400000
);

SELECT
  BURGLARY_ID,
  ROBBERY_ID,
  MOTOR_VEHICLE_THEFT_ID
FROM (
  SELECT
    BEAT,
    ID,
    TS,
    PRIMARY_TYPE
  FROM CRIMES
  WHERE PRIMARY_TYPE IN ('BURGLARY', 'ROBBERY', 'MOTOR VEHICLE THEFT')
) MATCH_RECOGNIZE(
  PARTITION BY BEAT
  ORDER BY TS, ID
  MEASURES
    A.ID AS BURGLARY_ID,
    CASE WHEN BC.PRIMARY_TYPE = 'ROBBERY' THEN BC.ID END AS ROBBERY_ID,
    CASE WHEN BC.PRIMARY_TYPE = 'MOTOR VEHICLE THEFT' THEN BC.ID END AS MOTOR_VEHICLE_THEFT_ID
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO NEXT ROW
  PATTERN (A GAP* BC)
  DEFINE
    A AS A.PRIMARY_TYPE = 'BURGLARY',
    GAP AS GAP.TS - A.TS <= 3600000,
    BC AS (BC.PRIMARY_TYPE = 'ROBBERY' OR BC.PRIMARY_TYPE = 'MOTOR VEHICLE THEFT') AND BC.TS - A.TS <= 3600000
);

SELECT
  id AS ROBBERY_ID,
  next_id AS BATTERY_ID,
  next2_id AS MOTOR_VEHICLE_THEFT_ID
FROM (
  SELECT
    id,
    primary_type,
    lag(primary_type) OVER (ORDER BY ts, id) AS prev_type,
    lead(id) OVER (ORDER BY ts, id) AS next_id,
    lead(primary_type) OVER (ORDER BY ts, id) AS next_type,
    lead(ts) OVER (ORDER BY ts, id) AS next_ts,
    lead(id, 2) OVER (ORDER BY ts, id) AS next2_id,
    lead(primary_type, 2) OVER (ORDER BY ts, id) AS next2_type
  FROM CRIMES
) t
WHERE primary_type = 'ROBBERY'
  AND prev_type = 'ASSAULT'
  AND next_type = 'BATTERY'
  AND next_ts BETWEEN 1755864000000 AND 1755873000000
  AND next2_type = 'MOTOR VEHICLE THEFT';

WITH filtered AS (
  SELECT
    BEAT,
    TS,
    ID,
    CASE
      WHEN PRIMARY_TYPE = 'ARSON' AND LOCATION_DESCRIPTION = 'RESIDENCE' THEN 'A'
      WHEN PRIMARY_TYPE = 'CRIMINAL SEXUAL ASSAULT' AND DOMESTIC = 'true' THEN 'B'
      WHEN YEAR = 2016 AND PRIMARY_TYPE = 'BATTERY' AND DISTRICT = '003' THEN 'C'
    END AS role
  FROM CRIMES
  WHERE (PRIMARY_TYPE = 'ARSON' AND LOCATION_DESCRIPTION = 'RESIDENCE')
     OR (PRIMARY_TYPE = 'CRIMINAL SEXUAL ASSAULT' AND DOMESTIC = 'true')
     OR (YEAR = 2016 AND PRIMARY_TYPE = 'BATTERY' AND DISTRICT = '003')
),
with_next AS (
  SELECT
    BEAT,
    TS,
    ID,
    role,
    LEAD(CASE WHEN role = 'B' THEN ID END) IGNORE NULLS OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next_b_id,
    LEAD(CASE WHEN role = 'C' THEN ID END) IGNORE NULLS OVER (PARTITION BY BEAT ORDER BY TS, ID) AS next_c_id
  FROM filtered
)
SELECT
  a.ID AS A_ID,
  b.ID AS B_ID,
  c.ID AS C_ID
FROM with_next a
JOIN with_next b ON a.BEAT = b.BEAT AND a.next_b_id = b.ID
JOIN with_next c ON b.BEAT = c.BEAT AND b.next_c_id = c.ID
WHERE a.role = 'A'
  AND b.role = 'B'
  AND c.role = 'C';

WITH lagged AS (
  SELECT
    ID,
    PRIMARY_TYPE = 'CRIMINAL DAMAGE' AND WARD = 24 AS is_c,
    LAG(ID, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS b_id,
    LAG(ID, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS a_id,
    LAG(PRIMARY_TYPE = 'BURGLARY' AND DISTRICT = '004' AND ARREST = 'false', 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS prev_is_b,
    LAG(PRIMARY_TYPE = 'ASSAULT' AND DISTRICT = '003' AND ARREST = 'false', 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS prev2_is_a
  FROM CRIMES
)
SELECT a_id AS A_ID, b_id AS B_ID, ID AS C_ID
FROM lagged
WHERE is_c AND prev_is_b AND prev2_is_a;

WITH leads AS (
  SELECT
    DISTRICT,
    ID,
    TS,
    PRIMARY_TYPE,
    LOCATION_DESCRIPTION,
    ARREST,
    LEAD(ID, 1) OVER w AS B_ID,
    LEAD(ID, 2) OVER w AS C_ID,
    LEAD(CASE WHEN PRIMARY_TYPE = 'MOTOR VEHICLE THEFT' AND LOCATION_DESCRIPTION = 'STREET' AND ARREST = 'false' THEN 1 ELSE 0 END, 1) OVER w AS next_is_B,
    LEAD(CASE WHEN PRIMARY_TYPE = 'NARCOTICS' AND LOCATION_DESCRIPTION = 'STREET' AND ARREST = 'true' THEN 1 ELSE 0 END, 2) OVER w AS next2_is_C
  FROM CRIMES
  WINDOW w AS (PARTITION BY DISTRICT ORDER BY TS, ID)
)
SELECT
  DISTRICT,
  ID AS A_ID,
  B_ID,
  C_ID
FROM leads
WHERE PRIMARY_TYPE = 'BATTERY' AND LOCATION_DESCRIPTION = 'STREET'
  AND next_is_B = 1
  AND next2_is_C = 1;

SELECT * FROM CRIMES MATCH_RECOGNIZE(
  PARTITION BY BLOCK
  ORDER BY TS, ID
  MEASURES 
    MATCH_NUMBER() AS MATCH_NO,
    CASE WHEN CLASSIFIER() IN ('GAP_ANY', 'GAP_NON_END') THEN 'GAP' ELSE CLASSIFIER() END AS EVENT_CLASS,
    START_EVENT.TS AS START_TS,
    FINAL LAST(END_EVENT.TS) AS END_TS,
    FINAL COUNT(GAP.*) AS EXCLUDED_COUNT
  ALL ROWS PER MATCH
  AFTER MATCH SKIP PAST LAST ROW
  PATTERN(START_EVENT GAP_ANY GAP_NON_END* END_EVENT+)
  SUBSET GAP = (GAP_ANY, GAP_NON_END)
  DEFINE
    START_EVENT AS PRIMARY_TYPE = 'BURGLARY' AND ARREST = 'false',
    GAP_ANY AS TS - START_EVENT.TS <= 86400000,
    GAP_NON_END AS TS - START_EVENT.TS <= 86400000 AND NOT (PRIMARY_TYPE = 'BURGLARY' AND ARREST = 'true'),
    END_EVENT AS PRIMARY_TYPE = 'BURGLARY' AND ARREST = 'true' AND TS - START_EVENT.TS <= 86400000
);

WITH crimes_with_v AS (
  SELECT
    ID,
    CASE_NUMBER,
    TS,
    BLOCK,
    IUCR,
    PRIMARY_TYPE,
    DESCRIPTION,
    LOCATION_DESCRIPTION,
    ARREST,
    DOMESTIC,
    BEAT,
    DISTRICT,
    WARD,
    COMMUNITY_AREA,
    FBI_CODE,
    X_COORDINATE,
    Y_COORDINATE,
    YEAR,
    UPDATED_ON,
    LATITUDE,
    LONGITUDE,
    LOCATION,
    CASE WHEN PRIMARY_TYPE IN ('ASSAULT', 'BATTERY', 'ROBBERY') THEN 1 ELSE 0 END AS is_v
  FROM CRIMES
),
segments AS (
  SELECT
    *,
    SUM(CASE WHEN is_v = 1 THEN 0 ELSE 1 END) OVER (PARTITION BY DISTRICT ORDER BY TS, ID) AS grp
  FROM crimes_with_v
),
v_rows AS (
  SELECT
    *,
    ROW_NUMBER() OVER (PARTITION BY DISTRICT, grp ORDER BY TS, ID) AS v_rn
  FROM segments
  WHERE is_v = 1
),
v_end AS (
  SELECT
    *,
    MAX(v_rn) OVER (PARTITION BY DISTRICT, grp ORDER BY TS RANGE BETWEEN CURRENT ROW AND CAST(3600000 AS BIGINT) FOLLOWING) AS end_rn
  FROM v_rows
),
v_start AS (
  SELECT
    *,
    CASE WHEN v_rn > COALESCE(LAG(end_rn) OVER (PARTITION BY DISTRICT, grp ORDER BY v_rn), 0) THEN 1 ELSE 0 END AS is_start
  FROM v_end
),
v_match_start AS (
  SELECT
    *,
    CASE WHEN is_start = 1 AND end_rn > v_rn THEN 1 ELSE 0 END AS match_start_flag,
    CASE WHEN is_start = 1 AND end_rn > v_rn THEN end_rn ELSE 0 END AS match_end
  FROM v_start
),
v_match_info AS (
  SELECT
    *,
    MAX(match_end) OVER (PARTITION BY DISTRICT, grp ORDER BY v_rn ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS max_match_end,
    MAX(CASE WHEN match_start_flag = 1 THEN v_rn ELSE 0 END) OVER (PARTITION BY DISTRICT, grp ORDER BY v_rn ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS match_start_v_rn
  FROM v_match_start
),
v_final AS (
  SELECT
    ID,
    match_start_flag,
    CASE WHEN v_rn <= max_match_end THEN 1 ELSE 0 END AS in_match,
    CASE WHEN v_rn <= max_match_end THEN max_match_end - match_start_v_rn + 1 ELSE 0 END AS match_size
  FROM v_match_info
),
all_rows AS (
  SELECT
    C.*,
    COALESCE(v_final.match_start_flag, 0) AS match_start_flag,
    COALESCE(v_final.in_match, 0) AS in_match,
    COALESCE(v_final.match_size, 0) AS match_size
  FROM CRIMES C
  LEFT JOIN v_final ON C.ID = v_final.ID
)
SELECT
  ID,
  CASE_NUMBER,
  TS,
  BLOCK,
  IUCR,
  PRIMARY_TYPE,
  DESCRIPTION,
  LOCATION_DESCRIPTION,
  ARREST,
  DOMESTIC,
  BEAT,
  DISTRICT,
  WARD,
  COMMUNITY_AREA,
  FBI_CODE,
  X_COORDINATE,
  Y_COORDINATE,
  YEAR,
  UPDATED_ON,
  LATITUDE,
  LONGITUDE,
  LOCATION,
  CASE WHEN in_match = 1 THEN SUM(match_start_flag) OVER (PARTITION BY DISTRICT ORDER BY TS, ID) ELSE 0 END AS MATCH_NO,
  CASE WHEN in_match = 1 THEN 'VIOLENT_EVENT' ELSE NULL END AS EVENT_CLASS,
  match_size AS MATCH_SIZE
FROM all_rows;

WITH base AS (
  SELECT *,
         CASE WHEN PRIMARY_TYPE = 'MOTOR VEHICLE THEFT' AND COMMUNITY_AREA = 29 AND ARREST = 'false' THEN 1 ELSE 0 END AS is_v
  FROM CRIMES
),
marked AS (
  SELECT *,
         CASE WHEN is_v = 1 AND (LAG(is_v) OVER (PARTITION BY DISTRICT ORDER BY TS, ID) IS DISTINCT FROM 1) THEN 1 ELSE 0 END AS is_run_start
  FROM base
),
runs AS (
  SELECT *,
         SUM(is_run_start) OVER (PARTITION BY DISTRICT ORDER BY TS, ID ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS run_id
  FROM marked
),
violent_rows AS (
  SELECT *,
         COUNT(*) OVER (PARTITION BY DISTRICT, run_id) AS match_size
  FROM runs
  WHERE is_v = 1
),
qualifying AS (
  SELECT *,
         DENSE_RANK() OVER (PARTITION BY DISTRICT ORDER BY run_id) AS match_no
  FROM violent_rows
  WHERE match_size >= 2
)

       match_no AS MATCH_NO,

SELECT
  ID,
  CASE_NUMBER,
  TS,
  BLOCK,
  IUCR,
  PRIMARY_TYPE,
  DESCRIPTION,
  LOCATION_DESCRIPTION,
  ARREST,
  DOMESTIC,
  BEAT,
  DISTRICT,
  WARD,
  COMMUNITY_AREA,
  FBI_CODE,
  X_COORDINATE,
  Y_COORDINATE,
  YEAR,
  UPDATED_ON,
  LATITUDE,
  LONGITUDE,
  LOCATION,
  DENSE_RANK() OVER (PARTITION BY BEAT ORDER BY run_id) AS MATCH_NO,
  'VIOLENT_EVENT' AS EVENT_CLASS,
  match_size AS MATCH_SIZE
FROM (
  SELECT *
  FROM (
    SELECT *,
           SUM(is_violent) OVER (PARTITION BY BEAT, run_id) AS match_size
    FROM (
      SELECT *,
             SUM(start) OVER (PARTITION BY BEAT ORDER BY TS, ID ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS run_id
      FROM (
        SELECT *,
               CASE WHEN is_violent = 1 AND (prev IS NULL OR prev = 0) THEN 1 ELSE 0 END AS start
        FROM (
          SELECT *,
                 CASE WHEN PRIMARY_TYPE = 'BATTERY' AND DISTRICT = '025' THEN 1 ELSE 0 END AS is_violent,
                 LAG(CASE WHEN PRIMARY_TYPE = 'BATTERY' AND DISTRICT = '025' THEN 1 ELSE 0 END) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS prev
          FROM CRIMES
        ) t1
      ) t2
    ) t3
  ) t4
  WHERE is_violent = 1 AND match_size >= 2
) t5;

SELECT * FROM CRIMES MATCH_RECOGNIZE(
  PARTITION BY BLOCK, PRIMARY_TYPE
  ORDER BY TS, ID
  MEASURES CLASSIFIER() AS EVENT_CLASS, RUNNING COUNT(FOLLOWUP.*) AS FOLLOWUPS_SO_FAR, FINAL COUNT(FOLLOWUP.*) AS FOLLOWUP_TOTAL, RUNNING LAST(FOLLOWUP.TS) AS CURRENT_END_TS, FINAL LAST(FOLLOWUP.TS) AS FINAL_END_TS
  ALL ROWS PER MATCH
  AFTER MATCH SKIP PAST LAST ROW
  PATTERN(START_EVENT FOLLOWUP+)
  DEFINE
    START_EVENT AS TRUE,
    FOLLOWUP AS TS - START_EVENT.TS <= 86400000
);

SELECT * FROM CRIMES MATCH_RECOGNIZE(
  PARTITION BY BEAT
  ORDER BY TS, ID
  MEASURES FIRST(PROPERTY_EVENT.TS) AS PROPERTY_START_TS, LAST(PROPERTY_EVENT.TS) AS PROPERTY_END_TS, LAST(ARRESTED.TS) AS ARREST_END_TS, COUNT(PROPERTY_EVENT.*) AS PROPERTY_COUNT, COUNT(ARRESTED.*) AS ARREST_COUNT
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO LAST PROPERTY_EVENT
  PATTERN(PROPERTY_EVENT{2} ARRESTED+)
  DEFINE
    PROPERTY_EVENT AS PRIMARY_TYPE IN ('THEFT', 'MOTOR VEHICLE THEFT', 'CRIMINAL DAMAGE') AND ARREST = 'false',
    ARRESTED AS PRIMARY_TYPE = 'CRIMINAL DAMAGE' AND ARREST = 'true' AND TS > 1640966400000 AND TS - PROPERTY_EVENT.TS <= 86400000
);

WITH base AS (
    SELECT
        DISTRICT,
        TS,
        ID,
        CASE WHEN PRIMARY_TYPE IN ('THEFT', 'CRIMINAL DAMAGE', 'BATTERY', 'ROBBERY')
             THEN 1 ELSE 0 END AS ab,
        CASE WHEN PRIMARY_TYPE = 'HOMICIDE'
             THEN 1 ELSE 0 END AS is_homicide
    FROM CRIMES
),
ordered AS (
    SELECT
        DISTRICT,
        TS,
        ab,
        is_homicide,
        ROW_NUMBER() OVER (PARTITION BY DISTRICT ORDER BY TS, ID) AS pos,
        SUM(1 - ab) OVER (
            PARTITION BY DISTRICT
            ORDER BY TS, ID
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ) AS grp
    FROM base
)
SELECT
    o.DISTRICT AS DISTRICT,
    o.TS AS START_TS,
    t.TS AS END_TS,
    t.pos - o.pos AS AMBIGUOUS_COUNT
FROM ordered o
JOIN ordered t
    ON t.grp = o.grp + 1
   AND t.DISTRICT IS NOT DISTINCT FROM o.DISTRICT
   AND t.is_homicide = 1
WHERE o.ab = 1;
