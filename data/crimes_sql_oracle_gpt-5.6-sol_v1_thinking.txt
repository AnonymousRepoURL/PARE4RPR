  PARTITION BY DISTRICT
  ORDER BY TS, ID
  MEASURES FIRST(PRECURSOR.TS) AS PRECURSOR_START_TS, FIRST_HOMICIDE.TS AS HOMICIDE_TS, LAST(FOLLOWUP.TS) AS FOLLOWUP_END_TS, COUNT(PRECURSOR.*) AS PRECURSOR_COUNT, COUNT(FOLLOWUP.*) AS FOLLOWUP_COUNT
  ONE ROW PER MATCH
  AFTER MATCH SKIP PAST LAST ROW
  PATTERN(PRECURSOR* FIRST_HOMICIDE FOLLOWUP*)
  DEFINE
    PRECURSOR AS PRECURSOR.PRIMARY_TYPE != 'HOMICIDE',
    FIRST_HOMICIDE AS FIRST_HOMICIDE.PRIMARY_TYPE = 'HOMICIDE',
    FOLLOWUP AS FOLLOWUP.TS - FIRST_HOMICIDE.TS <= 3456000000
);

SELECT *
FROM (
  SELECT BLOCK,
         PRIMARY_TYPE,
         TS,
         ID,
         FLOOR(TS / 86400000) AS DAY_NUMBER
  FROM CRIMES
  WHERE TS > 1786118400000
)
MATCH_RECOGNIZE (
  PARTITION BY BLOCK, PRIMARY_TYPE
  ORDER BY TS, ID
  MEASURES
    FIRST(F.TS) AS FIRST_DAY_TS,
    LAST(N.TS) AS LAST_DAY_TS,
    COUNT(*) AS INCIDENT_COUNT
  ONE ROW PER MATCH
  AFTER MATCH SKIP PAST LAST ROW
  PATTERN (F S* (N S*){2,})
  SUBSET X = (F, N)
  DEFINE
    F AS F.TS > 1786118400000,
    S AS S.DAY_NUMBER = PREV(S.DAY_NUMBER),
    N AS N.DAY_NUMBER = PREV(X.DAY_NUMBER) + 1
);

SELECT
  DISTRICT,
  COUNT(CASE WHEN PRIMARY_TYPE != 'BATTERY' THEN 1 END) AS MN_NON_BATTERY_EVENT
FROM CRIMES
WHERE DISTRICT = '011'
GROUP BY DISTRICT
HAVING COUNT(CASE WHEN PRIMARY_TYPE != 'BATTERY' THEN 1 END)
       + CASE
           WHEN MAX(PRIMARY_TYPE) KEEP (DENSE_RANK LAST ORDER BY TS, ID) = 'BATTERY'
           THEN 1
           ELSE 0
         END >= 2;

FROM (
  SELECT
    BEAT,
    TS AS START_TS,
    LEAD(TS, 6) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS END_TS,
    STATE_CODE,
    LEAD(STATE_CODE, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS STATE_2,
    LEAD(STATE_CODE, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS STATE_3,
    LEAD(STATE_CODE, 3) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS STATE_4,
    LEAD(STATE_CODE, 4) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS STATE_5,
    LEAD(STATE_CODE, 5) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS STATE_6,
    LEAD(STATE_CODE, 6) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS STATE_7
  FROM (
    SELECT
      BEAT,
      TS,
      ID,
      CASE
        WHEN PRIMARY_TYPE = 'ARSON'
         AND DISTRICT = '008'
         AND ARREST = 'false' THEN 1
        WHEN PRIMARY_TYPE = 'CRIM SEXUAL ASSAULT'
         AND DISTRICT = '019' THEN 2
        WHEN YEAR = 2017
         AND PRIMARY_TYPE = 'BURGLARY'
         AND DISTRICT = '008' THEN 3
        WHEN YEAR = 2011
         AND PRIMARY_TYPE = 'NARCOTICS'
         AND DISTRICT = '012' THEN 4
        WHEN YEAR = 2012
         AND PRIMARY_TYPE = 'CRIMINAL DAMAGE'
         AND DISTRICT = '017' THEN 5
        WHEN YEAR = 2006
         AND PRIMARY_TYPE = 'OTHER OFFENSE'
         AND DISTRICT = '010' THEN 6
        WHEN YEAR = 2004
         AND PRIMARY_TYPE = 'ROBBERY'
         AND DISTRICT = '006' THEN 7
      END AS STATE_CODE
    FROM CRIMES
  )
)
WHERE STATE_CODE = 1
  AND STATE_2 = 2
  AND STATE_3 = 3
  AND STATE_4 = 4
  AND STATE_5 = 5
  AND STATE_6 = 6
  AND STATE_7 = 7;

       ts AS start_ts,
       end_ts,
       7 AS match_size
FROM (
    SELECT beat,
           ts,
           pattern_code,
           LEAD(pattern_code, 1) OVER (PARTITION BY beat ORDER BY ts, id) AS code_2,
           LEAD(pattern_code, 2) OVER (PARTITION BY beat ORDER BY ts, id) AS code_3,
           LEAD(pattern_code, 3) OVER (PARTITION BY beat ORDER BY ts, id) AS code_4,
           LEAD(pattern_code, 4) OVER (PARTITION BY beat ORDER BY ts, id) AS code_5,
           LEAD(pattern_code, 5) OVER (PARTITION BY beat ORDER BY ts, id) AS code_6,
           LEAD(pattern_code, 6) OVER (PARTITION BY beat ORDER BY ts, id) AS code_7,
           LEAD(ts, 6) OVER (PARTITION BY beat ORDER BY ts, id) AS end_ts
    FROM (
        SELECT beat,
               ts,
               id,
               CASE
                   WHEN primary_type = 'ASSAULT'
                    AND location_description = 'RESIDENCE' THEN 1
                   WHEN primary_type = 'CRIMINAL DAMAGE'
                    AND location_description = 'RESIDENCE' THEN 2
                   WHEN primary_type = 'OTHER OFFENSE'
                    AND arrest = 'true' THEN 3
                   WHEN primary_type = 'BURGLARY'

                   WHEN primary_type = 'THEFT'

                   WHEN primary_type = 'WEAPONS VIOLATION'
                    AND domestic = 'false' THEN 6
                   WHEN primary_type = 'BATTERY'
                    AND district = '007' THEN 7
                   ELSE 0
               END AS pattern_code
        FROM crimes
    )
)
WHERE pattern_code = 1
  AND code_2 = 2
  AND code_3 = 3
  AND code_4 = 4
  AND code_5 = 5
  AND code_6 = 6
  AND code_7 = 7;

MATCH_RECOGNIZE (

  PATTERN (A X* B X* C)

WITH classified AS (
  SELECT
    id,
    ts,
    CASE
      WHEN latitude BETWEEN 41.6994435912654 AND 41.7146453168236
       AND longitude BETWEEN -87.6587416716118 AND -87.6330873595214
      THEN
        CASE primary_type
          WHEN 'ROBBERY' THEN 1
          WHEN 'BATTERY' THEN 2
          WHEN 'MOTOR VEHICLE THEFT' THEN 3
        END
    END AS event_type
  FROM crimes
),
previous_battery AS (
  SELECT
    id,
    ts,
    event_type,
    LAST_VALUE(CASE WHEN event_type = 2 THEN ts END) IGNORE NULLS OVER (
      ORDER BY ts, id
      ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
    ) AS previous_battery_ts,
    LAST_VALUE(CASE WHEN event_type = 2 THEN id END) IGNORE NULLS OVER (
      ORDER BY ts, id
      ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING
    ) AS previous_battery_id
  FROM classified
),
candidate_matches AS (
  SELECT
    id,
    ts,
    event_type,
    LAST_VALUE(
      CASE WHEN event_type = 3 THEN previous_battery_ts END
    ) IGNORE NULLS OVER (
      ORDER BY ts, id
      ROWS BETWEEN 1 FOLLOWING AND 49 FOLLOWING
    ) AS battery_ts,
    LAST_VALUE(
      CASE WHEN event_type = 3 THEN previous_battery_id END
    ) IGNORE NULLS OVER (
      ORDER BY ts, id
      ROWS BETWEEN 1 FOLLOWING AND 49 FOLLOWING
    ) AS battery_id,
    LAST_VALUE(
      CASE WHEN event_type = 3 THEN id END
    ) IGNORE NULLS OVER (
      ORDER BY ts, id
      ROWS BETWEEN 1 FOLLOWING AND 49 FOLLOWING
    ) AS motor_vehicle_theft_id
  FROM previous_battery
)
SELECT
  id AS robbery_id,
  battery_id,
  motor_vehicle_theft_id
FROM candidate_matches
WHERE event_type = 1
  AND (
    battery_ts > ts
    OR (battery_ts = ts AND battery_id > id)
  );

SELECT
  BEAT,
  CASE
    WHEN PRIMARY_TYPE = 'THEFT' THEN ID
    WHEN NEXT_PRIMARY_TYPE = 'THEFT' THEN NEXT_ID
    ELSE THIRD_ID
  END AS THEFT_ID,
  CASE
    WHEN PRIMARY_TYPE = 'BATTERY' THEN ID
    WHEN NEXT_PRIMARY_TYPE = 'BATTERY' THEN NEXT_ID
    ELSE THIRD_ID
  END AS BATTERY_ID,
  CASE
    WHEN PRIMARY_TYPE = 'CRIMINAL DAMAGE' THEN ID
    WHEN NEXT_PRIMARY_TYPE = 'CRIMINAL DAMAGE' THEN NEXT_ID
    ELSE THIRD_ID
  END AS DAMAGE_ID
FROM (
  SELECT
    BEAT,
    ID,
    PRIMARY_TYPE,
    LEAD(ID, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS NEXT_ID,
    LEAD(ID, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS THIRD_ID,
    LEAD(PRIMARY_TYPE, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS NEXT_PRIMARY_TYPE,
    LEAD(PRIMARY_TYPE, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS THIRD_PRIMARY_TYPE
  FROM CRIMES
)
WHERE PRIMARY_TYPE IN ('THEFT', 'BATTERY', 'CRIMINAL DAMAGE')
  AND NEXT_PRIMARY_TYPE IN ('THEFT', 'BATTERY', 'CRIMINAL DAMAGE')
  AND THIRD_PRIMARY_TYPE IN ('THEFT', 'BATTERY', 'CRIMINAL DAMAGE')
  AND PRIMARY_TYPE <> NEXT_PRIMARY_TYPE
  AND PRIMARY_TYPE <> THIRD_PRIMARY_TYPE
  AND NEXT_PRIMARY_TYPE <> THIRD_PRIMARY_TYPE;

SELECT BEAT,
       CASE
         WHEN CLS = 'A' THEN ID
         WHEN CLS_1 = 'A' THEN ID_1
         ELSE ID_2
       END AS A_ID,
       CASE
         WHEN CLS = 'B' THEN ID
         WHEN CLS_1 = 'B' THEN ID_1
         ELSE ID_2
       END AS B_ID,
       CASE
         WHEN CLS = 'C' THEN ID
         WHEN CLS_1 = 'C' THEN ID_1
         ELSE ID_2
       END AS C_ID
FROM (
  SELECT BEAT,
         ID,
         CLS,
         LEAD(ID, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS ID_1,
         LEAD(ID, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS ID_2,
         LEAD(CLS, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS CLS_1,
         LEAD(CLS, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS CLS_2
  FROM (
    SELECT BEAT,
           TS,
           ID,
           CASE
             WHEN PRIMARY_TYPE = 'ASSAULT'
              AND COMMUNITY_AREA = 28
              AND ARREST = 'true'
             THEN 'A'
             WHEN YEAR = 2024
              AND PRIMARY_TYPE = 'BATTERY'
              AND DISTRICT = '011'
             THEN 'B'
             WHEN PRIMARY_TYPE = 'OTHER OFFENSE'
              AND COMMUNITY_AREA = 33
              AND ARREST = 'false'
             THEN 'C'
           END AS CLS
    FROM CRIMES
  )
)
WHERE CLS IS NOT NULL
  AND CLS_1 IS NOT NULL
  AND CLS_2 IS NOT NULL
  AND CLS <> CLS_1
  AND CLS <> CLS_2
  AND CLS_1 <> CLS_2;

                  AND OPEN_EVENT.DOMESTIC = 'false',
    CLOSE_EVENT AS CLOSE_EVENT.PRIMARY_TYPE = 'THEFT'
                   AND CLOSE_EVENT.WARD = 44
);

SELECT *
FROM (
  SELECT DISTRICT, TS, ID, PRIMARY_TYPE
  FROM CRIMES
  WHERE PRIMARY_TYPE IN ('THEFT', 'BATTERY', 'ROBBERY')
)
MATCH_RECOGNIZE (
  PARTITION BY DISTRICT
  ORDER BY TS, ID
  MEASURES
    A.ID AS THEFT_ID,
    B.ID AS BATTERY_ID,
    C.ID AS ROBBERY_ID
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO NEXT ROW
  PATTERN (A X* B Y* C)
  DEFINE
    A AS A.PRIMARY_TYPE = 'THEFT',
    X AS X.TS - A.TS <= 86400000,
    B AS B.PRIMARY_TYPE = 'BATTERY'
         AND B.TS - A.TS <= 86400000,
    Y AS Y.TS - A.TS <= 86400000,
    C AS C.PRIMARY_TYPE = 'ROBBERY'
         AND C.TS - A.TS <= 86400000
);

SELECT *
FROM CRIMES MATCH_RECOGNIZE(
  PARTITION BY BEAT
  ORDER BY TS, ID
  MEASURES
    A.ID AS BURGLARY_ID,
    CASE WHEN D.PRIMARY_TYPE = 'ROBBERY' THEN D.ID END AS ROBBERY_ID,
    CASE WHEN D.PRIMARY_TYPE = 'MOTOR VEHICLE THEFT' THEN D.ID END AS MOTOR_VEHICLE_THEFT_ID
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO NEXT ROW
  PATTERN(A {- GAP* -} D)
  DEFINE
    A AS A.PRIMARY_TYPE = 'BURGLARY',
    GAP AS GAP.TS - A.TS <= 3600000,
    D AS D.PRIMARY_TYPE IN ('ROBBERY', 'MOTOR VEHICLE THEFT')
         AND D.TS - A.TS <= 3600000
);

SELECT ID AS ROBBERY_ID,
       BATTERY_ID,
       MOTOR_VEHICLE_THEFT_ID
FROM (
  SELECT ID,
         PRIMARY_TYPE,
         LAG(PRIMARY_TYPE) OVER (ORDER BY TS, ID) AS PREV_PRIMARY_TYPE,
         LEAD(ID) OVER (ORDER BY TS, ID) AS BATTERY_ID,
         LEAD(PRIMARY_TYPE) OVER (ORDER BY TS, ID) AS BATTERY_TYPE,
         LEAD(TS) OVER (ORDER BY TS, ID) AS BATTERY_TS,
         LEAD(ID, 2) OVER (ORDER BY TS, ID) AS MOTOR_VEHICLE_THEFT_ID,
         LEAD(PRIMARY_TYPE, 2) OVER (ORDER BY TS, ID) AS MOTOR_VEHICLE_THEFT_TYPE
  FROM CRIMES
)
WHERE PRIMARY_TYPE = 'ROBBERY'
  AND PREV_PRIMARY_TYPE = 'ASSAULT'
  AND BATTERY_TYPE = 'BATTERY'
  AND BATTERY_TS BETWEEN 1755864000000 AND 1755873000000
  AND MOTOR_VEHICLE_THEFT_TYPE = 'MOTOR VEHICLE THEFT';

SELECT BEAT, A_ID, B_ID, C_ID
FROM (
  SELECT
    BEAT,
    LAG(
      CASE
        WHEN PRIMARY_TYPE = 'ARSON'
         AND LOCATION_DESCRIPTION = 'RESIDENCE'
        THEN ID
      END,
      2
    ) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS A_ID,
    LAG(
      CASE
        WHEN PRIMARY_TYPE = 'CRIMINAL SEXUAL ASSAULT'
         AND DOMESTIC = 'true'
        THEN ID
      END
    ) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS B_ID,
    CASE
      WHEN YEAR = 2016
       AND PRIMARY_TYPE = 'BATTERY'
       AND DISTRICT = '003'
      THEN ID
    END AS C_ID
  FROM CRIMES
)
WHERE A_ID IS NOT NULL
  AND B_ID IS NOT NULL
  AND C_ID IS NOT NULL;

SELECT BEAT, ID AS A_ID, B_ID, C_ID
FROM (
  SELECT
    BEAT,
    ID,
    PRIMARY_TYPE,
    DISTRICT,
    ARREST,
    LEAD(ID, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS B_ID,
    LEAD(PRIMARY_TYPE, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS B_PRIMARY_TYPE,
    LEAD(DISTRICT, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS B_DISTRICT,
    LEAD(ARREST, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS B_ARREST,
    LEAD(ID, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS C_ID,
    LEAD(PRIMARY_TYPE, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS C_PRIMARY_TYPE,
    LEAD(WARD, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS C_WARD
  FROM CRIMES
)
WHERE PRIMARY_TYPE = 'ASSAULT'
  AND DISTRICT = '003'
  AND ARREST = 'false'
  AND B_PRIMARY_TYPE = 'BURGLARY'
  AND B_DISTRICT = '004'
  AND B_ARREST = 'false'
  AND C_PRIMARY_TYPE = 'CRIMINAL DAMAGE'
  AND C_WARD = 24;

SELECT DISTRICT, A_ID, B_ID, C_ID
FROM (
  SELECT
    DISTRICT,
    CASE
      WHEN PRIMARY_TYPE = 'BATTERY'
       AND LOCATION_DESCRIPTION = 'STREET'
      THEN ID
    END AS A_ID,
    LEAD(
      CASE
        WHEN PRIMARY_TYPE = 'MOTOR VEHICLE THEFT'
         AND LOCATION_DESCRIPTION = 'STREET'
         AND ARREST = 'false'
        THEN ID
      END,
      1
    ) OVER (PARTITION BY DISTRICT ORDER BY TS, ID) AS B_ID,
    LEAD(
      CASE
        WHEN PRIMARY_TYPE = 'NARCOTICS'
         AND LOCATION_DESCRIPTION = 'STREET'
         AND ARREST = 'true'
        THEN ID
      END,
      2
    ) OVER (PARTITION BY DISTRICT ORDER BY TS, ID) AS C_ID
  FROM CRIMES
)
WHERE A_ID IS NOT NULL
  AND B_ID IS NOT NULL
  AND C_ID IS NOT NULL;

    FIRST(PROPERTY_EVENT.TS) AS PROPERTY_START_TS,

    LAST(ARRESTED.TS) AS ARREST_END_TS,
    COUNT(PROPERTY_EVENT.*) AS PROPERTY_COUNT,
    COUNT(ARRESTED.*) AS ARREST_COUNT
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO LAST PROPERTY_EVENT
  PATTERN(PROPERTY_EVENT{2} ARRESTED+)
  DEFINE
    PROPERTY_EVENT AS PRIMARY_TYPE IN ('THEFT', 'MOTOR VEHICLE THEFT', 'CRIMINAL DAMAGE')
                      AND ARREST = 'false',
    ARRESTED AS PRIMARY_TYPE = 'CRIMINAL DAMAGE'
                AND ARREST = 'true'
                AND TS > 1640966400000
                AND TS - PROPERTY_EVENT.TS <= 86400000
);
