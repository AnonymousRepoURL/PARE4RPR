SELECT START_TS, BOTTOM_TS, END_TS, START_PRICE, BOTTOM_PRICE, END_PRICE
FROM (
  SELECT SYMBOL, TS, PRICE, TRADE_ID
  FROM NASDAQ
  WHERE TS >= 1722441600000000000
) AS N
MATCH_RECOGNIZE(
  PARTITION BY SYMBOL
  ORDER BY TS, TRADE_ID
  MEASURES
    STRT.TS AS START_TS,
    LAST(DOWN.TS) AS BOTTOM_TS,
    LAST(UP.TS) AS END_TS,
    STRT.PRICE AS START_PRICE,
    MIN(DOWN.PRICE) AS BOTTOM_PRICE,
    MAX(UP.PRICE) AS END_PRICE
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO LAST UP
  PATTERN (STRT DOWN+ UP+)
  DEFINE
    STRT AS STRT.TS BETWEEN 1722441600000000000 AND 1722528000000000000,
    DOWN AS DOWN.PRICE < PREV(DOWN.PRICE),
    UP AS UP.PRICE > PREV(UP.PRICE)
);

SELECT *
FROM NASDAQ MATCH_RECOGNIZE(
  PARTITION BY SYMBOL
  ORDER BY TS, TRADE_ID
  MEASURES
    A.TS AS START_TS,
    A.PRICE AS START_PRICE,
    LAST(B.PRICE) AS BOTTOM_PRICE,
    LAST(D.PRICE) AS RECOVERY_PRICE,
    COUNT(*) AS TRADE_COUNT
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO NEXT ROW
  PATTERN (A B+ C D+ E)
  DEFINE
    A AS PRICE >= 200,
    B AS PRICE < PREV(PRICE),
    C AS PRICE > PREV(PRICE) AND PRICE <= A.PRICE,
    D AS PRICE > PREV(PRICE),
    E AS PRICE <= PREV(PRICE) AND COUNT(*) <= 20
);

WITH CLASSIFIED AS (
    SELECT
        SYMBOL,
        EXCHANGE,
        TS,
        TRADE_ID,
        PRICE < 200 AS A_OK,
        PRICE BETWEEN 150 AND 250 AS B_OK,
        PRICE > 300 AND VOLUME >= 5000 AS C_OK
    FROM NASDAQ
),
SEGMENTED AS (
    SELECT
        *,
        SUM(CASE WHEN NOT (A_OK OR B_OK) THEN 1 ELSE 0 END) OVER (
            PARTITION BY SYMBOL, EXCHANGE
            ORDER BY TS, TRADE_ID
            ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING
        ) AS SEGMENT_ID
    FROM CLASSIFIED
),
MARKED AS (
    SELECT
        *,
        MAX(CASE WHEN C_OK THEN TS END) OVER (
            PARTITION BY SYMBOL, EXCHANGE, SEGMENT_ID
        ) AS END_TS
    FROM SEGMENTED
),
CANDIDATE_SEGMENTS AS (
    SELECT *
    FROM MARKED
    WHERE END_TS IS NOT NULL
),
NUMBERED AS (
    SELECT
        *,
        ROW_NUMBER() OVER (
            PARTITION BY SYMBOL, EXCHANGE, SEGMENT_ID
            ORDER BY TS, TRADE_ID
        ) AS RN,
        COUNT(*) OVER (
            PARTITION BY SYMBOL, EXCHANGE, SEGMENT_ID
        ) AS GROUP_SIZE
    FROM CANDIDATE_SEGMENTS
),
LOW_MARKED AS (
    SELECT
        *,
        MAX(CASE WHEN A_OK AND NOT B_OK THEN RN END) OVER (
            PARTITION BY SYMBOL, EXCHANGE, SEGMENT_ID
        ) AS LAST_A_ONLY_RN
    FROM NUMBERED
),
WITH_CUTOFF AS (
    SELECT
        *,
        MAX(
            CASE
                WHEN B_OK AND NOT A_OK AND RN < LAST_A_ONLY_RN THEN RN
            END
        ) OVER (
            PARTITION BY SYMBOL, EXCHANGE, SEGMENT_ID
        ) AS CUTOFF_RN
    FROM LOW_MARKED
)
SELECT
    SYMBOL,
    EXCHANGE,
    CASE WHEN RN <= LAST_A_ONLY_RN THEN TS END AS START_TS,
    END_TS,
    GROUP_SIZE - RN + 1 AS RELUCTANT_ROWS
FROM WITH_CUTOFF
WHERE RN > COALESCE(CUTOFF_RN, 0);

WITH classified AS (
    SELECT
        SYMBOL,
        TS,
        TRADE_ID,
        CASE
            WHEN CONDITIONS IS NOT NULL AND VOLUME <= 500 THEN 1
            WHEN CONDITIONS IS NULL AND VOLUME >= 5000 THEN 2
            ELSE 0
        END AS row_class
    FROM NASDAQ
),
partitions AS (
    SELECT
        SYMBOL,
        min(TS) FILTER (WHERE row_class = 1) AS START_TS,
        max(TS) FILTER (WHERE row_class = 2) AS END_TS,
        count_if(row_class = 1) AS PARTITION_ROWS,
        count_if(row_class = 2) AS b_rows,
        count_if(row_class = 0) AS invalid_rows,
        max(ROW(TS, TRADE_ID)) FILTER (WHERE row_class = 1) AS last_a,
        min(ROW(TS, TRADE_ID)) FILTER (WHERE row_class = 2) AS first_b
    FROM classified
    GROUP BY SYMBOL
)
SELECT
    SYMBOL,
    START_TS,
    END_TS,
    PARTITION_ROWS
FROM partitions
WHERE invalid_rows = 0
  AND PARTITION_ROWS <= 100
  AND b_rows BETWEEN 5 AND 100
  AND (PARTITION_ROWS = 0 OR last_a < first_b);

        n.*,

            PARTITION BY SYMBOL, EXCHANGE
            ORDER BY TS, TRADE_ID

            PARTITION BY SYMBOL, EXCHANGE
            ORDER BY TS, TRADE_ID

    FROM NASDAQ AS n
)

    SYMBOL,
    TS_ISO,
    TS,
    PRICE,
    VOLUME,

WHERE is_matching_row;

    WINDOW w AS (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID)
)
SELECT
    SYMBOL,
    START_TS,
    C_TS AS RANGE_TS,
    E_TS AS END_TS
FROM sequenced
CROSS JOIN UNNEST(
    IF(
        PREV_EXCHANGE = 'D' AND PREV_FEED <> 'sip',
        ARRAY[PREV_TS, CAST(NULL AS BIGINT)],
        ARRAY[CAST(NULL AS BIGINT)]
    )
) AS starts (START_TS)
WHERE EXCHANGE = 'D'
  AND FEED = 'sip'
  AND TS >= 1722450837513686408
  AND C_TS < 1722515175948760483
  AND D_FEED = 'sip'
  AND E_EXCHANGE IN ('Q', 'P', 'Z', 'K');
