WITH BUCKETED_ROWS AS(
    SELECT SYMBOL, TS_ISO, TS, PRICE, VOLUME, EXCHANGE, CONDITIONS, TRADE_ID, TAPE, FEED, FLOOR(TS / 1000) AS BK_NO
    FROM (SELECT * FROM NASDAQ WHERE SYMBOL IN (SELECT SYMBOL FROM NASDAQ GROUP BY SYMBOL HAVING MAX(CASE WHEN VOLUME >= 5000 THEN 1 ELSE 0 END) = 1))
), BK_NO_LIST AS(
  SELECT SYMBOL, BK_NO
  FROM BUCKETED_ROWS
  WHERE VOLUME >= 5000
   UNION
   SELECT SYMBOL, BK_NO + 1 AS BK_NO
  FROM BUCKETED_ROWS
  WHERE VOLUME >= 5000
), FILTERED_ROWS AS(
    SELECT SYMBOL, TS_ISO, TS, PRICE, VOLUME, EXCHANGE, CONDITIONS, TRADE_ID, TAPE, FEED
    FROM BUCKETED_ROWS PREFILTER_INPUT
    WHERE EXISTS (
        SELECT 1 FROM BK_NO_LIST PREFILTER_BUCKET
        WHERE PREFILTER_INPUT.BK_NO = PREFILTER_BUCKET.BK_NO AND (PREFILTER_INPUT.SYMBOL = PREFILTER_BUCKET.SYMBOL OR (PREFILTER_INPUT.SYMBOL IS NULL AND PREFILTER_BUCKET.SYMBOL IS NULL))
    )
)

FROM (SELECT SYMBOL, TS, TRADE_ID, PRICE, CASE WHEN VOLUME >= 5000 THEN 'a' WHEN VOLUME <= 999 THEN 'b' ELSE 'c' END AS TAG FROM FILTERED_ROWS)
