SELECT MR.REGION AS REGION, MR.TS AS TS, MR.ID AS ID, T_FULL.A1 AS A1, T_FULL.A2 AS A2, T_FULL.A3 AS A3
FROM (SELECT ID, REGION, TS, CASE WHEN (A1 BETWEEN 100 AND 500) THEN 'a' ELSE 'b' END AS TAG, CASE WHEN (A2 BETWEEN 300 AND 700) THEN 'a' ELSE 'b' END AS TAG_1 FROM SYN10M)
MATCH_RECOGNIZE(
    PARTITION BY REGION
    ORDER BY TS, ID
    MEASURES ID AS RPR_JOIN_PK_1, MATCH_NUMBER() AS RPR_MATCH_NO
    ALL ROWS PER MATCH
    PATTERN(A B{1,5} C D{1,5} E F+)
    DEFINE A AS TAG = 'a', B AS TAG_1 = 'a', C AS TAG = 'a', D AS TAG_1 = 'a', E AS TAG = 'a', F AS COUNT(*) <= 10
) MR
LEFT JOIN SYN10M T_FULL
ON T_FULL.ID = MR.RPR_JOIN_PK_1
ORDER BY MR.REGION, MR.RPR_MATCH_NO, MR.TS, MR.ID;
