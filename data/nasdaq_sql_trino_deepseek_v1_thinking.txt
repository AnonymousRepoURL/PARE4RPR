FROM (
  SELECT * FROM NASDAQ WHERE TS >= 1722441600000000000
) MATCH_RECOGNIZE(
  PARTITION BY SYMBOL
  ORDER BY TS, TRADE_ID
  MEASURES STRT.TS AS START_TS, LAST(DOWN.TS) AS BOTTOM_TS, LAST(UP.TS) AS END_TS, STRT.PRICE AS START_PRICE, LAST(DOWN.PRICE) AS BOTTOM_PRICE, LAST(UP.PRICE) AS END_PRICE
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO LAST UP
  PATTERN(STRT DOWN+ UP+)
  DEFINE
    STRT AS STRT.TS <= 1722528000000000000,
    DOWN AS DOWN.PRICE < PREV(DOWN.PRICE),
    UP AS UP.PRICE > PREV(UP.PRICE)
);

WITH ordered AS (
  SELECT
    SYMBOL,
    TS,
    VOLUME,
    ROW_NUMBER() OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS rn,
    MIN(TS) OVER (PARTITION BY SYMBOL) AS first_ts
  FROM NASDAQ
),
capped AS (
  SELECT
    SYMBOL,
    TS,
    VOLUME,
    rn,
    first_ts,
    VOLUME < 100 AND TS - first_ts <= 1000000000 AS is_open
  FROM ordered
  WHERE rn <= 10
),
prefixed AS (
  SELECT
    *,
    BOOL_AND(is_open) OVER (
      PARTITION BY SYMBOL
      ORDER BY rn
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS valid_prefix
  FROM capped
),
with_L AS (
  SELECT
    *,
    MAX(CASE WHEN valid_prefix THEN rn END) OVER (PARTITION BY SYMBOL) AS L
  FROM prefixed
)
SELECT
  SYMBOL,
  MAX(first_ts) AS FIRST_TS,
  MAX(TS) AS FIFTH_TS,
  SUM(VOLUME) AS OPENING_VOLUME
FROM with_L
WHERE L >= 5 AND rn <= L
GROUP BY SYMBOL;

WITH ordered AS (
  SELECT
    SYMBOL,
    TS,
    TRADE_ID,
    PRICE,
    VOLUME,
    CASE WHEN PRICE BETWEEN 100 AND 200 AND TS >= 1722441600000000000 THEN 1 ELSE 0 END AS is_base
  FROM NASDAQ
),
window_arrays AS (
  SELECT
    SYMBOL,
    TS,
    TRADE_ID,
    PRICE,
    VOLUME,
    is_base,
    array_agg(TS) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID ROWS BETWEEN CURRENT ROW AND 10 FOLLOWING) AS ts_arr,
    array_agg(CAST(PRICE AS DOUBLE)) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID ROWS BETWEEN CURRENT ROW AND 10 FOLLOWING) AS price_arr,
    array_agg(CAST(VOLUME AS DOUBLE)) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID ROWS BETWEEN CURRENT ROW AND 10 FOLLOWING) AS vol_arr,
    array_agg(is_base) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID ROWS BETWEEN CURRENT ROW AND 10 FOLLOWING) AS base_arr
  FROM ordered
),
matches AS (
  SELECT
    *,
    CASE
      WHEN array_sum(slice(base_arr, 1, 10)) = 10 AND element_at(ts_arr, 11) IS NOT NULL AND element_at(price_arr, 11) > array_sum(slice(price_arr, 1, 10)) / 10 * 1.01 AND element_at(vol_arr, 11) > array_sum(slice(vol_arr, 1, 10)) / 10 * 5 AND element_at(ts_arr, 11) - TS <= 60000000000 THEN 10
      WHEN array_sum(slice(base_arr, 1, 9)) = 9 AND element_at(ts_arr, 10) IS NOT NULL AND element_at(price_arr, 10) > array_sum(slice(price_arr, 1, 9)) / 9 * 1.01 AND element_at(vol_arr, 10) > array_sum(slice(vol_arr, 1, 9)) / 9 * 5 AND element_at(ts_arr, 10) - TS <= 60000000000 THEN 9
      WHEN array_sum(slice(base_arr, 1, 8)) = 8 AND element_at(ts_arr, 9) IS NOT NULL AND element_at(price_arr, 9) > array_sum(slice(price_arr, 1, 8)) / 8 * 1.01 AND element_at(vol_arr, 9) > array_sum(slice(vol_arr, 1, 8)) / 8 * 5 AND element_at(ts_arr, 9) - TS <= 60000000000 THEN 8
      WHEN array_sum(slice(base_arr, 1, 7)) = 7 AND element_at(ts_arr, 8) IS NOT NULL AND element_at(price_arr, 8) > array_sum(slice(price_arr, 1, 7)) / 7 * 1.01 AND element_at(vol_arr, 8) > array_sum(slice(vol_arr, 1, 7)) / 7 * 5 AND element_at(ts_arr, 8) - TS <= 60000000000 THEN 7
      WHEN array_sum(slice(base_arr, 1, 6)) = 6 AND element_at(ts_arr, 7) IS NOT NULL AND element_at(price_arr, 7) > array_sum(slice(price_arr, 1, 6)) / 6 * 1.01 AND element_at(vol_arr, 7) > array_sum(slice(vol_arr, 1, 6)) / 6 * 5 AND element_at(ts_arr, 7) - TS <= 60000000000 THEN 6
      WHEN array_sum(slice(base_arr, 1, 5)) = 5 AND element_at(ts_arr, 6) IS NOT NULL AND element_at(price_arr, 6) > array_sum(slice(price_arr, 1, 5)) / 5 * 1.01 AND element_at(vol_arr, 6) > array_sum(slice(vol_arr, 1, 5)) / 5 * 5 AND element_at(ts_arr, 6) - TS <= 60000000000 THEN 5
      WHEN array_sum(slice(base_arr, 1, 4)) = 4 AND element_at(ts_arr, 5) IS NOT NULL AND element_at(price_arr, 5) > array_sum(slice(price_arr, 1, 4)) / 4 * 1.01 AND element_at(vol_arr, 5) > array_sum(slice(vol_arr, 1, 4)) / 4 * 5 AND element_at(ts_arr, 5) - TS <= 60000000000 THEN 4
      WHEN array_sum(slice(base_arr, 1, 3)) = 3 AND element_at(ts_arr, 4) IS NOT NULL AND element_at(price_arr, 4) > array_sum(slice(price_arr, 1, 3)) / 3 * 1.01 AND element_at(vol_arr, 4) > array_sum(slice(vol_arr, 1, 3)) / 3 * 5 AND element_at(ts_arr, 4) - TS <= 60000000000 THEN 3
      ELSE NULL
    END AS max_k
  FROM window_arrays
)
SELECT
  SYMBOL,
  TS AS BASE_START_TS,
  CASE max_k
    WHEN 3 THEN element_at(ts_arr, 3)
    WHEN 4 THEN element_at(ts_arr, 4)
    WHEN 5 THEN element_at(ts_arr, 5)
    WHEN 6 THEN element_at(ts_arr, 6)
    WHEN 7 THEN element_at(ts_arr, 7)
    WHEN 8 THEN element_at(ts_arr, 8)
    WHEN 9 THEN element_at(ts_arr, 9)
    WHEN 10 THEN element_at(ts_arr, 10)
  END AS BASE_END_TS,
  CASE max_k
    WHEN 3 THEN array_sum(slice(price_arr, 1, 3)) / 3
    WHEN 4 THEN array_sum(slice(price_arr, 1, 4)) / 4
    WHEN 5 THEN array_sum(slice(price_arr, 1, 5)) / 5
    WHEN 6 THEN array_sum(slice(price_arr, 1, 6)) / 6
    WHEN 7 THEN array_sum(slice(price_arr, 1, 7)) / 7
    WHEN 8 THEN array_sum(slice(price_arr, 1, 8)) / 8
    WHEN 9 THEN array_sum(slice(price_arr, 1, 9)) / 9
    WHEN 10 THEN array_sum(slice(price_arr, 1, 10)) / 10
  END AS BASE_AVG_PRICE,
  CASE max_k
    WHEN 3 THEN array_sum(slice(vol_arr, 1, 3)) / 3
    WHEN 4 THEN array_sum(slice(vol_arr, 1, 4)) / 4
    WHEN 5 THEN array_sum(slice(vol_arr, 1, 5)) / 5
    WHEN 6 THEN array_sum(slice(vol_arr, 1, 6)) / 6
    WHEN 7 THEN array_sum(slice(vol_arr, 1, 7)) / 7
    WHEN 8 THEN array_sum(slice(vol_arr, 1, 8)) / 8
    WHEN 9 THEN array_sum(slice(vol_arr, 1, 9)) / 9
    WHEN 10 THEN array_sum(slice(vol_arr, 1, 10)) / 10
  END AS BASE_AVG_SIZE,
  CASE max_k
    WHEN 3 THEN element_at(ts_arr, 4)
    WHEN 4 THEN element_at(ts_arr, 5)
    WHEN 5 THEN element_at(ts_arr, 6)
    WHEN 6 THEN element_at(ts_arr, 7)
    WHEN 7 THEN element_at(ts_arr, 8)
    WHEN 8 THEN element_at(ts_arr, 9)
    WHEN 9 THEN element_at(ts_arr, 10)
    WHEN 10 THEN element_at(ts_arr, 11)
  END AS SPIKE_TS,
  CASE max_k
    WHEN 3 THEN element_at(price_arr, 4)
    WHEN 4 THEN element_at(price_arr, 5)
    WHEN 5 THEN element_at(price_arr, 6)
    WHEN 6 THEN element_at(price_arr, 7)
    WHEN 7 THEN element_at(price_arr, 8)
    WHEN 8 THEN element_at(price_arr, 9)
    WHEN 9 THEN element_at(price_arr, 10)
    WHEN 10 THEN element_at(price_arr, 11)
  END AS SPIKE_PRICE,
  CASE max_k
    WHEN 3 THEN element_at(vol_arr, 4)
    WHEN 4 THEN element_at(vol_arr, 5)
    WHEN 5 THEN element_at(vol_arr, 6)
    WHEN 6 THEN element_at(vol_arr, 7)
    WHEN 7 THEN element_at(vol_arr, 8)
    WHEN 8 THEN element_at(vol_arr, 9)
    WHEN 9 THEN element_at(vol_arr, 10)
    WHEN 10 THEN element_at(vol_arr, 11)
  END AS SPIKE_SIZE
FROM matches
WHERE max_k IS NOT NULL;

WITH prep AS (
  SELECT
    SYMBOL,
    TS,
    TRADE_ID,
    PRICE,
    VOLUME,
    EXCHANGE,
    CONDITIONS,
    LEAD(PRICE, 1) OVER w AS B_PRICE,
    LEAD(VOLUME, 2) OVER w AS C_VOLUME,
    LEAD(EXCHANGE, 3) OVER w AS D_EXCHANGE,
    LEAD(TS, 4) OVER w AS E_TS,
    LEAD(CONDITIONS, 4) OVER w AS E_CONDITIONS,
    LEAD(PRICE, 4) OVER w AS E_PRICE,
    LEAD(PRICE, 5) OVER w AS F_PRICE,
    LEAD(VOLUME, 6) OVER w AS G_VOLUME,
    LEAD(TS, 6) OVER w AS G_TS
  FROM NASDAQ
  WINDOW w AS (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID)
)
SELECT
  SYMBOL,
  START_TS,
  END_TS,
  TOTAL_SIZE
FROM prep
MATCH_RECOGNIZE (
  PARTITION BY SYMBOL
  ORDER BY TS, TRADE_ID
  MEASURES
    FIRST(TS) AS START_TS,
    LAST(TS) AS END_TS,
    SUM(VOLUME) AS TOTAL_SIZE
  ONE ROW PER MATCH
  AFTER MATCH SKIP PAST LAST ROW
  PATTERN (C S{6})
  DEFINE
    C AS C.VOLUME < 50
         AND B_PRICE > C.PRICE
         AND C_VOLUME >= 100
         AND D_EXCHANGE = 'Q'
         AND E_TS IS NOT NULL
         AND E_CONDITIONS IS NULL
         AND F_PRICE < E_PRICE
         AND G_VOLUME >= 500
         AND G_TS - C.TS <= 5000000000,
    S AS TRUE
);

WITH matched_pairs AS (

    A.PRICE AS A_PRICE,

    B_TS,
    B_PRICE,
    B_VOLUME,
    B_EXCHANGE,
    B_CONDITIONS,
    B_TRADE_ID,
    B_TAPE,

    *,

    LAG(is_H, 1, false) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS prev_is_H
  FROM base

  SELECT
    *,
    CASE WHEN is_L THEN SUM(CASE WHEN is_L AND NOT prev_is_L THEN 1 ELSE 0 END) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) END AS L_run_id,
    CASE WHEN is_H THEN SUM(CASE WHEN is_H AND NOT prev_is_H THEN 1 ELSE 0 END) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) END AS H_run_id
  FROM runs
),
enriched AS (
  SELECT
    *,

  FROM run_ids

    SYMBOL,

  FROM enriched
  WHERE is_H

  L.SYMBOL,

ORDER BY L.SYMBOL, L.START_TS;

WITH base AS (
  SELECT
    SYMBOL,
    TS,
    TRADE_ID,
    PRICE,
    VOLUME,
    EXCHANGE,
    CONDITIONS,
    TAPE,
    FEED,
    ROW_NUMBER() OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS rn,
    (FEED = 'SIP' AND VOLUME < 2000) AS is_b,
    (TAPE = 'C') AS is_c,
    (VOLUME >= 10000 AND CONDITIONS IS NOT NULL) AS is_d
  FROM NASDAQ
),
base_c AS (
  SELECT
    *,
    (is_c AND NOT is_b) AS is_c_only,
    (NOT (is_b OR is_c)) AS is_n
  FROM base
),
base_flags AS (
  SELECT
    *,
    (is_c_only AND LAG(is_c_only) OVER (PARTITION BY SYMBOL ORDER BY rn)) AS is_cc_end
  FROM base_c
),
base2 AS (
  SELECT
    *,
    MAX(CASE WHEN is_d THEN rn END) OVER (PARTITION BY SYMBOL ORDER BY rn ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) AS prev_d_rn,
    MAX(CASE WHEN is_n THEN rn END) OVER (PARTITION BY SYMBOL ORDER BY rn ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) AS last_N_before,
    MAX(CASE WHEN is_cc_end THEN rn END) OVER (PARTITION BY SYMBOL ORDER BY rn ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) AS last_CC_end_before,
    SUM(CASE WHEN is_b THEN 1 ELSE 0 END) OVER (PARTITION BY SYMBOL ORDER BY rn ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS cum_b,
    MIN(CASE WHEN is_b THEN rn END) OVER (PARTITION BY SYMBOL ORDER BY rn ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING) AS next_B
  FROM base_flags
),
d_rows AS (
  SELECT
    SYMBOL,
    TS,
    rn,
    COALESCE(prev_d_rn, 0) + 1 AS L,
    GREATEST(
      COALESCE(prev_d_rn, 0) + 1,
      COALESCE(last_N_before, 0) + 1,
      COALESCE(last_CC_end_before, 0) + 1
    ) AS lower_bound,
    cum_b
  FROM base2
  WHERE is_d
),
matches AS (
  SELECT
    d.SYMBOL,
    d.rn AS e_rn,
    d.TS AS END_TS,
    d.lower_bound,
    d.cum_b,
    COALESCE(LEAST(d.rn, b.next_B), d.rn) AS s_rn
  FROM d_rows d
  LEFT JOIN base2 b ON b.SYMBOL = d.SYMBOL AND b.rn = d.lower_bound
)
SELECT
  m.SYMBOL,
  CASE WHEN m.s_rn = m.e_rn THEN NULL ELSE s.TS END AS START_TS,
  m.END_TS,
  m.cum_b - COALESCE(s_prev.cum_b, 0) AS B_ROWS
FROM matches m
LEFT JOIN base2 s ON s.SYMBOL = m.SYMBOL AND s.rn = m.s_rn
LEFT JOIN base2 s_prev ON s_prev.SYMBOL = m.SYMBOL AND s_prev.rn = m.s_rn - 1;

WITH base AS (
  SELECT
    TS,
    TRADE_ID,
    PRICE,
    VOLUME,
    (PRICE BETWEEN 100 AND 300) AS a_flag
  FROM NASDAQ
  WHERE EXCHANGE = 'Q'
),
ordered AS (
  SELECT
    *,
    LAG(PRICE) OVER (ORDER BY TS, TRADE_ID) AS prev_price,
    LAG(a_flag) OVER (ORDER BY TS, TRADE_ID) AS prev_a_flag,
    LAG(TS) OVER (ORDER BY TS, TRADE_ID) AS prev_ts,
    ROW_NUMBER() OVER (ORDER BY TS, TRADE_ID) AS rn
  FROM base
),
flag AS (
  SELECT
    *,
    (VOLUME >= 10000 AND PRICE > prev_price AND prev_a_flag = true) AS b_flag,
    (a_flag AND (prev_a_flag IS NULL OR NOT prev_a_flag)) AS run_start
  FROM ordered
),
runs AS (
  SELECT
    *,
    SUM(CASE WHEN run_start THEN 1 ELSE 0 END) OVER (ORDER BY TS, TRADE_ID) AS run_id
  FROM flag
),
run_agg AS (
  SELECT
    run_id,
    MIN(TS) FILTER (WHERE a_flag) AS start_ts,
    MAX(TS) FILTER (WHERE a_flag) AS last_a_ts,
    COUNT(*) FILTER (WHERE a_flag) AS count_a,
    MIN(rn) FILTER (WHERE a_flag) AS first_a_rn,
    MIN(rn) FILTER (WHERE NOT a_flag) AS next_row_rn,
    MAX(rn) FILTER (WHERE a_flag AND b_flag) AS inside_b_rn
  FROM runs
  WHERE run_id > 0
  GROUP BY run_id
)
SELECT
  'Q' AS EXCHANGE,
  run_agg.start_ts AS START_TS,
  CASE
    WHEN next_row.b_flag THEN run_agg.last_a_ts
    WHEN inside_b.rn IS NOT NULL THEN inside_b.prev_ts
  END AS PREFIX_END_TS,
  CASE
    WHEN next_row.b_flag THEN next_row.TS
    WHEN inside_b.rn IS NOT NULL THEN inside_b.TS
  END AS END_TS,
  CASE
    WHEN next_row.b_flag THEN run_agg.count_a
    WHEN inside_b.rn IS NOT NULL THEN inside_b.rn - run_agg.first_a_rn
  END AS PREFIX_ROWS
FROM run_agg
LEFT JOIN runs next_row ON run_agg.run_id = next_row.run_id AND run_agg.next_row_rn = next_row.rn
LEFT JOIN runs inside_b ON run_agg.run_id = inside_b.run_id AND run_agg.inside_b_rn = inside_b.rn
WHERE next_row.b_flag = true OR inside_b.rn IS NOT NULL;

WITH cls AS (
    SELECT
        SYMBOL, TS_ISO, TS, PRICE, VOLUME, EXCHANGE, CONDITIONS, TRADE_ID, TAPE, FEED,
        CASE
            WHEN EXCHANGE = 'D' THEN 'A'
            WHEN EXCHANGE = 'Q' THEN 'B'
            WHEN EXCHANGE = 'P' THEN 'C'
            ELSE 'O'
        END AS cl
    FROM NASDAQ
),
seq AS (
    SELECT
        *,
        ROW_NUMBER() OVER (PARTITION BY SYMBOL ORDER BY TS DESC, TRADE_ID DESC) AS rn,
        LAG(cl) OVER (PARTITION BY SYMBOL ORDER BY TS DESC, TRADE_ID DESC) AS pcl,
        LEAD(cl) OVER (PARTITION BY SYMBOL ORDER BY TS DESC, TRADE_ID DESC) AS ncl
    FROM cls
),
flagged AS (
    SELECT
        *,
        CASE
            WHEN cl IN ('A', 'B') AND NOT COALESCE(pcl IN ('A', 'B'), FALSE) THEN 1
            ELSE 0
        END AS st
    FROM seq
),
grouped AS (
    SELECT
        *,
        SUM(st) OVER (
            PARTITION BY SYMBOL
            ORDER BY TS DESC, TRADE_ID DESC
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
        ) AS rid
    FROM flagged
),
sized AS (
    SELECT
        *,
        COUNT(CASE WHEN cl IN ('A', 'B') THEN 1 END) OVER (PARTITION BY SYMBOL, rid) AS rlen,
        ROW_NUMBER() OVER (PARTITION BY SYMBOL, rid ORDER BY rn) AS pos
    FROM grouped
),
matched_run AS (
    SELECT
        *,
        MAX(CASE WHEN pos = rlen AND ncl = 'C' AND rlen >= 2 THEN 1 ELSE 0 END)
            OVER (PARTITION BY SYMBOL, rid) AS rmatch
    FROM sized
),
with_lag AS (
    SELECT
        *,
        LAG(rmatch) OVER (PARTITION BY SYMBOL ORDER BY TS DESC, TRADE_ID DESC) AS prmatch,
        LAG(rid) OVER (PARTITION BY SYMBOL ORDER BY TS DESC, TRADE_ID DESC) AS prid
    FROM matched_run
),
rows_in_match AS (
    SELECT
        *,
        CASE
            WHEN cl IN ('A', 'B') AND rmatch = 1 AND pos >= GREATEST(1, rlen - 5) THEN rid
            WHEN cl = 'C' AND pcl IN ('A', 'B') AND prmatch = 1 THEN prid
        END AS mrid
    FROM with_lag
),
match_first AS (
    SELECT
        SYMBOL,
        mrid,
        MIN(rn) AS first_rn
    FROM rows_in_match
    WHERE mrid IS NOT NULL
    GROUP BY SYMBOL, mrid
),
numbered AS (
    SELECT
        r.*,
        ROW_NUMBER() OVER (PARTITION BY r.SYMBOL ORDER BY f.first_rn) AS mnum
    FROM rows_in_match r
    JOIN match_first f
        ON r.SYMBOL = f.SYMBOL AND r.mrid = f.mrid
    WHERE r.mrid IS NOT NULL
)

    SYMBOL,
    TS_ISO,
    TS,
    PRICE,
    VOLUME,
    EXCHANGE,
    CONDITIONS,
    TRADE_ID,
    TAPE,
    FEED,
    cl AS VAR_LABEL
FROM numbered
WHERE mnum <= 99;

WITH classified AS (
  SELECT SYMBOL, TS, TRADE_ID,
         CASE WHEN VOLUME >= 4328 THEN 'A'
              WHEN VOLUME >= 827 THEN 'B'
              WHEN VOLUME >= 106 THEN 'C'
              ELSE 'D' END AS CAT
  FROM NASDAQ
),
with_lags AS (
  SELECT SYMBOL, TS, TRADE_ID, CAT,
         LAG(CAT) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS prev_cat,
         LEAD(CAT) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS next_cat,
         LAG(TS) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS prev_ts,
         LEAD(TS) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS next_ts
  FROM classified
),
blocks AS (
  SELECT SYMBOL, TS, TRADE_ID, CAT, prev_cat, next_cat, prev_ts, next_ts,
         SUM(CASE WHEN CAT IN ('B','C') AND (prev_cat IS NULL OR prev_cat NOT IN ('B','C')) THEN 1 ELSE 0 END)
           OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS block_id
  FROM with_lags
)
SELECT SYMBOL,
       MIN(CASE WHEN prev_cat IS NULL OR prev_cat NOT IN ('B','C') THEN prev_ts END) AS START_TS,
       MAX(CASE WHEN next_cat IS NULL OR next_cat NOT IN ('B','C') THEN next_ts END) AS END_TS,
       COUNT(CASE WHEN CAT = 'B' THEN 1 END) AS B_ROWS,
       COUNT(CASE WHEN CAT = 'C' THEN 1 END) AS C_ROWS
FROM blocks
WHERE CAT IN ('B','C')
GROUP BY SYMBOL, block_id
HAVING MIN(CASE WHEN prev_cat IS NULL OR prev_cat NOT IN ('B','C') THEN prev_cat END) = 'A'
   AND MAX(CASE WHEN next_cat IS NULL OR next_cat NOT IN ('B','C') THEN next_cat END) = 'D';

WITH base AS (
    SELECT *,
        ROW_NUMBER() OVER (PARTITION BY EXCHANGE ORDER BY TS, TRADE_ID) AS rn,
        CASE WHEN CONDITIONS LIKE '%7%' THEN 1 ELSE 0 END AS is_a,
        CASE WHEN CONDITIONS LIKE '%4%' AND CONDITIONS NOT LIKE '%7%' AND SYMBOL IN ('AAPL', 'ADBE', 'AMAT', 'AMD', 'AMZN', 'AVGO', 'CSCO', 'GOOG', 'GOOGL', 'INTC') THEN 1 ELSE 0 END AS is_b,
        CASE WHEN CONDITIONS LIKE '%T%' AND CONDITIONS NOT LIKE '%4%' AND CONDITIONS NOT LIKE '%7%' THEN 1 ELSE 0 END AS is_c
    FROM NASDAQ
),
match_meta AS (
    SELECT
        EXCHANGE,
        MIN(rn) FILTER (WHERE is_a = 1) AS first_a_rn,
        MAX(rn) FILTER (WHERE is_c = 1) AS last_c_rn
    FROM base
    GROUP BY EXCHANGE
),
match_b AS (
    SELECT
        b.EXCHANGE,
        MAX(b.rn) FILTER (WHERE b.is_b = 1 AND b.rn > m.first_a_rn AND b.rn < m.last_c_rn) AS j_rn
    FROM base b
    JOIN match_meta m ON b.EXCHANGE = m.EXCHANGE
    GROUP BY b.EXCHANGE
),
match_c AS (
    SELECT
        b.EXCHANGE,
        MIN(b.rn) FILTER (WHERE b.is_c = 1 AND b.rn > mb.j_rn) AS k_rn
    FROM base b
    JOIN match_b mb ON b.EXCHANGE = mb.EXCHANGE
    GROUP BY b.EXCHANGE
)
SELECT
    b.SYMBOL,
    b.TS_ISO,
    b.TS,
    b.PRICE,
    b.VOLUME,
    b.EXCHANGE,
    b.CONDITIONS,
    b.TRADE_ID,
    b.TAPE,
    b.FEED,
    CASE WHEN b.rn = m.first_a_rn THEN b.TS END AS START_TS,
    CASE WHEN b.rn = mb.j_rn THEN b.TS END AS MIDDLE_TS,
    CASE WHEN b.rn = mc.k_rn THEN b.TS END AS END_TS
FROM base b
JOIN match_meta m ON b.EXCHANGE = m.EXCHANGE
JOIN match_b mb ON b.EXCHANGE = mb.EXCHANGE
JOIN match_c mc ON b.EXCHANGE = mc.EXCHANGE
WHERE (b.rn = m.first_a_rn OR b.rn = mb.j_rn OR b.rn = mc.k_rn)
  AND m.first_a_rn IS NOT NULL
  AND mb.j_rn IS NOT NULL
  AND mc.k_rn IS NOT NULL
ORDER BY b.EXCHANGE, b.rn;
