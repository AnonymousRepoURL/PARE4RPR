WITH BASE AS (
  SELECT /*+ MATERIALIZE */
         SYMBOL, TS_ISO, TS, PRICE, VOLUME, EXCHANGE, CONDITIONS, TRADE_ID, TAPE, FEED,
         ROW_NUMBER() OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS RN
  FROM NASDAQ
),
MATCHED AS (
  SELECT S.SYMBOL AS SYMBOL,
         S.RN AS SRN,
         S.PRICE AS START_PRICE,
         MIN(E.RN) AS ERN,
         MIN(E.PRICE) KEEP (DENSE_RANK FIRST ORDER BY E.RN) AS END_PRICE
  FROM BASE S
  JOIN BASE E
    ON E.SYMBOL = S.SYMBOL
   AND E.RN > S.RN
   AND E.VOLUME >= 500
   AND E.PRICE >= S.PRICE * 1.02
   AND E.TS - S.TS <= 500000000
  WHERE S.PRICE BETWEEN 300 AND 500
    AND S.VOLUME >= 500
  GROUP BY S.SYMBOL, S.RN, S.PRICE
),
NUMBERED AS (
  SELECT M.SYMBOL AS SYMBOL,
         M.SRN AS SRN,
         M.ERN AS ERN,
         M.START_PRICE AS START_PRICE,
         M.END_PRICE AS END_PRICE,
         ROW_NUMBER() OVER (PARTITION BY M.SYMBOL ORDER BY M.SRN) AS MATCH_NO
  FROM MATCHED M
)
SELECT B.SYMBOL, B.TS_ISO, B.TS, B.PRICE, B.VOLUME, B.EXCHANGE, B.CONDITIONS, B.TRADE_ID, B.TAPE, B.FEED,
       N.MATCH_NO,
       CASE WHEN B.RN = N.SRN THEN 'START_EVENT'
            WHEN B.RN = N.ERN THEN 'END_EVENT'
            ELSE 'GAP' END AS TRADE_CLASS,
       N.START_PRICE AS START_PRICE,
       N.END_PRICE AS END_PRICE
FROM NUMBERED N
JOIN BASE B
  ON B.SYMBOL = N.SYMBOL
 AND B.RN BETWEEN N.SRN AND N.ERN;

WITH top10 AS (
  SELECT
    SYMBOL,
    TS,
    VOLUME,
    ROW_NUMBER() OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS rn
  FROM NASDAQ
),
enriched AS (
  SELECT
    SYMBOL,
    TS,
    VOLUME,
    rn,
    FIRST_VALUE(TS) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS first_ts,
    SUM(VOLUME) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS cum_vol
  FROM top10
  WHERE rn <= 10
),
flagged AS (
  SELECT
    SYMBOL,
    rn,
    TS,
    first_ts,
    cum_vol,
    CASE WHEN VOLUME < 100 AND TS - first_ts <= 1000000000 THEN 1 ELSE 0 END AS ok
  FROM enriched
),
agg AS (
  SELECT
    SYMBOL,
    MIN(CASE WHEN ok = 0 THEN rn END) AS first_fail_rn,
    MAX(rn) AS max_rn
  FROM flagged
  GROUP BY SYMBOL
),
matches AS (
  SELECT
    SYMBOL,
    CASE
      WHEN first_fail_rn IS NULL THEN LEAST(10, max_rn)
      ELSE LEAST(10, first_fail_rn - 1)
    END AS m
  FROM agg
)
SELECT
  m.SYMBOL,
  f.first_ts AS FIRST_TS,
  f.TS AS FIFTH_TS,
  f.cum_vol AS OPENING_VOLUME
FROM matches m
JOIN flagged f
  ON f.SYMBOL = m.SYMBOL
 AND f.rn = m.m
WHERE m.m >= 5;

WITH ordered AS (
  SELECT SYMBOL, TS, TRADE_ID, VOLUME, PRICE, EXCHANGE, CONDITIONS,
         ROW_NUMBER() OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS rn,
         LEAD(TS, 6) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS ts6,
         LEAD(PRICE, 1) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS price1,
         LEAD(VOLUME, 2) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS vol2,
         LEAD(EXCHANGE, 3) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS exch3,
         LEAD(CONDITIONS, 4) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS cond4,
         LEAD(PRICE, 4) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS price4,
         LEAD(PRICE, 5) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS price5,
         LEAD(VOLUME, 6) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS vol6,
         SUM(VOLUME) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID ROWS BETWEEN CURRENT ROW AND 6 FOLLOWING) AS total_size
  FROM NASDAQ
),
candidates AS (
  SELECT SYMBOL, rn AS start_rn, TS AS start_ts, ts6 AS end_ts, total_size
  FROM ordered
  WHERE VOLUME < 50
    AND price1 > PRICE
    AND vol2 >= 100
    AND exch3 = 'Q'
    AND cond4 IS NULL
    AND price5 < price4
    AND vol6 >= 500
    AND ts6 - TS <= 5000000000
)
SELECT SYMBOL, START_TS, END_TS, TOTAL_SIZE
FROM candidates
MATCH_RECOGNIZE(

  AFTER MATCH SKIP PAST LAST ROW
  PATTERN (A X*)
  DEFINE X AS X.start_rn <= A.start_rn + 6

WITH numbered AS (
  SELECT SYMBOL, TS_ISO, TS, PRICE, VOLUME, EXCHANGE, CONDITIONS, TRADE_ID, TAPE, FEED,
         ROW_NUMBER() OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS RN
  FROM NASDAQ
),
a_rows AS (
  SELECT SYMBOL, RN, TS, VOLUME
  FROM numbered
  WHERE VOLUME >= 1000
),
matches AS (
  SELECT a.SYMBOL AS SYMBOL, a.RN AS A_RN, a.TS AS START_TS, MIN(b.RN) AS B_RN
  FROM a_rows a
  JOIN numbered b
    ON b.SYMBOL = a.SYMBOL
   AND b.RN > a.RN
   AND b.TS <= a.TS + 120000000000
   AND b.VOLUME >= a.VOLUME * 5
  GROUP BY a.SYMBOL, a.RN, a.TS
),
match_info AS (
  SELECT m.SYMBOL AS SYMBOL, m.A_RN AS A_RN, m.B_RN AS B_RN, m.START_TS AS START_TS,
         b.TS AS END_TS,
         ROW_NUMBER() OVER (PARTITION BY m.SYMBOL ORDER BY m.A_RN) AS MATCH_NO
  FROM matches m
  JOIN numbered b ON b.SYMBOL = m.SYMBOL AND b.RN = m.B_RN
)
SELECT n.SYMBOL, n.TS_ISO, n.TS, n.PRICE, n.VOLUME, n.EXCHANGE, n.CONDITIONS, n.TRADE_ID, n.TAPE, n.FEED,
       mi.MATCH_NO,
       CASE WHEN n.RN = mi.A_RN THEN 'A'
            WHEN n.RN = mi.B_RN THEN 'B'
            ELSE 'GAP' END AS TRADE_CLASS,
       mi.START_TS,
       mi.END_TS
FROM match_info mi
JOIN numbered n
  ON n.SYMBOL = mi.SYMBOL
 AND n.RN BETWEEN mi.A_RN AND mi.B_RN
ORDER BY n.SYMBOL, mi.MATCH_NO, n.RN;

WITH base AS (

    SYMBOL,

    CASE WHEN VOLUME BETWEEN 200 AND 400 THEN 1 ELSE 0 END AS is_B,
    CASE WHEN PRICE >= 500 THEN 1 ELSE 0 END AS is_C
  FROM NASDAQ
),
grp AS (
  SELECT
    SYMBOL,
    TS,
    rn,
    is_A,
    is_C,
    SUM(CASE WHEN is_A = 0 AND is_B = 0 THEN 1 ELSE 0 END) OVER (PARTITION BY SYMBOL ORDER BY rn) 
      - CASE WHEN is_A = 0 AND is_B = 0 THEN 1 ELSE 0 END AS prev_nonX
  FROM base
),
matches AS (
  SELECT
    SYMBOL,
    TS AS START_TS,
    rn AS start_rn,
    MAX(CASE WHEN is_C = 1 THEN rn END) OVER (
      PARTITION BY SYMBOL, prev_nonX
      ORDER BY rn
      ROWS BETWEEN 1 FOLLOWING AND 19 FOLLOWING
    ) AS end_rn
  FROM grp
  WHERE is_A = 1
)
SELECT
  m.SYMBOL,
  m.START_TS,
  e.TS AS END_TS
FROM matches m
JOIN base e
  ON e.SYMBOL = m.SYMBOL
 AND e.rn = m.end_rn
WHERE m.end_rn IS NOT NULL;

WITH base AS (
  SELECT SYMBOL, EXCHANGE, TS, TRADE_ID, PRICE, VOLUME,
         CASE WHEN PRICE > 300 AND VOLUME >= 5000 THEN 1 ELSE 0 END AS is_C,
         COUNT(CASE WHEN PRICE > 300 AND VOLUME >= 5000 THEN 1 END) 
           OVER (PARTITION BY SYMBOL, EXCHANGE ORDER BY TS, TRADE_ID ROWS BETWEEN UNBOUNDED PRECEDING AND 1 PRECEDING) AS seg_id
  FROM NASDAQ
),
flags AS (
  SELECT SYMBOL, EXCHANGE, TS, TRADE_ID, PRICE, VOLUME, is_C, seg_id,
         CASE WHEN is_C = 0 AND PRICE > 250 THEN 1 ELSE 0 END AS is_invalid,
         CASE WHEN is_C = 0 AND PRICE < 150 THEN 1 ELSE 0 END AS is_A_only,
         CASE WHEN is_C = 0 AND PRICE >= 200 AND PRICE <= 250 THEN 1 ELSE 0 END AS is_B_only
  FROM base
),
row_nums AS (
  SELECT SYMBOL, EXCHANGE, TS, TRADE_ID, PRICE, VOLUME, is_C, seg_id, is_invalid, is_A_only, is_B_only,
         ROW_NUMBER() OVER (PARTITION BY SYMBOL, EXCHANGE, seg_id ORDER BY TS, TRADE_ID) AS rn
  FROM flags
),
bad AS (
  SELECT SYMBOL, EXCHANGE, TS, TRADE_ID, PRICE, VOLUME, is_C, seg_id, is_invalid, is_A_only, is_B_only, rn,
         MAX(is_A_only) OVER (PARTITION BY SYMBOL, EXCHANGE, seg_id ORDER BY rn ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING) AS is_A_only_after
  FROM row_nums
),
bad_calc AS (
  SELECT SYMBOL, EXCHANGE, TS, TRADE_ID, PRICE, VOLUME, is_C, seg_id, is_invalid, is_A_only, is_B_only, rn,
         CASE WHEN is_invalid = 1 OR (is_B_only = 1 AND NVL(is_A_only_after, 0) = 1) THEN 1 ELSE 0 END AS is_bad
  FROM bad
),
seg_agg AS (
  SELECT SYMBOL, EXCHANGE, seg_id,
         MAX(CASE WHEN is_bad = 1 THEN rn END) AS max_bad_rn,
         MAX(CASE WHEN is_A_only = 1 THEN rn END) AS max_A_rn
  FROM bad_calc
  GROUP BY SYMBOL, EXCHANGE, seg_id
)
SELECT 
  c.SYMBOL,
  c.EXCHANGE,
  CASE WHEN NVL(a.max_A_rn, 0) > NVL(a.max_bad_rn, 0) THEN s.TS ELSE NULL END AS START_TS,
  c.TS AS END_TS,
  c.rn - NVL(a.max_bad_rn, 0) AS RELUCTANT_ROWS
FROM bad_calc c
JOIN seg_agg a ON c.SYMBOL = a.SYMBOL AND c.EXCHANGE = a.EXCHANGE AND c.seg_id = a.seg_id
JOIN row_nums s ON s.SYMBOL = c.SYMBOL AND s.EXCHANGE = c.EXCHANGE AND s.seg_id = c.seg_id AND s.rn = NVL(a.max_bad_rn, 0) + 1
WHERE c.is_C = 1;

WITH
  base AS (
    SELECT
      SYMBOL, TS, TRADE_ID, VOLUME, CONDITIONS, TAPE,
      ROW_NUMBER() OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS RN,
      LEAD(VOLUME) OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS NEXT_VOL
    FROM NASDAQ
  ),
  with_gt1000 AS (
    SELECT
      base.*,
      MIN(CASE WHEN VOLUME > 1000 THEN RN END) OVER (PARTITION BY SYMBOL ORDER BY RN ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING) AS NEXT_GT1000_RN
    FROM base
  ),
  with_blocks AS (
    SELECT
      with_gt1000.*,
      SUM(CASE WHEN VOLUME >= 500 AND (LAG(VOLUME) OVER (PARTITION BY SYMBOL ORDER BY RN) IS NULL OR LAG(VOLUME) OVER (PARTITION BY SYMBOL ORDER BY RN) < 500) THEN 1 ELSE 0 END)
        OVER (PARTITION BY SYMBOL ORDER BY RN) AS B_BLOCK_ID
    FROM with_gt1000
  ),
  block_info AS (
    SELECT
      SYMBOL,
      B_BLOCK_ID,
      MIN(RN) AS BLOCK_START_RN,
      MAX(RN) AS BLOCK_END_RN,
      MAX(CASE WHEN VOLUME >= 10000 AND CONDITIONS IS NOT NULL THEN RN END) AS LAST_C_RN
    FROM with_blocks
    WHERE VOLUME >= 500
    GROUP BY SYMBOL, B_BLOCK_ID
  )
SELECT
  S.TS AS START_TS,
  Q.TS AS END_TS,
  Q.RN - S.RN + 1 AS RELUCTANT_ROWS
FROM with_blocks S
JOIN with_blocks NEXT_ROW
  ON NEXT_ROW.SYMBOL = S.SYMBOL AND NEXT_ROW.RN = S.RN + 1
JOIN with_blocks C1
  ON C1.SYMBOL = S.SYMBOL AND C1.RN = NEXT_ROW.NEXT_GT1000_RN
JOIN block_info BI
  ON BI.SYMBOL = S.SYMBOL AND BI.B_BLOCK_ID = C1.B_BLOCK_ID
JOIN with_blocks Q
  ON Q.SYMBOL = S.SYMBOL AND Q.RN = BI.LAST_C_RN
WHERE S.TAPE = 'Y'
  AND NEXT_ROW.VOLUME <= 1000
  AND NEXT_ROW.NEXT_GT1000_RN IS NOT NULL
  AND BI.LAST_C_RN IS NOT NULL
  AND GREATEST(BI.BLOCK_START_RN, S.RN + 2) < BI.LAST_C_RN
ORDER BY S.RN;

    SUM(CASE WHEN is_a = 0 THEN 1 ELSE 0 END) OVER (PARTITION BY EXCHANGE ORDER BY rn ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS grp
  FROM base1 b
),
a_with_runs AS (
  SELECT
    b.*,
    MIN(rn) OVER (PARTITION BY grp) AS L_rn,
    MAX(rn) OVER (PARTITION BY grp) AS R_rn
  FROM base b
  WHERE is_a = 1
),
a_with_b AS (
  SELECT
    a.*,
    MAX(CASE WHEN is_b = 1 AND rn > L_rn THEN rn END) OVER (PARTITION BY grp) AS B_max_rn
  FROM a_with_runs a
),
run_matches AS (
  SELECT
    a.grp,
    a.L_rn,
    a.R_rn,
    a.B_max_rn,
    b_next.is_b AS B_next_flag
  FROM a_with_b a
  LEFT JOIN base b_next ON b_next.rn = a.R_rn + 1
  WHERE a.rn = a.L_rn
),
match_info AS (
  SELECT
    grp,
    L_rn,
    CASE
      WHEN B_next_flag = 1 THEN R_rn + 1
      WHEN B_max_rn IS NOT NULL THEN B_max_rn
      ELSE NULL
    END AS j_rn
  FROM run_matches
)
SELECT
  l.EXCHANGE,
  l.TS,
  l.TRADE_ID,
  l.TS AS START_TS,
  prev.TS AS PREFIX_END_TS,
  j.TS AS END_TS,
  m.j_rn - m.L_rn AS PREFIX_ROWS
FROM match_info m
JOIN base l ON l.rn = m.L_rn
JOIN base j ON j.rn = m.j_rn
JOIN base prev ON prev.rn = m.j_rn - 1
WHERE m.j_rn IS NOT NULL;

           PARTITION BY SYMBOL
           ORDER BY rn
           ROWS BETWEEN 2 FOLLOWING AND 11 FOLLOWING
         ) AS best_c_rn,
         MAX(CASE WHEN VOLUME >= 5000 THEN TS END) OVER (
           PARTITION BY SYMBOL
           ORDER BY rn
           ROWS BETWEEN 2 FOLLOWING AND 11 FOLLOWING

  FROM ranked

  SELECT A.SYMBOL, A.rn AS a_rn, A.TS AS a_ts,

         ROW_NUMBER() OVER (PARTITION BY A.SYMBOL, A.rn ORDER BY B.rn DESC) AS rn_b
  FROM ranked A
  JOIN b_candidates B ON B.SYMBOL = A.SYMBOL
                     AND B.rn BETWEEN A.rn + 3 AND A.rn + 11

WITH ordered AS (
  SELECT
    n.*,
    LAG(n.TRADE_ID, 1) OVER (PARTITION BY n.EXCHANGE ORDER BY n.TRADE_ID, n.TS) AS p1,
    LAG(n.TRADE_ID, 2) OVER (PARTITION BY n.EXCHANGE ORDER BY n.TRADE_ID, n.TS) AS p2,
    LEAD(n.TRADE_ID, 1) OVER (PARTITION BY n.EXCHANGE ORDER BY n.TRADE_ID, n.TS) AS n1,
    LEAD(n.TRADE_ID, 2) OVER (PARTITION BY n.EXCHANGE ORDER BY n.TRADE_ID, n.TS) AS n2
  FROM NASDAQ n
),
matched AS (
  SELECT
    o.*,
    CASE
      WHEN o.TRADE_ID <= 13 AND o.n1 > 13 AND o.n1 <= 105 AND o.n2 > 105 AND o.n2 <= 1687 THEN 1
      WHEN o.p1 <= 13 AND o.TRADE_ID > 13 AND o.TRADE_ID <= 105 AND o.n1 > 105 AND o.n1 <= 1687 THEN 2
      WHEN o.p2 <= 13 AND o.p1 > 13 AND o.p1 <= 105 AND o.TRADE_ID > 105 AND o.TRADE_ID <= 1687 THEN 3

  FROM ordered o
)

  SYMBOL, TS_ISO, TS, PRICE, VOLUME, EXCHANGE, CONDITIONS, TRADE_ID, TAPE, FEED,

  CASE role WHEN 1 THEN n1 WHEN 2 THEN TRADE_ID ELSE p1 END AS MIDDLE_ID,
  CASE role WHEN 1 THEN n2 WHEN 2 THEN n1 ELSE TRADE_ID END AS END_ID
FROM matched
WHERE role IS NOT NULL
ORDER BY EXCHANGE, TRADE_ID, TS;

  SUBSET U = (G1, G2)
  DEFINE
    A AS A.CONDITIONS LIKE '%7%' AND A.VOLUME >= 1000,
    G1 AS G1.VOLUME BETWEEN 100 AND 800,
    G2 AS G1.VOLUME < 100,
    B AS B.CONDITIONS LIKE '%4%' AND B.CONDITIONS NOT LIKE '%7%' AND COUNT(U.*) <= 10
);

SELECT TS AS START_TS, TS AS RANGE_TS, TS AS END_TS FROM NASDAQ WHERE 1=0;

WITH ordered AS (
  SELECT
    SYMBOL, TS_ISO, TS, PRICE, VOLUME, EXCHANGE, CONDITIONS, TRADE_ID, TAPE, FEED,
    ROW_NUMBER() OVER (PARTITION BY EXCHANGE ORDER BY TS, TRADE_ID) AS rn,
    CASE
      WHEN PRICE <= 21.3243 THEN 'S'
      WHEN PRICE <= 42.0270 THEN 'I'
      ELSE 'F'
    END AS row_type
  FROM NASDAQ
),
computed AS (
  SELECT
    ordered.*,
    FIRST_VALUE(CASE WHEN row_type != 'I' THEN rn END) IGNORE NULLS OVER (
      PARTITION BY EXCHANGE ORDER BY rn
      ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING
    ) AS next_non_i_rn,
    FIRST_VALUE(CASE WHEN row_type != 'I' THEN row_type END) IGNORE NULLS OVER (
      PARTITION BY EXCHANGE ORDER BY rn
      ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING
    ) AS next_non_i_type,
    FIRST_VALUE(CASE WHEN row_type != 'I' THEN TS END) IGNORE NULLS OVER (
      PARTITION BY EXCHANGE ORDER BY rn
      ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING
    ) AS next_non_i_ts
  FROM ordered
),
starts AS (
  SELECT
    computed.*,
    CASE
      WHEN row_type = 'S' AND next_non_i_type = 'F' AND (next_non_i_rn - rn - 1) BETWEEN 3 AND 9
      THEN 1 ELSE 0
    END AS is_valid_start
  FROM computed
)
SELECT
  SYMBOL, TS_ISO, TS, PRICE, VOLUME, EXCHANGE, CONDITIONS, TRADE_ID, TAPE, FEED,
  start_ts AS START_TS,
  end_ts AS END_TS,
  match_rows AS MATCH_ROWS
FROM (
  SELECT
    starts.*,
    LAST_VALUE(CASE WHEN is_valid_start = 1 THEN rn END) IGNORE NULLS OVER (
      PARTITION BY EXCHANGE ORDER BY rn
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS match_start_rn,
    LAST_VALUE(CASE WHEN is_valid_start = 1 THEN next_non_i_rn END) IGNORE NULLS OVER (
      PARTITION BY EXCHANGE ORDER BY rn
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS match_end_rn,
    LAST_VALUE(CASE WHEN is_valid_start = 1 THEN TS END) IGNORE NULLS OVER (
      PARTITION BY EXCHANGE ORDER BY rn
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS start_ts,
    LAST_VALUE(CASE WHEN is_valid_start = 1 THEN next_non_i_ts END) IGNORE NULLS OVER (
      PARTITION BY EXCHANGE ORDER BY rn
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS end_ts,
    LAST_VALUE(CASE WHEN is_valid_start = 1 THEN (next_non_i_rn - rn + 1) END) IGNORE NULLS OVER (
      PARTITION BY EXCHANGE ORDER BY rn
      ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
    ) AS match_rows
  FROM starts
)
WHERE rn BETWEEN match_start_rn AND match_end_rn;

WITH base AS (
  SELECT SYMBOL, TS, TRADE_ID, VOLUME,
         CASE WHEN VOLUME >= 4328 THEN 'A'
              WHEN VOLUME >= 827 AND VOLUME < 4328 THEN 'B'
              WHEN VOLUME >= 106 AND VOLUME < 827 THEN 'C'
              WHEN VOLUME < 106 THEN 'D'
         END AS cat
  FROM NASDAQ
),
numbered AS (
  SELECT SYMBOL, TS, TRADE_ID, VOLUME, cat,
         ROW_NUMBER() OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS rn
  FROM base
),
cumulated AS (
  SELECT SYMBOL, TS, TRADE_ID, VOLUME, cat, rn,
         SUM(CASE WHEN cat = 'B' THEN 1 ELSE 0 END) OVER (PARTITION BY SYMBOL ORDER BY rn ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS cum_B,
         SUM(CASE WHEN cat = 'C' THEN 1 ELSE 0 END) OVER (PARTITION BY SYMBOL ORDER BY rn ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS cum_C
  FROM numbered
),
windowed AS (
  SELECT SYMBOL, TS, TRADE_ID, VOLUME, cat, rn, cum_B, cum_C,
         MIN(CASE WHEN cat = 'A' THEN rn END) OVER (PARTITION BY SYMBOL ORDER BY rn ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING) AS next_A_rn,
         MIN(CASE WHEN cat = 'D' THEN rn END) OVER (PARTITION BY SYMBOL ORDER BY rn ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING) AS next_D_rn,
         MIN(CASE WHEN cat = 'D' THEN TS END) OVER (PARTITION BY SYMBOL ORDER BY rn ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING) AS next_D_TS,
         MIN(CASE WHEN cat = 'D' THEN cum_B END) OVER (PARTITION BY SYMBOL ORDER BY rn ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING) AS next_D_cum_B,
         MIN(CASE WHEN cat = 'D' THEN cum_C END) OVER (PARTITION BY SYMBOL ORDER BY rn ROWS BETWEEN 1 FOLLOWING AND UNBOUNDED FOLLOWING) AS next_D_cum_C
  FROM cumulated
),
valid_matches AS (
  SELECT SYMBOL,
         TS AS START_TS,
         next_D_TS AS END_TS,
         next_D_cum_B - cum_B AS B_ROWS,
         next_D_cum_C - cum_C AS C_ROWS
  FROM windowed
  WHERE cat = 'A'
    AND next_D_rn IS NOT NULL
    AND (next_A_rn IS NULL OR next_D_rn < next_A_rn)
    AND next_D_rn > rn + 1
)
SELECT * FROM valid_matches;

SELECT * FROM (SELECT * FROM NASDAQ WHERE FEED = 'sip') MATCH_RECOGNIZE(
  PARTITION BY TAPE
  ORDER BY TS, TRADE_ID
  MEASURES A.TS AS A_TS, B.TS AS B_TS, C.TS AS C_TS, D.TS AS END_TS
  ALL ROWS PER MATCH
  AFTER MATCH SKIP PAST LAST ROW
  PATTERN(PERMUTE(A, B, C) D)
  DEFINE
    A AS A.CONDITIONS LIKE '%F%' AND A.EXCHANGE = 'A',
    B AS B.CONDITIONS LIKE '%F%' AND B.EXCHANGE = 'N',
    C AS C.CONDITIONS LIKE '%F%' AND C.EXCHANGE IN ('P', 'Z'),
    D AS D.CONDITIONS NOT LIKE '%F%' AND D.EXCHANGE = 'D' AND MATCH_NUMBER() < 2000
);

SELECT * FROM NASDAQ MATCH_RECOGNIZE(
  PARTITION BY SYMBOL
  ORDER BY TS, TRADE_ID
  MEASURES A.TS AS START_TS, B.TS AS END_TS, COUNT(*) - 2 AS GAP_ROWS
  PATTERN(A Z{2,20} B)
  DEFINE
    A AS A.TAPE = 'A' AND A.EXCHANGE = 'H',
    B AS B.TAPE = 'C' OR (B.TAPE = 'A' AND B.EXCHANGE IN ('K', 'Z'))
);

SELECT *
FROM (
  SELECT *
  FROM NASDAQ
  WHERE CONDITIONS LIKE '%7%'
     OR (CONDITIONS LIKE '%4%' AND CONDITIONS NOT LIKE '%7%'
         AND SYMBOL IN ('AAPL', 'ADBE', 'AMAT', 'AMD', 'AMZN', 'AVGO', 'CSCO', 'GOOG', 'GOOGL', 'INTC'))
     OR (CONDITIONS LIKE '%T%' AND CONDITIONS NOT LIKE '%4%' AND CONDITIONS NOT LIKE '%7%')
) MATCH_RECOGNIZE(
  PARTITION BY EXCHANGE
  ORDER BY TS, TRADE_ID
  MEASURES A.TS AS START_TS, B.TS AS MIDDLE_TS, C.TS AS END_TS
  ALL ROWS PER MATCH
  AFTER MATCH SKIP PAST LAST ROW
  PATTERN(A {- G1* -} B {- G2*? -} C)
  DEFINE
    A AS A.CONDITIONS LIKE '%7%',
    B AS B.CONDITIONS LIKE '%4%' AND B.CONDITIONS NOT LIKE '%7%' AND B.SYMBOL IN ('AAPL', 'ADBE', 'AMAT', 'AMD', 'AMZN', 'AVGO', 'CSCO', 'GOOG', 'GOOGL', 'INTC'),
    C AS C.CONDITIONS LIKE '%T%' AND C.CONDITIONS NOT LIKE '%4%' AND C.CONDITIONS NOT LIKE '%7%'
);

SELECT * FROM NASDAQ MATCH_RECOGNIZE(
  PARTITION BY SYMBOL
  ORDER BY TRADE_ID, TS
  MEASURES A.TRADE_ID AS START_ID, B.TRADE_ID AS RANGE_ID, LAST(C.TRADE_ID) AS END_ID
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO FIRST C
  PATTERN(A B C{1,3})
  DEFINE
    A AS A.TRADE_ID <= 13 AND A.TS_ISO LIKE '%-07-31%',
    B AS B.TRADE_ID > 13 AND B.TRADE_ID <= 105,
    C AS C.TRADE_ID > 105 AND C.TRADE_ID <= 1687 AND C.TS_ISO LIKE '%2024-07-31%'
);

WITH src AS (
  SELECT SYMBOL, TS, TS_ISO, EXCHANGE,
         CASE WHEN EXCHANGE IN ('C','X','M') THEN 1 ELSE 0 END AS isB,
         CASE WHEN EXCHANGE IN ('K','Z') THEN 1 ELSE 0 END AS isC,
         CASE WHEN EXCHANGE = 'D' THEN 1 ELSE 0 END AS isD,
         CASE WHEN EXCHANGE = 'D' AND TS_ISO LIKE '%-08-01%' THEN 1 ELSE 0 END AS isA,
         CASE WHEN EXCHANGE IN ('C','X','M','K','Z') THEN 1 ELSE 0 END AS bc
  FROM NASDAQ
  WHERE FEED = 'sip'
),
r AS (
  SELECT src.*,
         ROW_NUMBER() OVER (PARTITION BY SYMBOL ORDER BY TS, TRADE_ID) AS rn
  FROM src
),
lagged AS (
  SELECT r.*,
         LAG(bc) OVER (PARTITION BY SYMBOL ORDER BY rn) AS prev_bc
  FROM r
),
marked AS (
  SELECT lagged.*,
         MAX(CASE WHEN bc = 1 AND NVL(prev_bc, 0) = 0 THEN rn END)
             OVER (PARTITION BY SYMBOL ORDER BY rn ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS isl
  FROM lagged
),
islands AS (
  SELECT SYMBOL,
         isl AS island_id,
         MAX(rn) AS end_rn,
         SUM(isB) AS cnt_b,
         SUM(isC) AS cnt_c,
         MAX(CASE WHEN rn = isl THEN isB END) AS first_isb
  FROM marked
  WHERE bc = 1
  GROUP BY SYMBOL, isl
)
SELECT i.SYMBOL AS SYMBOL,
       pa.TS AS START_TS,
       fd.TS AS END_TS,
       i.cnt_b AS B_ROWS,
       i.cnt_c AS C_ROWS
FROM islands i
JOIN r pa ON pa.SYMBOL = i.SYMBOL AND pa.rn = i.island_id - 1
JOIN r fd ON fd.SYMBOL = i.SYMBOL AND fd.rn = i.end_rn + 1
WHERE pa.isA = 1 AND i.first_isb = 1 AND fd.isD = 1;

SELECT SYMBOL, TS_ISO, TS, PRICE, VOLUME, EXCHANGE, CONDITIONS, TRADE_ID, TAPE, FEED, MN, END_TS
FROM (
  SELECT SYMBOL, TS_ISO, TS, PRICE, VOLUME, EXCHANGE, CONDITIONS, TRADE_ID, TAPE, FEED, PRICE * VOLUME AS PV
  FROM NASDAQ
) MATCH_RECOGNIZE(
  PARTITION BY TAPE
  ORDER BY TS, TRADE_ID
  MEASURES MATCH_NUMBER() AS MN, Z.TS AS END_TS
  ALL ROWS PER MATCH
  PATTERN(X{2,5} Y{2,5} Z)
  DEFINE
    X AS X.PV >= 117042,
    Y AS Y.PV >= 25445 AND (Y.PV < 117042 OR Y.PV >= 713836),
    Z AS Z.PV < 25445
);

SELECT * FROM (SELECT * FROM NASDAQ WHERE TAPE IN ('A', 'C')) MATCH_RECOGNIZE(
  PARTITION BY SYMBOL
  ORDER BY TS, TRADE_ID
  MEASURES A.TS AS START_TS, B.TS AS MIDDLE_TS, C.TS AS END_TS, COUNT(Z.*) AS GAP_ROWS
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO LAST B
  PATTERN(A Z{2,20} B Z{2,10} C)
  DEFINE
    A AS A.VOLUME * A.PRICE >= 5000 AND A.VOLUME > 50,
    B AS B.VOLUME * B.PRICE >= 4 * (A.VOLUME * A.PRICE),
    C AS C.VOLUME * C.PRICE >= 4 * (B.VOLUME * B.PRICE)
);
