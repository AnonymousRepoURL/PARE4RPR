WITH BUCKETED_ROWS AS(
    SELECT ID, CASE_NUMBER, TS, BLOCK, IUCR, PRIMARY_TYPE, DESCRIPTION, LOCATION_DESCRIPTION, ARREST, DOMESTIC, BEAT, DISTRICT, WARD, COMMUNITY_AREA, FBI_CODE, X_COORDINATE, Y_COORDINATE, YEAR, UPDATED_ON, LATITUDE, LONGITUDE, LOCATION, FLOOR(TS / 1800000) AS BK_NO
    FROM CRIMES
), RANGES AS(
    SELECT A_BUCKET_NO.BK_NO AS FLAG_NO
    FROM BUCKETED_ROWS A_BUCKET_NO INNER JOIN BUCKETED_ROWS B_BUCKET_NO
      ON A_BUCKET_NO.BK_NO = B_BUCKET_NO.BK_NO
    WHERE A_BUCKET_NO.PRIMARY_TYPE = 'ROBBERY' AND B_BUCKET_NO.PRIMARY_TYPE = 'BATTERY' AND B_BUCKET_NO.DISTRICT = A_BUCKET_NO.DISTRICT AND (B_BUCKET_NO.TS - A_BUCKET_NO.TS) <= 1800000
    UNION
    SELECT A_BUCKET_NO.BK_NO AS FLAG_NO
    FROM BUCKETED_ROWS A_BUCKET_NO INNER JOIN BUCKETED_ROWS B_BUCKET_NO
      ON A_BUCKET_NO.BK_NO + 1 = B_BUCKET_NO.BK_NO
    WHERE A_BUCKET_NO.PRIMARY_TYPE = 'ROBBERY' AND B_BUCKET_NO.PRIMARY_TYPE = 'BATTERY' AND B_BUCKET_NO.DISTRICT = A_BUCKET_NO.DISTRICT AND (B_BUCKET_NO.TS - A_BUCKET_NO.TS) <= 1800000
), BK_NO_LIST AS(
    SELECT FLAG_NO AS BK_NO
    FROM RANGES
    UNION
    SELECT FLAG_NO + 1 AS BK_NO
    FROM RANGES
), FILTERED_ROWS AS(
    SELECT ID, CASE_NUMBER, TS, BLOCK, IUCR, PRIMARY_TYPE, DESCRIPTION, LOCATION_DESCRIPTION, ARREST, DOMESTIC, BEAT, DISTRICT, WARD, COMMUNITY_AREA, FBI_CODE, X_COORDINATE, Y_COORDINATE, YEAR, UPDATED_ON, LATITUDE, LONGITUDE, LOCATION
    FROM BUCKETED_ROWS
    WHERE BK_NO IN (SELECT BK_NO FROM BK_NO_LIST)
)
