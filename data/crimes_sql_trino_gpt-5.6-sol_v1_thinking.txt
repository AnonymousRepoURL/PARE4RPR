    BEAT,
    CASE WHEN PRIMARY_TYPE = 'THEFT' THEN ID END AS A_ID,
    CASE WHEN PRIMARY_TYPE = 'ROBBERY' THEN ID END AS B_ID,
    CASE WHEN NEXT_PRIMARY_TYPE = 'BATTERY' THEN NEXT_ID END AS C_ID,
    CASE WHEN NEXT_PRIMARY_TYPE = 'ASSAULT' THEN NEXT_ID END AS D_ID
FROM (
    SELECT
        BEAT,
        ID,
        TS,
        PRIMARY_TYPE,
        LEAD(ID) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS NEXT_ID,
        LEAD(TS) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS NEXT_TS,
        LEAD(PRIMARY_TYPE) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS NEXT_PRIMARY_TYPE
    FROM CRIMES
) AS CRIME_PAIRS
WHERE PRIMARY_TYPE IN ('THEFT', 'ROBBERY')
  AND NEXT_PRIMARY_TYPE IN ('BATTERY', 'ASSAULT')
  AND NEXT_TS - TS <= 1800000;

SELECT
    BEAT,
    CASE
        WHEN PRIMARY_TYPE = 'THEFT' THEN ID
        WHEN PRIMARY_TYPE_1 = 'THEFT' THEN ID_1
        ELSE ID_2
    END AS THEFT_ID,
    CASE
        WHEN PRIMARY_TYPE = 'BATTERY' THEN ID
        WHEN PRIMARY_TYPE_1 = 'BATTERY' THEN ID_1
        ELSE ID_2
    END AS BATTERY_ID,
    CASE
        WHEN PRIMARY_TYPE = 'CRIMINAL DAMAGE' THEN ID
        WHEN PRIMARY_TYPE_1 = 'CRIMINAL DAMAGE' THEN ID_1
        ELSE ID_2
    END AS DAMAGE_ID
FROM (
    SELECT
        BEAT,
        ID,
        PRIMARY_TYPE,
        LEAD(ID, 1) OVER w AS ID_1,
        LEAD(PRIMARY_TYPE, 1) OVER w AS PRIMARY_TYPE_1,
        LEAD(ID, 2) OVER w AS ID_2,
        LEAD(PRIMARY_TYPE, 2) OVER w AS PRIMARY_TYPE_2
    FROM CRIMES
    WINDOW w AS (PARTITION BY BEAT ORDER BY TS, ID)
) AS ordered_crimes
WHERE PRIMARY_TYPE IN ('THEFT', 'BATTERY', 'CRIMINAL DAMAGE')
  AND PRIMARY_TYPE_1 IN ('THEFT', 'BATTERY', 'CRIMINAL DAMAGE')
  AND PRIMARY_TYPE_2 IN ('THEFT', 'BATTERY', 'CRIMINAL DAMAGE')
  AND PRIMARY_TYPE <> PRIMARY_TYPE_1
  AND PRIMARY_TYPE <> PRIMARY_TYPE_2
  AND PRIMARY_TYPE_1 <> PRIMARY_TYPE_2;

WITH classified AS (
    SELECT
        BEAT,
        TS,
        ID,
        CASE
            WHEN PRIMARY_TYPE = 'ASSAULT'
                AND COMMUNITY_AREA = 28
                AND ARREST = 'true' THEN 1
            WHEN YEAR = 2024
                AND PRIMARY_TYPE = 'BATTERY'
                AND DISTRICT = '011' THEN 2
            WHEN PRIMARY_TYPE = 'OTHER OFFENSE'
                AND COMMUNITY_AREA = 33
                AND ARREST = 'false' THEN 3
            ELSE 0
        END AS role
    FROM CRIMES
),
sequences AS (
    SELECT
        BEAT,
        ID,
        role,
        LEAD(ID, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS id_1,
        LEAD(ID, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS id_2,
        LEAD(role, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS role_1,
        LEAD(role, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS role_2
    FROM classified
)
SELECT
    BEAT,
    CASE
        WHEN role = 1 THEN ID
        WHEN role_1 = 1 THEN id_1
        WHEN role_2 = 1 THEN id_2
    END AS A_ID,
    CASE
        WHEN role = 2 THEN ID
        WHEN role_1 = 2 THEN id_1
        WHEN role_2 = 2 THEN id_2
    END AS B_ID,
    CASE
        WHEN role = 3 THEN ID
        WHEN role_1 = 3 THEN id_1
        WHEN role_2 = 3 THEN id_2
    END AS C_ID
FROM sequences
WHERE role * role_1 * role_2 = 6;

WITH classified AS (
    SELECT
        BEAT,
        TS,
        ID,
        CASE
            WHEN YEAR = 2004
                AND PRIMARY_TYPE = 'ASSAULT' THEN 'A'
            WHEN PRIMARY_TYPE = 'BURGLARY'
                AND DISTRICT = '006' THEN 'B'
            WHEN PRIMARY_TYPE = 'CRIMINAL DAMAGE'
                AND WARD = 17
                AND ARREST = 'false' THEN 'C'
        END AS pattern_label
    FROM CRIMES
),
sequenced AS (
    SELECT
        BEAT,
        ID,
        pattern_label,
        LEAD(ID, 1) OVER (
            PARTITION BY BEAT
            ORDER BY TS, ID
        ) AS next_id,
        LEAD(ID, 2) OVER (
            PARTITION BY BEAT
            ORDER BY TS, ID
        ) AS third_id,
        LEAD(pattern_label, 1) OVER (
            PARTITION BY BEAT
            ORDER BY TS, ID
        ) AS next_pattern_label,
        LEAD(pattern_label, 2) OVER (
            PARTITION BY BEAT
            ORDER BY TS, ID
        ) AS third_pattern_label
    FROM classified
)
SELECT
    BEAT,
    CASE
        WHEN pattern_label = 'A' THEN ID
        WHEN next_pattern_label = 'A' THEN next_id
        ELSE third_id
    END AS A_ID,
    CASE
        WHEN pattern_label = 'B' THEN ID
        WHEN next_pattern_label = 'B' THEN next_id
        ELSE third_id
    END AS B_ID,
    CASE
        WHEN pattern_label = 'C' THEN ID
        WHEN next_pattern_label = 'C' THEN next_id
        ELSE third_id
    END AS C_ID
FROM sequenced
WHERE pattern_label IS NOT NULL
  AND next_pattern_label IS NOT NULL
  AND third_pattern_label IS NOT NULL
  AND pattern_label <> next_pattern_label
  AND pattern_label <> third_pattern_label
  AND next_pattern_label <> third_pattern_label;

WITH classified AS (
    SELECT
        BEAT,
        TS,
        ID,
        CASE
            WHEN PRIMARY_TYPE = 'BURGLARY' AND ARREST = 'false' THEN 1
            WHEN PRIMARY_TYPE = 'CRIMINAL TRESPASS' AND ARREST = 'false' THEN 2
            WHEN PRIMARY_TYPE = 'CRIMINAL DAMAGE' AND LOCATION_DESCRIPTION = 'STREET' THEN 4
        END AS kind
    FROM CRIMES
),
windowed AS (
    SELECT
        BEAT,
        ID,
        kind,
        LEAD(ID, 1) OVER w AS id_1,
        LEAD(ID, 2) OVER w AS id_2,
        LEAD(kind, 1) OVER w AS kind_1,
        LEAD(kind, 2) OVER w AS kind_2
    FROM classified
    WINDOW w AS (PARTITION BY BEAT ORDER BY TS, ID)
)
SELECT
    BEAT,
    CASE
        WHEN kind = 1 THEN ID
        WHEN kind_1 = 1 THEN id_1
        ELSE id_2
    END AS A_ID,
    CASE
        WHEN kind = 2 THEN ID
        WHEN kind_1 = 2 THEN id_1
        ELSE id_2
    END AS B_ID,
    CASE
        WHEN kind = 4 THEN ID
        WHEN kind_1 = 4 THEN id_1
        ELSE id_2
    END AS C_ID
FROM windowed
WHERE kind + kind_1 + kind_2 = 7;

    END_EVENT AS END_EVENT.PRIMARY_TYPE = 'MOTOR VEHICLE THEFT' AND END_EVENT.ARREST = 'false'
);

SELECT DISTRICT, TS, ID
FROM (
    SELECT
        DISTRICT,
        TS,
        ID,
        rn,
        MIN(CASE WHEN PRIMARY_TYPE = 'HOMICIDE' THEN rn END)
            OVER (PARTITION BY DISTRICT) AS first_homicide_rn
    FROM (
        SELECT
            DISTRICT,
            TS,
            ID,
            PRIMARY_TYPE,
            ROW_NUMBER() OVER (
                PARTITION BY DISTRICT
                ORDER BY TS, ID
            ) AS rn
        FROM CRIMES
    ) ranked
    WHERE rn <= 51
) prefix
WHERE rn <= COALESCE(first_homicide_rn, 50);

WITH relevant_crimes AS (
    SELECT district, ts, id, primary_type
    FROM crimes
    WHERE primary_type IN ('THEFT', 'BATTERY', 'ROBBERY')
)
SELECT *
FROM relevant_crimes MATCH_RECOGNIZE(
    PARTITION BY district
    ORDER BY ts, id
    MEASURES
        A.id AS theft_id,
        B.id AS battery_id,
        C.id AS robbery_id
    ONE ROW PER MATCH
    AFTER MATCH SKIP TO NEXT ROW
    PATTERN(A {- X* -} B {- Y* -} C)
    DEFINE
        A AS A.primary_type = 'THEFT',
        X AS X.ts - A.ts <= 86400000,
        B AS B.primary_type = 'BATTERY'
             AND B.ts - A.ts <= 86400000,
        Y AS Y.ts - A.ts <= 86400000,
        C AS C.primary_type = 'ROBBERY'
             AND C.ts - A.ts <= 86400000
);

SELECT *
FROM (
  SELECT BEAT, TS, ID, PRIMARY_TYPE
  FROM CRIMES
  WHERE PRIMARY_TYPE IN ('BURGLARY', 'ROBBERY', 'MOTOR VEHICLE THEFT')
) AS FILTERED_CRIMES MATCH_RECOGNIZE(
  PARTITION BY BEAT
  ORDER BY TS, ID
  MEASURES
    A.ID AS BURGLARY_ID,
    B.ID AS ROBBERY_ID,
    C.ID AS MOTOR_VEHICLE_THEFT_ID
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO NEXT ROW
  PATTERN(A {- GAP* -} (B | C))
  DEFINE
    A AS A.PRIMARY_TYPE = 'BURGLARY',
    GAP AS GAP.TS - A.TS <= 3600000,
    B AS B.PRIMARY_TYPE = 'ROBBERY' AND B.TS - A.TS <= 3600000,
    C AS C.PRIMARY_TYPE = 'MOTOR VEHICLE THEFT' AND C.TS - A.TS <= 3600000
);

SELECT
    robbery_id,
    battery_id,
    motor_vehicle_theft_id
FROM (
    SELECT
        id AS robbery_id,
        primary_type,
        LAG(primary_type) OVER w AS previous_primary_type,
        LEAD(id) OVER w AS battery_id,
        LEAD(primary_type) OVER w AS battery_primary_type,
        LEAD(ts) OVER w AS battery_ts,
        LEAD(id, 2) OVER w AS motor_vehicle_theft_id,
        LEAD(primary_type, 2) OVER w AS motor_vehicle_theft_primary_type
    FROM CRIMES
    WINDOW w AS (ORDER BY ts, id)
)
WHERE primary_type = 'ROBBERY'
  AND previous_primary_type = 'ASSAULT'
  AND battery_primary_type = 'BATTERY'
  AND battery_ts BETWEEN 1755864000000 AND 1755873000000
  AND motor_vehicle_theft_primary_type = 'MOTOR VEHICLE THEFT';

SELECT
    BEAT,
    ID AS A_ID,
    B_ID,
    C_ID
FROM (
    SELECT
        BEAT,
        ID,
        PRIMARY_TYPE = 'ARSON'
            AND LOCATION_DESCRIPTION = 'RESIDENCE' AS IS_A,
        LEAD(
            PRIMARY_TYPE = 'CRIMINAL SEXUAL ASSAULT'
                AND DOMESTIC = 'true',
            1
        ) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS IS_B,
        LEAD(
            YEAR = 2016
                AND PRIMARY_TYPE = 'BATTERY'
                AND DISTRICT = '003',
            2
        ) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS IS_C,
        LEAD(ID, 1) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS B_ID,
        LEAD(ID, 2) OVER (PARTITION BY BEAT ORDER BY TS, ID) AS C_ID
    FROM CRIMES
) AS SEQUENCES
WHERE IS_A AND IS_B AND IS_C;

SELECT BEAT, A_ID, B_ID, C_ID
FROM (
    SELECT
        BEAT,
        ID AS A_ID,
        LEAD(ID) OVER w AS B_ID,
        LEAD(ID, 2) OVER w AS C_ID,
        PRIMARY_TYPE = 'ASSAULT'
            AND DISTRICT = '003'
            AND ARREST = 'false' AS IS_A,
        LEAD(
            PRIMARY_TYPE = 'BURGLARY'
                AND DISTRICT = '004'
                AND ARREST = 'false'
        ) OVER w AS IS_B,
        LEAD(
            PRIMARY_TYPE = 'CRIMINAL DAMAGE'
                AND WARD = 24,
            2
        ) OVER w AS IS_C
    FROM CRIMES
    WINDOW w AS (PARTITION BY BEAT ORDER BY TS, ID)
)
WHERE IS_A AND IS_B AND IS_C;

SELECT DISTRICT, ID AS A_ID, B_ID, C_ID
FROM (
    SELECT
        DISTRICT,
        ID,
        MATCH_KIND,
        LEAD(ID, 1) OVER W AS B_ID,
        LEAD(ID, 2) OVER W AS C_ID,
        LEAD(MATCH_KIND, 1) OVER W AS B_KIND,
        LEAD(MATCH_KIND, 2) OVER W AS C_KIND
    FROM (
        SELECT
            DISTRICT,
            TS,
            ID,
            CASE
                WHEN PRIMARY_TYPE = 'BATTERY'
                    AND LOCATION_DESCRIPTION = 'STREET' THEN 1
                WHEN PRIMARY_TYPE = 'MOTOR VEHICLE THEFT'
                    AND LOCATION_DESCRIPTION = 'STREET'
                    AND ARREST = 'false' THEN 2
                WHEN PRIMARY_TYPE = 'NARCOTICS'
                    AND LOCATION_DESCRIPTION = 'STREET'
                    AND ARREST = 'true' THEN 3
                ELSE 0
            END AS MATCH_KIND
        FROM CRIMES
    ) S
    WINDOW W AS (PARTITION BY DISTRICT ORDER BY TS, ID)
) S
WHERE MATCH_KIND = 1
  AND B_KIND = 2
  AND C_KIND = 3;
