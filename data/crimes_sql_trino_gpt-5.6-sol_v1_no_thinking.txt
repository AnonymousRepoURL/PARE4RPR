SELECT
  robbery_id,
  battery_id,
  motor_vehicle_theft_id
FROM (
  SELECT
    id AS robbery_id,
    lead(id, 1) OVER (ORDER BY ts, id) AS battery_id,
    lead(id, 2) OVER (ORDER BY ts, id) AS motor_vehicle_theft_id,
    primary_type,
    lag(primary_type, 1) OVER (ORDER BY ts, id) AS previous_primary_type,
    lead(primary_type, 1) OVER (ORDER BY ts, id) AS battery_primary_type,
    lead(ts, 1) OVER (ORDER BY ts, id) AS battery_ts,
    lead(primary_type, 2) OVER (ORDER BY ts, id) AS motor_vehicle_theft_primary_type
  FROM crimes
)
WHERE primary_type = 'ROBBERY'
  AND previous_primary_type = 'ASSAULT'
  AND battery_primary_type = 'BATTERY'
  AND battery_ts BETWEEN 1755864000000 AND 1755873000000
  AND motor_vehicle_theft_primary_type = 'MOTOR VEHICLE THEFT';

  MEASURES CLASSIFIER() AS EVENT_CLASS, RUNNING COUNT(FOLLOWUP.*) AS FOLLOWUPS_SO_FAR, FINAL COUNT(FOLLOWUP.*) AS FOLLOWUP_TOTAL, RUNNING LAST(FOLLOWUP.TS) AS CURRENT_END_TS, FINAL LAST(FOLLOWUP.TS) AS FINAL_END_TS
  ALL ROWS PER MATCH
  AFTER MATCH SKIP PAST LAST ROW
  PATTERN(START_EVENT FOLLOWUP+)
  DEFINE
    START_EVENT AS 1 = 1,
    FOLLOWUP AS PRIMARY_TYPE = PREV(PRIMARY_TYPE) AND TS - START_EVENT.TS <= 86400000
);

SELECT * FROM CRIMES MATCH_RECOGNIZE(
  PARTITION BY BEAT
  ORDER BY TS, ID
  MEASURES FIRST(PROPERTY_EVENT.TS) AS PROPERTY_START_TS, LAST(PROPERTY_EVENT.TS) AS PROPERTY_END_TS, LAST(ARRESTED.TS) AS ARREST_END_TS, COUNT(PROPERTY_EVENT.*) AS PROPERTY_COUNT, COUNT(ARRESTED.*) AS ARREST_COUNT
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO LAST PROPERTY_EVENT
  PATTERN((THEFT_EVENT | DAMAGE_EVENT){2} ARRESTED+)
  SUBSET PROPERTY_EVENT = (THEFT_EVENT, DAMAGE_EVENT)
  DEFINE
    THEFT_EVENT AS PRIMARY_TYPE IN ('THEFT', 'MOTOR VEHICLE THEFT') AND ARREST = 'false',
    DAMAGE_EVENT AS PRIMARY_TYPE = 'CRIMINAL DAMAGE' AND ARREST = 'false',
    ARRESTED AS PRIMARY_TYPE = 'CRIMINAL DAMAGE' AND ARREST = 'true' AND TS > 1640966400000 AND TS - PROPERTY_EVENT.TS <= 86400000
);

SELECT * FROM CRIMES MATCH_RECOGNIZE(
  PARTITION BY DISTRICT
  ORDER BY TS, ID
  MEASURES FIRST(AMBIGUOUS_EVENT.TS) AS START_TS, HOMICIDE_EVENT.TS AS END_TS, COUNT(AMBIGUOUS_EVENT.*) AS AMBIGUOUS_COUNT
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO NEXT ROW
  PATTERN(((COMMON_A | COMMON_B)+)+ HOMICIDE_EVENT)
  SUBSET AMBIGUOUS_EVENT = (COMMON_A, COMMON_B)
  DEFINE
    COMMON_A AS PRIMARY_TYPE IN ('THEFT', 'CRIMINAL DAMAGE', 'BATTERY'),
    COMMON_B AS PRIMARY_TYPE IN ('CRIMINAL DAMAGE', 'BATTERY', 'ROBBERY'),
    HOMICIDE_EVENT AS PRIMARY_TYPE = 'HOMICIDE'
);
