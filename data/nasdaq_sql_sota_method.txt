FROM (SELECT * FROM NASDAQ WHERE SYMBOL IN (SELECT SYMBOL FROM NASDAQ WHERE ((TS BETWEEN 1722441600000000000 AND 1722528000000000000))))
MATCH_RECOGNIZE(
    PARTITION BY SYMBOL
    ORDER BY TS, TRADE_ID
    MEASURES STRT.TS AS START_TS, LAST(DOWN.TS) AS BOTTOM_TS, LAST(UP.TS) AS END_TS, STRT.PRICE AS START_PRICE, MIN(DOWN.PRICE) AS BOTTOM_PRICE, MAX(UP.PRICE) AS END_PRICE
    AFTER MATCH SKIP TO LAST UP
    PATTERN(STRT DOWN+ UP+)
    DEFINE STRT AS TS BETWEEN 1722441600000000000 AND 1722528000000000000, DOWN AS PRICE < PREV(DOWN.PRICE), UP AS PRICE > PREV(UP.PRICE)
);

  ORDER BY TS, TRADE_ID
  MEASURES A.TS AS START_TS, C.TS AS RANGE_TS, E.TS AS END_TS
  ONE ROW PER MATCH
  AFTER MATCH SKIP TO NEXT ROW
  PATTERN(A? B C D E)
  DEFINE
    A AS A.EXCHANGE = 'D' AND A.FEED != 'sip',
    B AS B.EXCHANGE = 'D' AND B.FEED = 'sip',
    C AS PREV(C.TS, 1) >= 1722450837513686408 AND C.TS < 1722515175948760483,
    E AS E.EXCHANGE IN ('Q', 'P', 'Z', 'K') AND D.FEED = 'sip'
);

    DEFINE A AS (PRICE * VOLUME) >= 1000000, V1 AS 1 = 1, C AS (1.1 * PRICE) < FIRST(A.PRICE) AND COUNT(*) <= 15
);
